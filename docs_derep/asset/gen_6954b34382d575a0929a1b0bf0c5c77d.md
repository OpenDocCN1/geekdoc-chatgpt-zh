def generate_data(n_samples):

    """

    创建可复现的圆形二分类样本

    参数：

        n_samples (int): 需要生成的样本个数

    返回：

        X (ndarray): 二维特征集，维度为 (n_samples, 2)

        y (ndarray): 对应标签集，维度为 (n_samples,)

    """

    np.random.seed(0)            # 锁定随机种子，保证结果可复现

    X = []                       # 暂存特征的列表

    y = []                       # 暂存标签的列表

    for i in range(n_samples):

        # 在区间[-5,5)内均匀采样得到 x1、x2

        x1 = np.random.uniform(-5, 5)

        x2 = np.random.uniform(-5, 5)

        if x1 ** 2 + x2 ** 2 < 10:  # 判断点是否落在圆内

            X.append([x1, x2])     # 加入特征列表

            y.append(1)            # 标记为正例

        else:

            X.append([x1, x2])     # 加入特征列表

            y.append(-1)           # 标记为负例

    return np.array(X), np.array(y)  # 转换为数组后返回

# 构造30个样本作为训练集
X_train, Y_train = generate_data(30)

# 构造10个样本作为测试集
X_test, Y_test = generate_data(10)

import numpy as np
import matplotlib.pyplot as plt

# 实现线性核
def linear_kernel(x1, x2):
    """
    计算线性核值
    参数:
        x1: 第一个样本
        x2: 第二个样本
    返回:
        两个样本的点积结果
    """
    return np.dot(x1, x2)

# 实现多项式核
def polynomial_kernel(x1, x2, degree=2):
    """
    计算多项式核值
    参数：
        x1: 第一个样本
        x2: 第二个样本
        degree: 核阶数，默认2
    返回：
        多项式核的计算结果
    """
    return (np.dot(x1, x2) + 1) ** degree

def smo(X, y, C, tol, max_iter, kernel_function):
    """
    序列最小优化求解SVM参数
    参数：
        X: 训练特征
        y: 训练标签
        C: 松弛惩罚系数
        tol: 容差阈值
        max_iter: 最大迭代轮数
        kernel_function: 核函数句柄
    返回：
        alpha: 乘子向量
        b: 偏置
    """
    m = len(X)                                      # 样本总数
    alpha = np.zeros(m)                             # 初始化乘子为零
    b = 0                                           # 初始化偏置
    iter_count = 0                                  # 记录停滞轮数
    while iter_count < max_iter:                    # 循环直到达到最大停滞轮数
        alpha_changed = 0                           # 本轮变化计数
        for i in range(m):                          # 遍历每个样本
            error_i = np.dot(alpha * y, kernel_function(X, X[i])) + b - y[i]  # 计算第i个样本的误差
            if np.any((y[i] * error_i < -tol) & (alpha[i] < C)) or np.any((y[i] * error_i > tol) & (alpha[i] > 0)):  # 判断是否需要更新
                j = np.random.choice([x for x in range(m) if x != i])   # 随机挑选第二个样本
                error_j = np.dot(alpha * y, kernel_function(X, X[j])) + b - y[j]  # 计算第j个样本的误差
                alpha_old_i = alpha[i]  # 保存旧值
                alpha_old_j = alpha[j]  # 保存旧值
                if y[i] != y[j]:        # 标签不同时确定边界
                    L = max(0, alpha[j] - alpha[i])
                    H = min(C, C + alpha[j] - alpha[i])
                else:
                    L = max(0, alpha[i] + alpha[j] - C)
                    H = min(C, alpha[i] + alpha[j])
                if L == H:              # 边界重合则跳过
                    continue
                eta = kernel_function(X[i], X[i]) + kernel_function(X[j], X[j]) - 2 * kernel_function(X[i], X[j])  # 核函数差异
                if eta <= 0:            # eta无效则跳过
                    continue
                alpha[j] = alpha[j] + float(y[j] * (error_i - error_j) / eta)   # 更新alpha_j
                alpha[j] = np.clip(alpha[j], L, H)      # 限制到合法区间
                if abs(alpha[j] - alpha_old_j) < 1e-5:  # 变化太小则跳过
                    continue
                alpha[i] += y[i] * y[j] * (alpha_old_j - alpha[j])      # 协同更新alpha_i
                b1 = b - error_i - y[i] * (alpha[i] - alpha_old_i) * kernel_function(X[i], X[i]) - y[j] * (alpha[j] - alpha_old_j) * kernel_function(X[i], X[j])
                # 计算候选偏置b1
                b2 = b - error_j - y[i] * (alpha[i] - alpha_old_i) * kernel_function(X[i], X[j]) - y[j] * (alpha[j] - alpha_old_j) * kernel_function(X[j], X[j])  # 计算候选偏置b2
                if 0 < alpha[i] < C:  # 根据alpha状态更新b
                    b = b1
                elif 0 < alpha[j] < C:
                    b = b2
                else:
                    b = (b1 + b2) / 2
                alpha_changed += 1  # 记录一次更新
        if alpha_changed == 0:     # 无更新则计数
            iter_count += 1
        else:
            iter_count = 0
    return alpha, b    # 返回最优乘子与偏置
