先看调用方式的变化。完成式接口用 text-davinci-003，温度固定为 0，最大生成 10 token， top_p 1，频率和存在惩罚都关掉，最后取 choices[0] 的 text 去掉首尾换行。聊天式接口则是 gpt-3.5-turbo，messages 里放 user 角色，参数保持一致。两种写法在微调前后对比时都很关键。

def complete(prompt: str, model: str, max_tokens: int) -> str:

    response = openai.Completion.create(

        prompt=prompt,

        temperature=0,

        max_tokens=max_tokens,

        top_p=1,

        frequency_penalty=0,

        presence_penalty=0,

        model=model

    )

    ans = response["choices"][0]["text"].strip(" \n")

    return ans
