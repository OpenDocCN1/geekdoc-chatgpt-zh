def get_data(num_points):

    # Create sine waveforms

    wave_2 = 3.6 * np.sin(np.arange(0, num_points))

    wave_3 = 1.1 * np.sin(np.arange(0, num_points))

    wave_4 = 4.7 * np.sin(np.arange(0, num_points))
