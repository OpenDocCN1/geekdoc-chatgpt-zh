# PyTorch DDP 多 GPU 训练模板

下面是一份可直接运行的多卡数据并行骨架，演示如何在 CIFAR-10 上用 DistributedDataParallel 训练一个轻量卷积模型。网络里先做两次 3×3 卷积配合 2×2 最大池化，把特征图逐步压缩，随后展平接 128 维隐藏层并映射到 10 个类别。进程初始化采用 NCCL 后端，MASTER_ADDR 设为 localhost、端口 12355，各 rank 分别绑定 cuda:{rank}。数据方面使用 DistributedSampler 按世界大小切分样本，配合 Resize、ToTensor 与 0.5 归一化，DataLoader 批大小 128、num_workers 2 并开启 pin_memory。优化选用 Adam 学习率 0.001，损失为 CrossEntropy，训练循环中每轮调用 set_epoch，100 步打印一次均值损失，结束时输出总耗时。

import torch

import torch.nn as nn

import torch.optim as optim

import torchvision

import torchvision.transforms as transforms

import torch.distributed as dist

import torch.multiprocessing as mp

import os

import time

# 定义一个简单的卷积神经网络模型

class SimpleCNN(nn.Module):

    def __init__(self):

        super(SimpleCNN, self).__init__()

        # 卷积层

        # 输入通道为 3，输出通道为 32，卷积核为 3*3

        self.conv1=nn.Conv2d(3, 32, kernel_size=3, padding=1)

        # 输入通道为 32，输出通道为 64，卷积核为 3*3

        self.conv2=nn.Conv2d(32, 64, kernel_size=3, padding=1)

        # 全连接层

        self.fc1=nn.Linear(64*8*8, 128)  # 输入特征为 64*8*8，输出特征为 128

        self.fc2=nn.Linear(128, 10)  # 输入特征为 128，输出特征为 10（类别数）

        # 激活函数

        self.relu=nn.ReLU()

        # 池化层

        self.pool=nn.MaxPool2d(2, 2)  # 2*2 最大池化

    def forward(self, x):

        x=self.relu(self.conv1(x))  # 卷积 1+ReLU

        x=self.pool(x)  # 池化 1

        x=self.relu(self.conv2(x))  # 卷积 2+ReLU

        x=self.pool(x)  # 池化 2

        x=x.view(-1, 64*8*8)  # 展平

        x=self.relu(self.fc1(x))  # 全连接 1+ReLU

        x=self.fc2(x)  # 全连接 2

        return x

# 初始化分布式环境

def setup(rank, world_size):

    os.environ['MASTER_ADDR']='localhost'

    os.environ['MASTER_PORT']='12355'

    # 使用 NCCL 作为后端

    dist.init_process_group("nccl", rank=rank, world_size=world_size)

# 清理分布式环境

def cleanup():

    dist.destroy_process_group()

# 训练函数

def train(rank, world_size, epochs=5):

    print(f"开始训练，进程号：{rank}")

    setup(rank, world_size)

    torch.manual_seed(0)

    # 设置设备

    device=torch.device(f'cuda:{rank}')

    # 定义模型并将其移动到对应的 GPU 上

    model=SimpleCNN().to(device)

    # 实现分布式数据并行训练

    model=nn.parallel.DistributedDataParallel(model, device_ids=[rank])

    # 定义损失函数和优化器

    criterion=nn.CrossEntropyLoss().to(device)

    optimizer=optim.Adam(model.parameters(), lr=0.001)

    # 数据预处理

    transform=transforms.Compose(

        [transforms.Resize((32, 32)),

         transforms.ToTensor(),

         transforms.Normalize((0.5, 0.5, 0.5), (0.5, 0.5, 0.5))])

    # 加载 CIFAR-10 训练集

    trainset=torchvision.datasets.CIFAR10(root='./data', train=True,

                                           download=True, transform=transform)

    train_sampler=torch.utils.data.distributed.DistributedSampler(trainset,

                                                                   num_replicas=world_size,

                                                                   rank=rank)

    trainloader=torch.utils.data.DataLoader(trainset, batch_size=128,

                                             shuffle=False, num_workers=2,

                                             pin_memory=True, sampler=train_sampler)

    # 记录训练时间

    start_time=time.time()

    # 训练过程

    for epoch in range(epochs):

        train_sampler.set_epoch(epoch)

        running_loss=0.0

        for i, data in enumerate(trainloader, 0):

            inputs, labels=data[0].to(device, non_blocking=True), data[1].to(device, non_blocking=True)

            optimizer.zero_grad()

            outputs=model(inputs)

            loss=criterion(outputs, labels)

            loss.backward()

            optimizer.step()

            running_loss += loss.item()

            if i % 100 == 99:  # 每 100 批输出一次

                print(f"[进程 {rank}, Epoch {epoch+1}, Batch {i+1}] 损失: {running_loss/100:.3f}")

                running_loss=0.0

    end_time=time.time()

    total_time=end_time-start_time

    print(f"进程 {rank} 训练完成, 总训练时间: {total_time:.2f}秒")

    cleanup()

# 测试函数

    print(f"开始测试, 进程号: {rank}")

    setup(rank, world_size)

    torch.manual_seed(0)

    # 设置设备
