# Transformer 输出形状与实战代码速览

译码侧的 logits 形状是 `[batch_size * tgt_len, tgt_vocab_size]`。换个说法，就是把整批序列的所有位置摊平成一行，行数是批次乘目标长度，列数是目标词表大小。每个位置都在词表上做一次多分类，最终挑概率最高的词作为预测结果。整体流程上，Transformer 的工作本质就是先走 Encoder 再走 Decoder，最后把结果吐出来。

训练循环采用 1024 个 epoch，借助 tqdm 呈现进度。每一轮把编码输入、解码输入、解码目标全部迁到设备上，前向得到 `outputs: [batch_size * tgt_len, tgt_vocab_size]` 以及三组注意力。损失用交叉熵计算，目标先展平为一维。梯度清零、反向、参数更新后，进度条实时显示当前 epoch 与训练损失，结束后把模型权重保存到指定路径。

for epoch in range(1024):

    pbar = tqdm((loader), total=len(loader))  # 显示进度条

    for enc_inputs, dec_inputs, dec_outputs in pbar:

        enc_inputs, dec_inputs, dec_outputs = enc_inputs.to(device), dec_inputs.to(device), dec_outputs.to(device)

        # outputs: [batch_size * tgt_len, tgt_vocab_size]

        outputs, enc_self_attns, dec_self_attns, dec_enc_attns = model(enc_inputs, dec_inputs)

        loss = criterion(outputs, dec_outputs.view(-1))

        optimizer.zero_grad()

        loss.backward()

        optimizer.step()

        pbar.set_description(f"epoch {epoch + 1} : train loss {loss.item():.6f} ")  # : learn_rate {lr_scheduler.get_last_lr()[0]:.6f}

    torch.save(model.state_dict(), save_path)

class Memory:

    def __init__(self):

        """初始化"""

        self.actions = []  # 行动(共4种)

        self.states = []   # 状态, 由8个数字组成

        self.logprobs = [] # 概率

        self.rewards = []  # 奖励

        self.is_dones = [] # 游戏是否结束: is_terminals?

    def clear_memory(self):

        del self.actions[:]

        del self.states[:]

        del self.logprobs[:]

        del self.rewards[:]

        del self.is_dones[:]

class Action(torch.nn.Module):
