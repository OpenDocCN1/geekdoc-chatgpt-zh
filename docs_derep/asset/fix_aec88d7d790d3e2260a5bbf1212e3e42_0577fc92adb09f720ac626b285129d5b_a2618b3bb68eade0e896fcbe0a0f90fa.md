把给定的原文 {{input}} 规范化成 JSON 返回。返回对象只包含三个键：title 承载标题，content 承载若干文本片段构成的数组，tags 承载主题关键词构成的数组。

######## 示例

{
    "title": "职场女性就该当「乖学生」？",

    "content": [

        "是迎合规则？",

        "保持谦逊？",

        "隐藏锋芒？",

        "都不是！"

    ],

    "tags": [

        "自我觉醒",

        "价值提升",

        "职场真相"

    ]

}

async def main(args: Args) -> Output:
    params = args.params
    input = params.get("input")
    length = len(input)
    # 构建输出对象
    ret: Output = {
        "length" : length
    }
    return ret
