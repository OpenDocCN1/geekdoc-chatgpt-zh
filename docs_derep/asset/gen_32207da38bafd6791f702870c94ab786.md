    def merge_heads(self, x):
        # x 已是注意力权重对 value 加权后的结果
        # 此时维度是 (batch_size, num_head, seq_len, head_features)
        # 先把维度重排为 (batch_size, seq_len, num_head, head_features)
        x = x.permute(0, 2, 1, 3).contiguous()
        # 目标是把头维度和特征维压成一维
        new_x_shape = x.size()[:-2] + (x.size(-2) * x.size(-1),)
        # 把 num_head 与 head_features 合并为 all_head_size
        # 最终形状变为 (batch_size, seq_len, all_head_size)
        return x.view(*new_x_shape)

    def split_heads(self, x, k=False):
        # 输入一般是 query / key / value，目标形状 (batch_size, seq_len, num_head, head_features)
        new_x_shape = x.size()[:-1] + (self.n_head, x.size(-1) // self.n_head)
        x = x.view(*new_x_shape)
        # key 需要转成 (batch_size, num_head, head_features, seq_len) 方便与 query 做点积
        if k:
            return x.permute(0, 2, 3, 1)
        # query / value 转成 (batch_size, num_head, seq_len, head_features)
        else:
            return x.permute(0, 2, 1, 3)
