    # 校验输入类型：boxes 需为列表或元组，rel_codes 需为张量

    assert isinstance(rel_codes, torch.Tensor)

def decode_single(self, rel_codes, boxes):

    """
    解码过程：依据参考框坐标与偏移量，推算出预测框位置
    """

    # 统一数据类型，避免精度不匹配
    boxes = boxes.to(rel_codes.dtype)

    # 计算参考框的宽高与中心点
    widths = boxes[:, :, 2] - boxes[:, :, 0]
    heights = boxes[:, :, 3] - boxes[:, :, 1]
    ctr_x = boxes[:, :, 0] + 0.5 * widths
    ctr_y = boxes[:, :, 1] + 0.5 * heights

    # 取出编码权重
    wx, wy, ww, wh = self.weights

    # 将归一化偏移还原为绝对偏移
    dx = rel_codes[:, :, 0:4] / wx
    dy = rel_codes[:, :, 1:4] / wy
    dw = rel_codes[:, :, 2:4] / ww
    dh = rel_codes[:, :, 3:4] / wh

    # 对宽高偏移做上界钳制，防止预测框过大
    # 后续仍需 filter_proposals 剔除越界框
    dw = torch.clamp(dw, max=self.bbox_xform_clip)
    dh = torch.clamp(dh, max=self.bbox_xform_clip)

    # 计算预测框中心点
    pred_ctr_x = dx * widths[:, :, None] + ctr_x[:, :, None]
    pred_ctr_y = dy * heights[:, :, None] + ctr_y[:, :, None]

    # 计算预测框宽高
    pred_w = torch.exp(dw) * widths[:, :, None]
    pred_h = torch.exp(dh) * heights[:, :, None]

    # 转换为 (x1, y1, x2, y2) 格式
    pred_boxes1 = pred_ctr_x - 0.5 * pred_w
    pred_boxes2 = pred_ctr_y - 0.5 * pred_h
    pred_boxes3 = pred_ctr_x + 0.5 * pred_w
    pred_boxes4 = pred_ctr_y + 0.5 * pred_h

    # 堆叠并展平最后两维
    pred_boxes = torch.stack((pred_boxes1, pred_boxes2, pred_boxes3,
        pred_boxes4), dim=2).flatten(1)

return pred_boxes

@torch.jit.script
def encode_boxes(reference_boxes, proposals, weights):

    """
    编码过程：计算真实框相对于锚框的归一化偏移量
    """

    # 权重向量在 __init__ 时确定，用于平衡回归损失与分类损失量级
    wx, wy, ww, wh = weights[0], weights[1], weights[2], weights[3]

    # 将坐标拆分为独立列向量，便于广播运算
    prop_x1 = proposals[:, 0].unsqueeze(1)
    prop_y1 = proposals[:, 1].unsqueeze(1)
    prop_x2 = proposals[:, 2].unsqueeze(1)
    prop_y2 = proposals[:, 3].unsqueeze(1)

    ref_x1 = reference_boxes[:, 0].unsqueeze(1)
    ref_y1 = reference_boxes[:, 1].unsqueeze(1)
    ref_x2 = reference_boxes[:, 2].unsqueeze(1)
    ref_y2 = reference_boxes[:, 3].unsqueeze(1)

    # 锚框几何属性
    ex_widths = prop_x2 - prop_x1
    ex_heights = prop_y2 - prop_y1
    ex_ctr_x = prop_x1 + 0.5 * ex_widths
    ex_ctr_y = prop_y1 + 0.5 * ex_heights

    # 真实框几何属性
    gt_widths = ref_x2 - ref_x1
    gt_heights = ref_y2 - ref_y1
    gt_ctr_x = ref_x1 + 0.5 * gt_widths
    gt_ctr_y = ref_y1 + 0.5 * gt_heights

    # 计算四个维度的编码目标
    targets_dx = wx * (gt_ctr_x - ex_ctr_x) / ex_widths
    targets_dy = wy * (gt_ctr_y - ex_ctr_y) / ex_heights
    targets_dw = ww * torch.log(gt_widths / ex_widths)
    targets_dh = wh * torch.log(gt_heights / ex_heights)

    # 拼接为 (N, 4) 张量返回
    targets = torch.cat((targets_dx, targets_dy, targets_dw, targets_dh), dim=1)
    return targets

def filter_proposals(self, proposals, objectness, image_shapes, num_anchors_per_level):
    """
    候选框筛选流程：
    1. 按置信度排序并截取 Top-K
    2. 裁剪越界框
    3. 移除过小框
    4. 阈值过滤低置信度框
    5. 分层级执行 NMS
    6. 最终截取 Post-NMS Top-K
    """

    num_images = proposals.shape[0]
    device = proposals.device

    # 分离计算图，reshape 为 (B, 总锚框数)
    objectness = objectness.detach().reshape(num_images, -1)

    # 构建层级索引张量，形状 (1, 总锚框数) -> (B, 总锚框数)
    level_indices = torch.cat([
        torch.full((n,), idx, dtype=torch.int64, device=device)
        for idx, n in enumerate(num_anchors_per_level)
    ], 0).reshape(1, -1).expand_as(objectness)

    # 分层级取 Top-K 索引
    top_n_idx = self._get_top_n_idx(objectness, num_anchors_per_level)

    # 批次索引广播
    batch_idx = torch.arange(num_images, device=device)[:, None]

    # 按索引采样
    objectness = objectness[batch_idx, top_n_idx]
    level_indices = level_indices[batch_idx, top_n_idx]
    proposals = proposals[batch_idx, top_n_idx]

    final_scores = []
    final_boxes = []

    # 逐图像后处理
    for boxes, scores, lvl, img_shape in zip(proposals, objectness, level_indices, image_shapes):
        # 1. 裁剪至图像边界
        boxes = box_ops.clip_boxes_to_image(boxes, img_shape)

        # 2. 剔除面积过小框
        keep = box_ops.remove_small_boxes(boxes, self.min_size)
        boxes, scores, lvl = boxes[keep], scores[keep], lvl[keep]

        # 3. 置信度阈值过滤
        keep = torch.where(scores >= self.score_thresh)[0]
        boxes, scores, lvl = boxes[keep], scores[keep], lvl[keep]

        # 4. 分层级 NMS
        keep = box_ops.batched_nms(boxes, scores, lvl, self.nms_thresh)

        # 5. 最终 Top-K 截断
        keep = keep[:self.post_nms_top_n()]

        boxes, scores = boxes[keep], scores[keep]
        final_boxes.append(boxes)
        final_scores.append(scores)

    return final_boxes, final_scores

def _get_top_n_idx(self, objectness, num_anchors_per_level):
    """
    分层级选取预置信度最高的 pre_nms_top_n 个索引
    """
    indices_list = []
    offset = 0

    # 按层级切分 objectness
    for ob in objectness.split(num_anchors_per_level, 1):
        # ob: (B, H×W×A)
        num_anchors = ob.shape[1]
        k = min(self.pre_nms_top_n(), num_anchors)

        # 降序 Top-K，返回索引
        _, top_k_idx = ob.topk(k, dim=1)
        indices_list.append(top_k_idx + offset)
        offset += num_anchors

    return torch.cat(indices_list, dim=1)
