def build_index(self, documents: List[str]):
    """构建倒排索引。"""
    for doc_id, content in enumerate(documents):
        for word in content.split():
            self.index[word].append(doc_id)

def search(self, query: str) -> List[int]:
    """在倒排索引中查询关键词。"""
    return self.index.get(query, [])
