async def run_client():

    print("=== 客户端等待 5 秒后开始连接 ... ===")

    await asyncio.sleep(5)

    # MCP stdio transport

    from mcp.client.stdio import StdioServerParameters, stdio_client

    server_params = StdioServerParameters(

        command="python",

        args=[os.path.abspath(__file__)]

    )

    async with stdio_client(server_params) as (read, write):

        async with ClientSession(read, write) as session:

            print("[Client] Finance QA session init...")

            await session.initialize()

            # Step1: 查询金融术语

            glossary_res = await session.call_tool("tool_finance_glossary", {"term": "Yield Curve"})

            print("[Client] 金融术语查询:", glossary_res)

            # Step2: 模拟某产品收益

            product_sim = await session.call_tool("tool_product_simulation", {

                "product_id": "FundA",

                "principal": 10000,

                "years": 2

            })

            print("[Client] 产品收益模拟:", product_sim)

            # Step3: 查看用户账户

            account_view = await session.call_tool("tool_check_account", {"user_id": "U3002"})

            print("[Client] 用户账户信息:", account_view)

            # Step4: 风险评估问卷

            answers = ["中", "longterm"]

            risk_eval = await session.call_tool("tool_risk_survey", {

                "answers": answers

            })

            print("[Client] 风险偏好:", risk_eval)

            # Step5: 列出可投产品

            prod_list = await session.call_tool("tool_list_products", {})

            print("[Client] 可投产品列表:", prod_list)

async def main():

    import asyncio

    server_task = asyncio.create_task(run_server())

    client_task = asyncio.create_task(run_client())

    await asyncio.gather(server_task, client_task)

if __name__ == "__main__":

    asyncio.run(main())

#### 10.1.3 工作流平台上的 MCP 实践方案

在大模型与智能体协同演进的背景下，编排复杂任务、支撑多智能体协作的工作流平台成为关键基础设施。MCP 作为标准化语境协议，提供语境建模、工具调用、资源调度与语义封装等能力，为搭建稳定、可扩展、可控的智能体协作框架提供统一交互与编排底座。基于该协议的平台目标是打通智能体间的语义壁垒，实现任务驱动下的信息持续流转、状态可追踪、操作可原子化执行，尤其适用于企业高并发、多轮交互与异步任务链的管控场景。

在架构层面，每个智能体被视作 MCP 下的能力体 Capability Entity，生命周期由语境槽 Slot 承载，调度则依靠 MCP 工具系统 Tool API 触发。工作流引擎在此之上负责任务分派、状态维护与路径规划，执行时智能体响应 ToolCall 指令，完成能力落地并回写反馈。为支持多阶段链路，MCP 允许状态注入，例如用 Slot 记录当前阶段、任务状态及中间产出，从而保持语义一致与语境同步的可靠性。

任务通常由业务侧发起，平台通过 MCP 客户端发出构建请求，服务端依据已注册智能体的能力声明 Capabilities 匹配合适的执行体。这不是传统 RPC 的指令式调用，而是一种语义化消息触发，天然带有信息传递、资源注入与状态持久化的特点。一个任务可拆解为多个有序或并发的步骤，每个步骤由一个或多个智能体承担，工具调用机制让这些步骤以 Tool 形式声明，并能热插拔式接入工作流。

平台内置 Slot 生命周期管理，用于动态维护执行过程中产生的语境，例如初始请求、智能体中间产出、最终结构化结果，都通过 Slot 的写入与更新实现跨智能体共享与边界隔离。再加上 MCP 标准中的错误治理能力，如错误分级、异常恢复、语境重试与回滚，使得生产环境下的智能体协同更可控、可观测。
