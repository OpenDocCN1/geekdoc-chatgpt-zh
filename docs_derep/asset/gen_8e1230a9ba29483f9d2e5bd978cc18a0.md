# 分词策略决定模型上限：词级 vs 子词实战拆解

在自然语言处理里，分词方式直接影响模型能不能读懂文本。粗暴地按词切，表达清晰但一碰生词就卡壳；用子词切，能把未登录词拆开，词汇表压力小得多，理解力自然上去。词级方案更适合任务明确、词表稳定的场景，子词方案则在开放域模型里几乎成了标配，尤其 BERT 这类预训练模型默认就靠子词来消化海量词汇。

下面用三句典型中文做对比演示，先看最简单的词级切分，再看基于 Hugging Face 的子词切分，接着用手写的 BPE 和官方 WordPiece 做进一步对照，从输出就能看出粒度差异。

import re

from transformers import AutoTokenizer

# 示例文本

texts=["自然语言处理的分词方法可以提升模型的理解能力。",

       "词级分词与子词分词适用于不同的NLP任务。",

       "BERT模型常用子词分词以处理大量词汇。" ]

# 词级分词实现

def word_level_tokenization(text):

    """

    实现简单的词级分词

    参数:

    -text: str, 输入的文本

    返回:

    -tokens: list of str, 词级分词结果

    """

    # 使用空格和标点符号分隔文本

    tokens=re.findall(r"\b\w+\b", text)

    return tokens

# 使用Hugging Face的Tokenizer进行子词分词

model_name="bert-base-chinese"

tokenizer=AutoTokenizer.from_pretrained(model_name)

def subword_level_tokenization(text, tokenizer):

    """

    使用子词分词将文本分解为子词单元

    参数:

    -text: str, 输入的文本

    -tokenizer: 预训练模型的分词器

    返回:

    -subwords: list of str, 子词分词结果

    """

    # 通过Tokenizer进行分词

    tokenized=tokenizer.tokenize(text)

    return tokenized

# 执行词级分词和子词分词

for text in texts:

    word_tokens=word_level_tokenization(text)

    subword_tokens=subword_level_tokenization(text, tokenizer)

    print(f"原始文本: {text}")

    print(f"词级分词结果: {word_tokens}")

    print(f"子词分词结果: {subword_tokens}\n")

词级实现在纯中文里几乎把整句当成一个单元拿下，而 bert-base-chinese 的子词器会把句子拆成更细的片段，英文缩写 NLP、BERT 更是被拆到字符/子串级别，这正是子词对罕见词和跨语言符号的容错能力。

import re

from collections import Counter, defaultdict

# 初始化词汇表和示例文本

vocab={"自然语言", "处理", "的", "分词", "方法", "可以",

       "提升", "模型", "理解", "能力"}

texts=[

    "自然语言处理的分词方法可以提升模型的理解能力。",]

# 构建BPE分词表

def build_bpe_vocab(vocab, num_merges=5):

    """

    构建BPE分词词汇表

    参数:

    -vocab: set, 初始词汇表

    -num_merges: int, 合并次数

    返回:

    -bpe_vocab: dict, BPE词汇表

    """

    bpe_vocab={word: list(word)+["</w>"] for word in vocab}

    for i in range(num_merges):

        # 统计词对的出现频率

        pairs=defaultdict(int)

        for word, tokens in bpe_vocab.items():

            for j in range(len(tokens)-1):

                pairs[(tokens[j], tokens[j+1])] += 1

        # 找到出现频率最高的词对

        if not pairs:

            break

        best_pair=max(pairs, key=pairs.get)

        # 合并词对

        for word, tokens in bpe_vocab.items():

            j=0

            while j < len(tokens)-1:

                if tokens[j]==best_pair[0] and tokens[j+1]==best_pair[1]:

                    tokens[j:j+2]=["".join(best_pair)]

                j += 1

    return bpe_vocab

bpe_vocab=build_bpe_vocab(vocab)                # 执行BPE分词

# 对文本进行BPE编码

def bpe_encode(text, bpe_vocab):

    """

    使用BPE分词对文本编码

    参数:

    -text: str, 输入的文本

    -bpe_vocab: dict, BPE词汇表

    返回:

    -encoded_text: list, 编码后的子词列表

    """

    tokens=list(text)

    encoded_text=[]

    while tokens:

        for i in range(len(tokens), 0, -1):

            subword="".join(tokens[:i])

            if subword in bpe_vocab:

                encoded_text.append(subword)

                tokens=tokens[i:]

                break

        else:

            encoded_text.append(tokens[0])

            tokens=tokens[1:]

    return encoded_text

# 对示例文本进行BPE编码

for text in texts:

    encoded_text=bpe_encode(text, bpe_vocab)

    print(f"原始文本: {text}")

    print(f"BPE编码结果: {encoded_text}\n")

手写的 BPE 在小词汇表下已经能把“自然语言”“分词”等高频词保留下来，同时把未见的标点单独处理。接下来用真实的 WordPiece 分词器看看工业级效果。

BPE 和 WordPiece 在思想上很接近，都是贪心合并，但 WordPiece 在训练时带概率选择，中文里会倾向保留“自然”“语言”等完整词，同时对“可”“以”“提”“升”这类常见偏旁做细化。两者对比能帮助你判断在自己的语料上该偏向更保守的词级，还是更灵活的子词。
