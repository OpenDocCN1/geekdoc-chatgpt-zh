def register_user(username, password):

    """

    注册新用户

    :param username: 用户名

    :param password: 密码

    :return: 注册结果

    """

    # 检查用户名是否已存在

    if username_exists(username):

        return "用户名已存在，请选择其他用户名。"

    # 加密密码

    hashed_password=hash_password(password)

    # 将用户信息存储到数据库

    save_user_to_db(username, hashed_password)

    return "注册成功！"
