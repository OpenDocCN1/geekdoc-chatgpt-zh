# 重算换空间：checkpoint 的内存收益实测

在大模型训练或推理里，碰到的第一道瓶颈常常是显存而非算力。PyTorch 的 checkpoint 机制提供了一种显式交换：在反向阶段主动丢弃部分中间激活，需要梯度时再把前向重新跑一遍。代价是时间，收益是峰值占用下降。下面用两组对比把这种取舍量化出来，并附一个可直接复用的骨架。

第一组聚焦推理。卷积网络的两个阶段分别封装，基准实现走完整图，重算版本把 layer1、layer2 包进 checkpoint。测试用 16 张 224×224 输入，连续 10 次取峰值占用与总时长，测量前后都清空缓存并重置统计，保证两次环境一致。实测中推理阶段的内存降幅明显，耗时只小幅上浮。

class CheckpointedCNN(nn.Module):

    def __init__(self):

        super(CheckpointedCNN,self).__init__()

        self.layer1=nn.Sequential(

            nn.Conv2d(3,64,kernel_size=3,padding=1),

            nn.ReLU(),

            nn.Conv2d(64,64,kernel_size=3,padding=1),

            nn.ReLU(),

            nn.MaxPool2d(2)

        )

        self.layer2=nn.Sequential(

            nn.Conv2d(64,128,kernel_size=3,padding=1),

            nn.ReLU(),

            nn.Conv2d(128,128,kernel_size=3,padding=1),

            nn.ReLU(),

            nn.MaxPool2d(2)

        )

        self.fc=nn.Linear(128*56*56,1000)

    def forward(self,x):

        # 使用检查点函数来节省显存

import torch

import torch.nn as nn

import torch.utils.checkpoint as checkpoint

import time

# 定义一个深层神经网络模型

class DeepNet(nn.Module):

    def __init__(self):

        super(DeepNet,self).__init__()

        self.layers=nn.Sequential(

            nn.Linear(1024,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU(),

            nn.Linear(2048,1024),

            nn.ReLU(),

            nn.Linear(1024,10)

        )

    def forward(self,x):

        return self.layers(x)

# 定义使用梯度检查点的深层神经网络模型

class CheckpointedDeepNet(nn.Module):

    def __init__(self):

        super(CheckpointedDeepNet,self).__init__()

        self.layer1=nn.Sequential(

            nn.Linear(1024,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU()

        )

        self.layer2=nn.Sequential(

            nn.Linear(2048,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU()

        )

        self.layer3=nn.Sequential(

            nn.Linear(2048,1024),

            nn.ReLU(),

            nn.Linear(1024,10)

        )

    def forward(self,x):

        x=checkpoint.checkpoint(self.layer1,x)

        x=checkpoint.checkpoint(self.layer2,x)

        x=self.layer3(x)

        return x

        optimizer.zero_grad()

        output=model(input)

        loss=criterion(output,target)

        loss.backward()

        optimizer.step()
