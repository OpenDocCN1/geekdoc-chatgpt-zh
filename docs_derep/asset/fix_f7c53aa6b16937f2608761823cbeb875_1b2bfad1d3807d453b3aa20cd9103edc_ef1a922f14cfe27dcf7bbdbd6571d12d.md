###### 定义一个名为 'factorial' 的函数，它接收一个参数 n
def factorial(n):
    # 如果 n 等于 0
    if n == 0:
        # 返回 1，因为 0 的阶乘在数学定义上就是 1
        return 1
    else:

        # 这就是递归的过程，函数调用了自身，每次调用时 n 都减 1，直到 n 等于 0 时返回 1，结束递归
        return n * factorial(n - 1)
