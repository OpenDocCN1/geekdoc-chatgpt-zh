# 行为调度内核：抢占式裁决与共存契约

调度循环的心跳由两大机制驱动：**强制清场**与**兼容准入**。当高优指令触达，现有执行集瞬间清零，新指令独占跑道并刷新优先级水位线；常规指令则须接受“全员互斥”审查——唯有与全部在场任务零冲突，方能挤入活跃池并同步抬高门槛。维护环节则执行“存亡过滤”：遍历池中实体，终结者自动剔除，存续者继续推进。

class ActionCombination extends Action:

    # 保存子动作

    actions: Action[]

    function interrupt() -> bool:

        # 如果有任何子动作可以中断，则中断该动作

        for action in actions:

            if action.interrupt():

                return true

        return false

    function canDoBoth(other: Action) -> bool:

        # 如果所有子动作都可以执行，则执行组合动作

        for action in actions:

            if not action.canDoBoth(other):

                return false

        return true

    function isComplete() -> bool:

        # 如果所有子动作都已完成，则可以认为动作已经完成

        for action in actions:

            if not action.isComplete():

                return false

        return true

    function execute():

        # 执行所有子动作

        for action in actions:

            action.execute()

function interrupt() -> bool:

    # 如果第一个子动作可以中断，则中断它

    return actions[0].interrupt()

function canDoBoth(other: Action) -> bool:

    # 如果所有子动作都可以执行，则执行组合动作

    # 如果仅测试了第一个，则可能会出现一种危险

    # 即在序列中间突然发现不兼容的情况

    for action in actions:

        if not action.canDoBoth(other):

            return False

    return true

function isComplete() -> bool:

    # 如果所有子动作都已完成，则可以认为动作已经完成

    return activeIndex >= len(actions)

function execute():

    # 执行当前动作

    actions[activeIndex].execute()

    # 如果当前动作已经完成，则跳转到下一个动作

    if actions[activeIndex].isComplete():

        activeIndex += 1
