#### 4. 破解内容分级：携带 Cookie 绕过登录验证

不少网站设有内容分级机制，强制要求登录后才能浏览。以编程论坛的搜索页面（https://bbs.bccn.net/search.php）为例，界面如下图。若未填用户名直接点击“搜索”，系统会弹出提示框，要求会员登录或注册后方可进入。

该论坛靠 Cookie 判断会员身份，因此我们可以在 `requests` 请求头里塞入 Cookie 来跳过验证拦截。获取 Cookie 的操作流程如下：

1.  先注册一个账号并完成登录。
2.  按 `F12` 打开开发者工具，切到 **Network（网络）** 面板。
3.  按 `F5` 刷新页面，捕获全部网络请求记录。
4.  在列表中点选 `search.php`，查看右侧 **Headers → General** 下的 **Request URL**，并复制完整 Cookie 字符串，参考下图。

import time

def web_scraping_bot(urls):

    eng_words = []

    for url in urls:

        file = url.split("/")[-1]

        print("获取: " + file + " 网络数据中...")

        r = get_resource(url)

        if r.status_code == requests.codes.ok:

            soup = parse_html(r.text)

            words = get_words(soup, file)

            eng_words = eng_words + words

            print("等待 5 秒钟...")

            time.sleep(5)

        else:

            print("HTTP 请求错误...")

    return eng_words

def get_words(soup, file):

    words = []

    count = 0

    for wordlist_table in soup.find_all(class_="wordlist"):

        count += 1

        for word_entry in wordlist_table.find_all("tr"):

            new_word = []

            new_word.append(file)

            new_word.append(str(count))

            new_word.append(word_entry.th.text)

            new_word.append(word_entry.td.text)

            words.append(new_word)

    return words
