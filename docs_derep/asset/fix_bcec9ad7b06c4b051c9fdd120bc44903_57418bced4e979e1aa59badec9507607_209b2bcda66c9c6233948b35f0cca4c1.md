# 初始化 dv3 环境

用 conda 生成一个独立的工作区，把 Python 固定在 3.11 以避免依赖冲突。
conda create --name dv3 python=3.11

对 conda 做一次初始化，确保后续命令可被识别。
conda init

让 shell 重新读取配置文件，使初始化生效。
source ~/.bashrc

切到新环境，让当前会话使用 dv3 的 Python。
conda activate dv3

# 配置 Jupyter 内核

先装 JupyterLab 作为交互界面。
conda install jupyterlab

再装 ipykernel，以便在 notebook 中调用当前环境。
conda install ipykernel

把 dv3 注册为 Jupyter 可用内核，显示名称设为 Python (dv3)。
python -m ipykernel install --user --name dv3 --display-name "Python (dv3)"

# 安装模型管理库

modelscope 用于模型下载与加载，先安装工具链。
pip install modelscope

# 本地缓存并下载量化模型

新建本地目录用于存放权重。
mkdir ./deepseek

使用 modelscope 将指定的 int4 量化模型拉到本地目录。
modelscope download --model OPEA/DeepSeek-V3-int4-sym-gptq-inc --local_dir ./deepseek

import torch

from modelscope import AutoTokenizer, AutoModelForCausalLM, GenerationConfig

# 指定模型存储路径（修改为上一步中模型权重下载到的本地路径）

model_name = "./deepseek"

# 加载对应的分词器（Tokenizer）

tokenizer = AutoTokenizer.from_pretrained(model_name)

# 加载 CausalLM（因果语言模型），设置数据类型为 bfloat16，并自动映射到可用设备（如 GPU）

model = AutoModelForCausalLM.from_pretrained(model_name, torch_dtype=torch.bfloat16, device_map="auto")

# 加载生成配置（GenerationConfig），用于调整模型的生成策略

model.generation_config = GenerationConfig.from_pretrained(model_name)

model.generation_config.pad_token_id = model.generation_config.eos_token_id
