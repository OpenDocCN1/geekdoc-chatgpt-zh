# 4. 添加 Docker 官方源
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# 5. 安装 Docker 核心组件
sudo apt install -y docker-ce docker-ce-cli containerd.io

# 6. 验证 Docker 运行状态
sudo systemctl status docker

# 7. 授予普通用户 Docker 使用权限
sudo usermod -aG docker $USER

# 8. 启用开机自启
sudo systemctl enable docker

**注意：** 执行完上述操作后，需退出当前终端并重新登录，普通用户的组权限才能生效。

2. Docker Compose 安装
Ubuntu 默认源内的 Compose 版本通常滞后，兼容性不佳，建议直接从官方发布页获取最新稳定版，具体步骤如下：

# 1. 获取最新版 Docker Compose
sudo curl -L "https://github.com/docker/compose/releases/latest/download/docker-compose-$(uname -s)-$(uname -m)" -o /usr/local/bin/docker-compose

# 2. 赋予执行权限
sudo chmod +x /usr/local/bin/docker-compose

# 3. 确认安装结果
docker-compose --version

在部署 OpenClaw 之前，需要先把官方镜像拉取到本机。镜像地址为 ghcr.io/openclaw/openclaw:latest，latest 标签指向当前稳定版本。执行 docker pull ghcr.io/openclaw/openclaw:latest 后，Docker 引擎会从 GitHub 仓库将镜像下载到本地服务器。

容器启动前要先明确数据卷与目录映射的关系。Docker 容器默认是无状态的，停止或删除后容器内部产生的所有文件都会被丢弃，配置、日志等一并丢失。OpenClaw 的关键资产例如 openclaw.json 配置、会话历史、已安装的技能插件都需长期保留，因此要把宿主机目录映射到容器内部。官方说明中，容器内的数据统一放在 /root/.openclaw。

### 8.4.3 编写 Docker Compose 文件
直接使用一条冗长的 docker run 命令也可以启动容器，但参数较多且维护困难。实际场景更推荐使用 Docker Compose，通过结构清晰的 YAML 文件统一管理各项配置。

在服务器上新建一个用于存放 OpenClaw 数据的目录，并在其中创建 docker-compose.yml。文件中定义一个名为 openclaw-gateway 的服务，指定官方镜像，并配置 restart: unless-stopped 策略，确保异常或重启后可自动恢复。

端口方面将容器内部的 3000 端口映射到宿主机的 3000 端口，以便外部访问。数据卷是核心，将宿主机当前目录下的 ./data 文件夹挂载到容器内部的 /root/.openclaw，实现持久化保存。

还可以通过 environment 字段注入环境变量，例如设置时区为 Asia/Shenzhen，或将大模型 API Key 以变量形式传入，避免明文写入配置。最后使用 command: ["gateway"] 指定启动网关服务。

OpenClaw 官网提供的标准 Docker Compose 配置如下：

services:
  openclaw-gateway:
    image: ${OPENCLAW_IMAGE:-openclaw:local}
    environment:
      HOME: /home/node
      TERM: xterm-256color
      OPENCLAW_GATEWAY_TOKEN: ${OPENCLAW_GATEWAY_TOKEN:-}
      OPENCLAW_ALLOW_INSECURE_PRIVATE_WS: ${OPENCLAW_ALLOW_INSECURE_PRIVATE_WS:-}
      CLAUDE_AI_SESSION_KEY: ${CLAUDE_AI_SESSION_KEY:-}
      CLAUDE_WEB_SESSION_KEY: ${CLAUDE_WEB_SESSION_KEY:-}
      CLAUDE_WEB_COOKIE: ${CLAUDE_WEB_COOKIE:-}
      TZ: ${OPENCLAW_TZ:-UTC}
    volumes:
      - ${OPENCLAW_CONFIG_DIR}:/home/node/.openclaw
      - ${OPENCLAW_WORKSPACE_DIR}:/home/node/.openclaw/workspace
    ports:
      - "${OPENCLAW_GATEWAY_PORT:-18789}:18789"
      - "${OPENCLAW_BRIDGE_PORT:-18790}:18790"
    init: true
    restart: unless-stopped
    command:
      [
        "node",
        "dist/index.js",
        "gateway",
        "--bind",
        "${OPENCLAW_GATEWAY_BIND:-lan}",
        "--port",
        "18789",
      ]
    healthcheck:
      test:
        [
          "CMD",
          "node",
          "-e",
          "fetch('http://127.0.0.1:18789/healthz').then((r)=>process.exit (r.ok?0:1)).catch(()=>process.exit(1))",
        ]
      interval: 30s
      timeout: 5s
      retries: 5
      start_period: 20s
  openclaw-cli:
    image: ${OPENCLAW_IMAGE:-openclaw:local}
    network_mode: "service:openclaw-gateway"
    cap_drop:
      - NET_RAW
      - NET_ADMIN
    security_opt:
      - no-new-privileges:true
    environment:
      HOME: /home/node
      TERM: xterm-256color
      OPENCLAW_GATEWAY_TOKEN: ${OPENCLAW_GATEWAY_TOKEN:-}
      OPENCLAW_ALLOW_INSECURE_PRIVATE_WS: ${OPENCLAW_ALLOW_INSECURE_PRIVATE_WS:-}
      BROWSER: echo
      CLAUDE_AI_SESSION_KEY: ${CLAUDE_AI_SESSION_KEY:-}
      CLAUDE_WEB_SESSION_KEY: ${CLAUDE_WEB_SESSION_KEY:-}
      CLAUDE_WEB_COOKIE: ${CLAUDE_WEB_COOKIE:-}
      TZ: ${OPENCLAW_TZ:-UTC}
    volumes:
      - ${OPENCLAW_CONFIG_DIR}:/home/node/.openclaw
      - ${OPENCLAW_WORKSPACE_DIR}:/home/node/.openclaw/workspace
    stdin_open: true
    tty: true
    init: true
    entrypoint: ["node", "dist/index.js"]
    depends_on:
      - openclaw-gateway

### 8.4.4 启动与管理服务
配置文件完成后，进入 docker-compose.yml 所在目录执行 docker-compose up -d，即可后台守护模式启动 OpenClaw 容器。

运行期间如需排查问题，可执行 docker-compose logs -f 实时查看日志。若要进行初始化配置或安装插件，不能在宿主机直接执行，需先用 docker exec -it openclaw-gateway /bin/bash 进入运行中的容器，再在容器内执行 openclaw configure 等命令。

版本更新过程很平滑：在同一目录先执行 docker-compose pull 拉取新镜像，再执行 docker-compose up -d 用新镜像重建容器。借助数据卷挂载，./data 中的配置文件和历史数据会完整保留并自动接入新版本。

## 9.1 集成飞书
飞书由字节跳动打造，以流畅体验和开放平台能力著称。OpenClaw 对飞书提供原生支持，核心在于采用飞书开放平台的 WebSocket 长连接事件订阅，而非传统 Webhook。这样无需在服务器暴露公网 URL，也省去反向代理与 SSL 配置，安全性更高。

### 9.1.1 创建飞书企业应用与机器人
首先在飞书开放平台创建自建应用作为身份凭证。登录后进入开发者后台，点击创建企业自建应用，填写名称如“OpenClaw 智能助手”、描述并上传图标。

创建成功后进入详情页，在凭证与基础信息中获取两个关键参数：
• App ID，通常以 cli_ 开头；
• App Secret，应用密钥。
这两个值相当于机器人的账号密码，后续校验身份时会用到，需妥善保存。

随后在添加应用功能中打开机器人开关，机器人即成为 OpenClaw 在飞书里的交互入口。

### 9.1.2 配置权限与事件订阅
创建应用后需申请 API 权限。在开发配置 → 权限管理 → 应用身份权限中，可批量导入所需权限。以下 JSON 涵盖机器人运行的核心范围：

{
  "scopes": {
    "tenant": [
      "im:chat.members:bot_access",
      "im:message",
      "im:message.group_at_msg:readonly",
      "im:message.p2p_msg:readonly",
      "im:message:readonly",
      "im:message:send_as_bot",
      "im:resource"

      "contact:contact.base:readonly"
    ]
  }
}

其中 user 下的 contact:contact.base:readonly 用于只读访问企业通讯录基础信息，是机器人识别用户身份、完成上下文匹配的前提。若未开通，用户发起指令时机器人将因缺少数据而失效。
