## 朴素子串定位
先做合法性检查，再用两个下标同步走。不相等时让主串指针回退到下一个可能起点，继续尝试。成功则返回首次出现位置，失败返回 -1。样例在 xyzsabcd 中找 abc。

[PRE1]
"""

**** 方法功能：判断 p 是否是 s 的子串，如果是返回 p 在 s 中第一次出现的下标，否则返回-1

**** 输入参数：s 和 p 分别为主串和模式串

"""

def match(s,p):

    # 检查参数是否合理

    if s == None or p == None:

        print("参数不合理")

        return -1

    slength = len(s)

    plength = len(p)

    # p 肯定不是 s 的子串

    if slength < plength:

        return -1

    i = 0

    j = 0

    while i < slength and j < plength:

        if list(s)[i] == list(p)[j]:

            # 如果相同，那么继续比较后面的字符

            i += 1

            j += 1

        else:

            # 后退回去重新比较

            i = i - j + 1

            j = 0

            if i > (slength - plength):

                return -1

    if j >= plength:      # 匹配成功

        return i - plength

    return -1

if __name__ == "__main__":

    p = "abc"

    print(match(s,p))

## 最长公共子串的递推实现
用二维表记录以对应尾部结尾的最长相等后缀，相等时继承左上角并加一，否则清零。遍历后按最大值回溯得到子串。说明后给出实现。

[PRE5]
**** 方法功能：获取两个字符串的最长公共子串

**** 输入参数：str1 和 str2 为指向字符的指针

[PRE6]
def getMaxSubStr(str1,str2):

    len1 = len(str1)

    len2 = len(str2)

    sb = ""

    maxs = 0

    maxI = 0

    # 申请新的空间来记录公共子串长度信息

    M = [[0 for i in range(len1+1)] for j in range(len2+1)]

    # 通过利用通用公式写新建立的二维数组（公共子串的长度信息）

    i = 0

    while i < len1+1:

        M[i][0] = 0

        i += 1

    j = 0

    while j < len2+1:

        M[0][j] = 0

        j += 1

    i = 1

    while i < len1+1:

        j = 1

        while j < len2+1:

            if list(str1)[i-1] == list(str2)[j-1]:

                M[i][j] = M[i-1][j-1] + 1

                if M[i][j] > maxs:

                    maxs = M[i][j]

                    maxI = i

            else:

                M[i][j] = 0

            j += 1

        i += 1

    # 找出公共子串

    i = maxI - maxs

    while i < maxI:

        sb += "".join(list(str1)[i])

        i += 1

    return sb
