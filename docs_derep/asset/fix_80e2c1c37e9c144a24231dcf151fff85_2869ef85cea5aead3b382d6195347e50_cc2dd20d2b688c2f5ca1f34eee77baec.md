def findWithBinary(array,data):

    if array == None or len(array) < 1:

        print("参数不合理")

        return False

    # 从二维数组右上角元素开始遍历

    i = 0

    rows = len(array)

    columns = len(array[0])

    j = columns - 1

    while i < rows and j >= 0:

        # 在数组中找到 data

        if array[i][j] == data:

            return True

        # 当前遍历到数组中的值大于 data, data 肯定不在这一列中

        elif array[i][j] > data:

            j -= 1

        # 当前遍历到数组中的值小于 data, data 肯定不在这一行中

        else:

            i += 1

    return False
