# 若候选集为空，则自动从全量规则库收集可触发项

function runOne(object: Object,
    triggered: Rule[]) -> Rule[]:

    if not triggered:
        triggered = getTriggeredRules(object: Object)
        if not triggered:
            return null
        triggered.shuffle()

    rule = triggered.peek()
    rule.fire(object)
    return triggered
