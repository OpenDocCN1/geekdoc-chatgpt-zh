class WebSearch(BaseTool):

    name: str = "web_search"

    description: str = """发起网络检索并返回相关网址列表。
默认会调用主要搜索引擎接口获取最新数据，出现异常时自动切换至备选源。"""

    parameters: dict = {

        "type": "object", "properties": {

            "query": {

                "type": "string",

                "description": "(必需) 传入搜索引擎的查询语句。",

            },

            "num_results": {

                "type": "integer",

                "description": "(可选) 需要返回的条目数，未指定默认 10。",

                "default": 10,

            },

            "required": ["query"],

        }

    }

    _search_engine: dict[str, WebSearchEngine] = {

        "google": GoogleSearchEngine(),

        "baidu": BaiduSearchEngine(),

        "duckduckgo": DuckDuckGoSearchEngine(),

        "bing": BingSearchEngine(),

    }

    async def execute(self, query: str, num_results: int = 10) -> List[str]:

        """

        向外发起搜索并给出 URL 列表

        会按照既定顺序逐个尝试可用引擎，单点失效时立即换下一个

        当全部引擎均不可用时，将按间隔休眠后重试，直至用完允许的重试次数

        参数:

            query (str): 发往搜索引擎的查询词

            num_results (int, 可选): 需要的链接数量，未提供则取 10

        返回:

            List[str]: 与查询相符的网址列表。

        """

        # 读取重试参数

        retry_delay = 60  # 默认间隔 60 秒

        max_retries = 3  # 默认最多重试 3 次

        if config.search_config:

            retry_delay = getattr(config.search_config, "retry_delay", 60)

            max_retries = getattr(config.search_config, "max_retries", 3)

        # 按顺序尝试各引擎，全部失败则进入重试

        for retry_count in range(

            max_retries + 1

        ):  # +1 来计入首次尝试而不算作重试

            links = await self._try_all_engines(query, num_results)

            if links:

                return links

            if retry_count < max_retries:

                # 全部失效则休眠后重试

                logger.warning(

                    f" 所有搜索源均失效，{retry_delay} 秒后进行第 {retry_count + 1}/{max_retries} 次重试...")

                await asyncio.sleep(retry_delay)

            else:

                logger.error(

                    f" 经 {max_retries} 次重试后全部搜索源仍不可用，终止本次查询。")

        return []

    async def _try_all_engines(

        self, query: str, num_results: int) -> List[str]:

        """

        按预设顺序轮询各个搜索引擎

        参数:

            query (str): 传给引擎的检索词

            num_results (int): 期望返回的链接条数

        返回:

            List[str]: 匹配结果的网址列表，全部失败时返回空列表

        """

        engine_order = self._get_engine_order()

        failed_engines = []

        for engine_name in engine_order:

            engine = self._search_engine[engine_name]

            try:

                logger.info(f"🔍 正在用 {engine_name.capitalize()} 执行检索...")

                links = await self._perform_search_with_engine(

                    engine, query, num_results)

                if links:

                    if failed_engines:

                        logger.info(

                            f" {engine_name.capitalize()} 检索成功，之前已尝试过：{', '.join(failed_engines)}")

                    return links

            except Exception as e:

                failed_engines.append(engine_name.capitalize())

                is_rate_limit = (

                    "429" in str(e) or "Too Many Requests" in str(e))

                if is_rate_limit:

                    logger.warning(

                        f"⚠️ {engine_name.capitalize()} 请求受限，切换到下一个搜索源 ...")

                else:

                    logger.warning(

                        f"⚠ {engine_name.capitalize()} 检索异常：{e}"

                    )

        if failed_engines:

            logger.error(f"全部引擎均报错：{', '.join(failed_engines)}")

        return []
