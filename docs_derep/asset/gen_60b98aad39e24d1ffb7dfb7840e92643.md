词干提取把不同形式的词压缩到同一根。同一批词用 Porter、Lancaster、Snowball 三种实现对比效果。

from nltk.stem.porter import PorterStemmer
from nltk.stem.lancaster import LancasterStemmer
from nltk.stem.snowball import SnowballStemmer

input_words = ['writing', 'calves', 'be', 'branded', 'horse', 'randomize', 
        'possibly', 'provision', 'hospital', 'kept', 'scratchy', 'code']

# Create various stemmer objects
porter = PorterStemmer()
lancaster = LancasterStemmer()
snowball = SnowballStemmer('english') 
[PRE13]
# Create a list of stemmer names for display
stemmer_names = ['PORTER', 'LANCASTER', 'SNOWBALL']
formatted_text = '{:>16}' * (len(stemmer_names) + 1)

最后是常用工程技巧：把长文本按固定词数切块，便于批处理和训练。先读取 Brown语料前 12000 个词，再用 700 词为单位分组并预览。

[PRE20]python
import numpy as np
from nltk.corpus import brown 
[PRE21]python
# Split the input text into chunks, where
# each chunk contains N words
def chunker(input_data, N):
    input_words = input_data.split(' ')
    output = [] 
[PRE22]python
cur_chunk = []
count = 0
for word in input_words:
    cur_chunk.append(word)
    count += 1
    if count == N:
        output.append(' '.join(cur_chunk))
        count, cur_chunk = 0, []
output.append(' '.join(cur_chunk))
return output 
[PRE23]python
if __name__=='__main__':
    # Read the first 12000 words from the Brown corpus
    input_data = ' '.join(brown.words()[:12000]) 
[PRE24]python
# Define the number of words in each chunk
chunk_size = 700 
[PRE25]python
chunks = chunker(input_data, chunk_size)
print('\nNumber of text chunks =', len(chunks), '\n')
for i, chunk in enumerate(chunks):
    print('Chunk', i+1, '==>', chunk[:50])
