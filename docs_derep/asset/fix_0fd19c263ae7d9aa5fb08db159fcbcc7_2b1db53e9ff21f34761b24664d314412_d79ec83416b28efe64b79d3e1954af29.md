    """

    :param continue_buildingsample_num: 这个参数指的是在输入的prompt_token后再输出多少个字符

    :param prompt_token: 这是需要转换成Token的内容，这里需要输入一个list

    :param top_k:

    :return: 输出一个Token序列

    """

    for i in range(continue_buildingsample_num):

        _token_inp = torch.tensor([prompt_token_new]).to("cuda")

        if self.use_rlhf:

            result, _ = self.forward(_token_inp)

        else:

            result = self.forward(_token_inp)

        logits = result[:, -1, :]

        probs = torch.softmax(logits / temperature, dim=-1)

        next_token = self.sample_top_p(probs, top_p)  # 预设的top_p = 0.95

        next_token = next_token.reshape(-1)

        prompt_token_new.append(next_token.item())

    return prompt_token_new

def sample_top_p(self, probs, p):

    probs_sort, probs_idx = torch.sort(probs, dim=-1, descending=True)

    probs_sum = torch.cumsum(probs_sort, dim=-1)

    mask = probs_sum - probs_sort > p

    probs_sort[mask] = 0.0

    probs_sort.div_(probs_sort.sum(dim=-1, keepdim=True))

    next_token = torch.multinomial(probs_sort, num_samples=1)

    next_token = torch.gather(probs_idx, -1, next_token)

    return next_token

接下来讲解散度计算模块AdaptiveKLController，即PPO2算法。无论是经典PPO还是自定义PPO，KL divergence的计算都至关重要，它用于衡量两个分布之间的距离。

本实现采用AdaptiveKLController，在梯度计算中加入clip操作，形成PPO2。

其机理是：当优势函数为正，需要提升当前动作被选中的概率时，会限制新旧分布在当前状态动作下比值的上限，超过阈值则停止策略更新；当优势函数为负，需要降低被选中概率时，则限制比值的下限，超过阈值同样停止更新。这样可在参数更新的同时保证新旧分布距离保持在设定范围内，如图15-15所示。

[PRE9]

图15-15 PPO2算法演示

该方法能通过动态调节KL约束项的惩罚系数，在保证训练稳定的前提下让策略分布产生足够变化，即更新幅度要小以保证稳定性，又要足够大以改变策略。

如图15-16所示的策略，对更新前后的KL距离设定一个可调超参数target作为目标约束，直接限制KL divergence的最大变化。

与直接约束KL divergence不同，该做法对优势函数进行了裁剪。其中当重要性采样系数超出固定区间，一般设为[-0.2, 0.2]，见下方代码时，该次更新会被丢弃，裁剪后的损失与参数无关，不产生梯度。本质上是忽略新旧策略差异过大带来的优势值，以保证训练稳定和梯度更新的单调性。
