def summarize_text(text, model_name):

    response = openai.Completion.create(

        engine=model_name,

        prompt=f"请对以下文本进行总结，注意总结的凝练性，将总结字数控制在 20 个字以内:\n{text}",

        temperature=0.3,

        max_tokens=100,

    )

    summarized_text = response.choices[0].text.strip()

    return summarized_text
