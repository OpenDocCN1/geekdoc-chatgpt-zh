def send_messages_tools(messages):

    """发送消息并调用工具"""

    response = client.chat.completions.create(

        model="deepseek-chat",  # 使用的模型

        messages=messages,  # 消息列表

        tools=tools,  # 工具列表

        tool_choice="auto",  # 让模型自动选择是否调用工具

    )

    return response.choices[0].message
