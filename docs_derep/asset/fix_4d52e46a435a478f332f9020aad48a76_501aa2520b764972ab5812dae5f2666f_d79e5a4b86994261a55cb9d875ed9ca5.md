轻量MNIST实战：深度可分离膨胀卷积

MNIST手写数字常被用来验证轻量化结构的可行性。下面这个实现把逐通道卷积与1x1点卷积串联，并借助膨胀率扩大感受野，在参数大幅压缩的前提下保留表达力，同时给出一套可直接跑通的训练闭环。

数据与设备
标签通过numpy从文件读取，图像张量在第1轴上做一次升维，变成可供卷积处理的四维格式，打印一次shape用来确认通道位置。运行时根据CUDA是否可用选择计算设备，模型实例化后送到目标设备并开启torch.compile加速。

网络骨架
整体堆叠分为三段：首层把单通道展开到12通道，采用7x7标准卷积并接ReLU；中间插入自制的深度可分离膨胀模块，再次激活；末尾用24到6通道的3x3卷积收敛特征。特征图经einops的Rearrange从b c h w展平为b (c h w)，最后接1536到10的线性层输出logits。

可直接运行的完整脚本
y_train_label = np.load("../dataset/mnist/y_train_label.npy")

x_train = np.expand_dims(x_train,axis=1)

print(x_train.shape)

depth_conv = nn.Conv2d(in_channels=12, out_channels=12, kernel_size=3, groups=6,dilation=2)

point_conv = nn.Conv2d(in_channels=12, out_channels=24, kernel_size=1)

# 深度、可分离、膨胀卷积的定义

depthwise_separable_conv = torch.nn.Sequential(depth_conv, point_conv)

class MnistNetword(nn.Module):

    def __init__(self):

        super(MnistNetword, self).__init__()

        self.convs_stack = nn.Sequential(

            nn.Conv2d(1,12,kernel_size=7),

            nn.ReLU(),

            depthwise_separable_conv,

            nn.ReLU(),

            nn.Conv2d(24,6,kernel_size=3)

        )

        self.logits_layer = nn.Linear(in_features=1536,out_features=10)

    def forward(self,inputs):

        image = inputs

        x = self.convs_stack(image)

        x = elt.Rearrange("b c h w -> b (c h w)")(x)

        logits = self.logits_layer(x)

        return logits

device = "cuda" if torch.cuda.is_available() else "cpu"

#注意需要将model发送到GPU计算

model = MnistNetword().to(device)

model = torch.compile(model)

loss_fn = nn.CrossEntropyLoss()

optimizer = torch.optim.SGD(model.parameters(), lr=1e-4)

batch_size = 128

for epoch in range(63):

    train_num = len(x_train)//128

    train_loss = 0.

    for i in range(train_num):

        start = i * batch_size

        end = (i + 1) * batch_size

        x_batch = torch.tensor(x_train[start:end]).to(device)

        y_batch = torch.tensor(y_train_label[start:end]).to(device)

        pred = model(x_batch)

        loss = loss_fn(pred, y_batch)

        optimizer.zero_grad()

        loss.backward()

        optimizer.step()

        train_loss += loss.item()  # 记录每个批次的损失值

    # 计算并打印损失值

    train_loss /= train_num

    accuracy = (pred.argmax(1) == y_batch).type(torch.float32).sum().item() / batch_size

    print("epoch: ",epoch,"train_loss:",

    round(train_loss,2),"accuracy:",round(accuracy,2))

训练细节
采用CrossEntropyLoss配合SGD，学习率设为1e-4。每个epoch按128为步长切片数据，完成梯度清零、反向传播、参数更新。批次损失累加后求均值，并在epoch末用当前批次的预测计算准确率，打印epoch、loss、accuracy三项指标。

深度可分离的原理
PyTorch的Conv2d通过groups控制通道拆分，当groups等于输入通道数时退化为逐通道的空间卷积，再配合1x1卷积实现跨通道融合。构造方式如下：

import torch
import torch.nn as nn
import numpy as np
import einops.layers.torch as elt
#下面是自定义的深度、可分离、膨胀卷积的定义
depth_conv = nn.Conv2d(in_channels=12, out_channels=12, kernel_size=3, groups=6,dilation=2)
point_conv = nn.Conv2d(in_channels=12, out_channels=24, kernel_size=1)
depthwise_separable_conv = torch.nn.Sequential(depth_conv, point_conv)
class MnistNetword(nn.Module):
    def __init__(self):
        super(MnistNetword, self).__init__()
        self.convs_stack = nn.Sequential(
            nn.Conv2d(1,12,kernel_size=7),
            nn.ReLU(),
            depthwise_separable_conv,    #使用自定义卷积替代了原生卷积层
            nn.ReLU(),
            nn.Conv2d(24,6,kernel_size=3)
        )
        self.logits_layer = nn.Linear(in_features=1536,out_features=10)
    def forward(self,inputs):
        image = inputs
        x = self.convs_stack(image)
        x = elt.Rearrange("b c h w -> b (c h w)")(x)
        logits = self.logits_layer(x)
        return logits
