from collections import deque

import random

class PolicyNetwork(nn.Module):

    def __init__(self, state_dim, action_dim):

        super(PolicyNetwork, self).__init__()

        self.fc = nn.Sequential(

            nn.Linear(state_dim, 64),

            nn.ReLU(),

            nn.Linear(64, 32),

            nn.ReLU(),

            nn.Linear(32, action_dim),

            nn.Softmax(dim=-1)

        )

    def forward(self, x):

        return self.fc(x)

class RLAgent:

    def __init__(self, state_dim, action_dim):

        self.policy_net = PolicyNetwork(state_dim, action_dim)

        self.optimizer = optim.Adam(self.policy_net.parameters(), lr=0.001)

        self.gamma = 0.99

        self.memory = deque(maxlen=10000)

        self.batch_size = 32

    def select_action(self, state):

        state = torch.FloatTensor(state)

        probs = self.policy_net(state)

        action = torch.multinomial(probs, 1).item()

        return action

    def store_transition(self, state, action, reward, next_state, done):

        self.memory.append((state, action, reward, next_state, done))

    def train(self):

        if len(self.memory) < self.batch_size:

            return

        batch = random.sample(self.memory, self.batch_size)

        states, actions, rewards, next_states, dones = zip(*batch)

        states = torch.FloatTensor(np.array(states))

        actions = torch.LongTensor(actions)

        rewards = torch.FloatTensor(rewards)

        next_states = torch.FloatTensor(np.array(next_states))

        dones = torch.FloatTensor(dones)

        # 计算折扣回报（从后往前）

        discounted_rewards = []

        running_reward = 0

        for reward, done in zip(reversed(rewards), reversed(dones)):

            if done:

                running_reward = 0

            running_reward = reward + self.gamma * running_reward

            discounted_rewards.insert(0, running_reward)

        discounted_rewards = torch.FloatTensor(discounted_rewards)

        # 标准化回报（提升稳定性）

        discounted_rewards = (discounted_rewards - discounted_rewards.mean()) / (discounted_rewards.std() + 1e-7)

        # 计算 log 概率和损失

        log_probs = torch.log(self.policy_net(states).gather(1, actions.unsqueeze(1)).squeeze())

        loss = -torch.mean(log_probs * discounted_rewards)

        self.optimizer.zero_grad()

        loss.backward()

        self.optimizer.step()

# ============ 训练主循环 ============

if __name__ == "__main__":

    agent = RLAgent(state_dim=4, action_dim=2)

    episode_rewards = []  # 记录每个 episode 的总奖励

### 5.5.3 常见问题与解决方案

在AI智能体研发中，检索噪声、信息冗余和长上下文处理是制约决策效率与输出质量的关键瓶颈。以下从问题本质、成因和对策三方面展开。

#### 1. 检索噪声

##### 1) 定义与影响
检索噪声指从外部知识库、文档库或互联网检索时，返回结果混入不相关、错误或低质量内容。这些噪声会干扰推理，导致错误回答或无效行动，长期损害用户信任。

##### 2) 成因
- (1) 查询意图理解偏差：用户输入模糊，如“苹果的最新动态”，难以区分水果或公司，造成检索方向错误。
- (2) 检索算法缺陷：传统关键词匹配对同义词如“医生/医师”、多义词如“Java”语言/岛屿不敏感，易产生误匹配。
- (3) 知识库质量不足：文档存在过时政策、重复内容，或缺乏事实与观点的标注。

##### 3) 应对措施
(1) 提升查询理解精度：
- 查询改写与扩展：用LLM将“苹果的价格”改写为“2025年水果苹果市场零售价”，明确实体和时间范围；通过同义词扩展覆盖更多检索词。
- 意图识别与分类：结合基于BERT的分类器或LLM零样本能力，预先判断查询意图如事实查询、情感咨询、指令执行，针对性调整策略。

(2) 检索算法升级：从字面匹配转向语义匹配：
- 语义检索：把查询和文档转为高维向量，通过Sentence-BERT、OpenAI Embeddings计算余弦相似度判断相关性。例如“如何缓解头痛”与“头痛应对方法”表述不同但语义相近，可被准确召回。
- 向量数据库支撑：使用Milvus、Pinecone、Weaviate等存储文档向量，实现大规模近似最近邻搜索。

(3) 结果过滤与重排：
- 相关性打分过滤：用LLM作为裁判对检索文档打1~5分，过滤低于阈值的内容。例如查询“苹果公司股价”时，可过滤掉“水果苹果种植技术”。
- 交叉注意力重排：通过查询与文档细粒度交互重排结果。例如检索“爱因斯坦相对论贡献”时，优先保留含“狭义相对论”“质能方程”的文档。

(4) 知识库清洗与增强：
- 数据预处理：通过规则或SimHash去重，清理错误重复内容；对文档进行质量标注，优先检索学术论文、官网等高质量来源。
- 知识增强：构建领域知识图谱如医疗“疾病-症状-治疗”，将非结构化文档转为结构化实体关系，减少歧义，例如在图谱中明确“苹果”区分为公司与水果。
