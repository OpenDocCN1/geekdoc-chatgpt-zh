##### Swish激活的实现

class Swish(nn.Module):

    def __init__(self, beta=1.0):

        super(Swish, self).__init__()

        self.beta = beta

    def forward(self, x):

        return x * torch.sigmoid(self.beta * x)

##### PyTorch实现的SwiGLU模块

class Swiglu(torch.nn.Module):

    """多层感知机实现。

    输入为h维隐藏状态，先升维到4h进行非线性变换，随后降维回h维。
    """

    # 构造器
    def __init__(self, hidden_size=384, add_bias_linear=False):

        # 初始化父类
        super(Swiglu, self).__init__()

        # 线性层是否带偏置
        self.add_bias = add_bias_linear

        # 记录隐藏大小
        self.hidden_size = hidden_size

        # 建立h到4h的线性映射
        # 参考论文 https://arxiv.org/pdf/2002.05202.pdf，使用SwiGLU时中间宽度加倍
        self.dense_h_to_4h = torch.nn.Linear(
            hidden_size,
            hidden_size * 4,
            bias=self.add_bias
        )

        # 内部定义SwiGLU激活
        def swiglu(x):
            # 沿最后维度切分为两段
            x = torch.chunk(x, 2, dim=-1)
            # 用SiLU处理第一段后与第二段点乘
            return torch.nn.functional.silu(x[0]) * x[1]

        # 把激活函数保存为实例属性
        self.activation_func = swiglu

        # 建立2h到h的回投影
        # 激活后维度减半，故输入为2h
        self.dense_4h_to_h = torch.nn.Linear(
            hidden_size * 2,
            hidden_size,
            bias=self.add_bias
        )

    # 前向计算
    def forward(self, hidden_states):

        # 将隐藏状态映射到4h
        intermediate_parallel = self.dense_h_to_4h(hidden_states)

        # 执行SwiGLU激活
        intermediate_parallel = self.activation_func(intermediate_parallel)

        # 映射回h维
        output = self.dense_4h_to_h(intermediate_parallel)

        # 返回结果
        return output
