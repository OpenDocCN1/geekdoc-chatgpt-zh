ChatGLM-6B 用 LoRA 做轻量化改写实战

在显存吃紧的情况下想给 6B 级别的基础模型注入特定能力，LoRA 是性价比最高的路径。下面把一次从推理验证到低秩微调落地的完整链路梳理出来，供复用。

gMASK 驱动的自回归生成片段首先演示了推理时的滑动窗口做法。把提示词、来源文本以及两个特殊标记拼接成 token 列表，转成 id 后进入循环，每一步只读取倒数第三位的 logits，经 0.95 温度缩放后做 softmax，再以 top_p 0.95 采样。窗口更新时保留 [gMASK] 与 <sop> 的相对位置，遇到终止 id 130005 即退出，最后解码输出。

LoRA 的本质是冻结原始权重，仅用低秩增量逼近变化。原权重记作 W0，增量 ΔW 通过两个小矩阵分解得到，进而影响前向计算。
[PRE1]

$W = W_0 + \Delta W$
[PRE2]

$W_0 \in R^{d \times k}$
[PRE3]

$\Delta W = B \times A$
[PRE4]

$H = W_0 x + \Delta W x = W_0 x + B A x$

def print_trainable_parameters(model):

    trainable_params = 0

    all_param = 0

    for _, param in model.named_parameters():

        num_params = param.numel()

        if num_params == 0 and hasattr(param, "ds_numel"):

            num_params = param.ds_numel

        all_param += num_params

        if param.requires_grad:

            trainable_params += num_params

    print(

        f"trainable params: {trainable_params} || all params: {all_param} || trainable%: {100 * trainable_params / all_param}")

def name_is_lora(name):

    return (

        len(name.split(".")) >= 4

        and (name.split(".")[-4]) == "parametrizations"

        and name.split(".")[-1] in ["lora_A", "lora_B"]

    )

def get_lora_state_dict(model):

    return {k: v for k, v in model.state_dict().items() if name_is_lora(k)}

def get_params_by_name(model, print_shapes=False, name_filter=None):

    for n, p in model.named_parameters():

        if name_filter is None or name_filter(n):

            if print_shapes:

                print(n, p.shape)

            yield p

def get_lora_params(model, print_shapes=False):

    return get_params_by_name(model, print_shapes=print_shapes,

                             name_filter=name_is_lora)
