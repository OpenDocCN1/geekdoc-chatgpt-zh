# 行为调度核心：抢占优先与共存验证机制

调度器运作遵循“来者居上”与“和谐共处”双轨制。当高优先级指令介入，现行任务队列瞬间清空，新指令独占执行权并重设优先级阈值；普通指令欲加入则须逐一通过在场任务的冲突检测，全票通过方能入池并同步抬升门槛。每帧结算期，系统自动扫描活跃池：已终结任务即时移出，其余继续驱动推进。

class ActionCombination extends Action:

    # 保存子动作

    actions: Action[]

    function interrupt() -> bool:

        # 如果有任何子动作可以中断，则中断该动作

        for action in actions:

            if action.interrupt():

                return true

        return false

    function canDoBoth(other: Action) -> bool:

        # 如果所有子动作都可以执行，则执行组合动作

        for action in actions:

            if not action.canDoBoth(other):

                return false

        return true

    function isComplete() -> bool:

        # 如果所有子动作都已完成，则可以认为动作已经完成

        for action in actions:

            if not action.isComplete():

                return false

        return true

    function execute():

        # 执行所有子动作

        for action in actions:

            action.execute()

function interrupt() -> bool:

    # 如果第一个子动作可以中断，则中断它

    return actions[0].interrupt()

function canDoBoth(other: Action) -> bool:

    # 如果所有子动作都可以执行，则执行组合动作

    # 如果仅测试了第一个，则可能会出现一种危险

    # 即在序列中间突然发现不兼容的情况

    for action in actions:

        if not action.canDoBoth(other):

            return False

    return true

function isComplete() -> bool:

    # 如果所有子动作都已完成，则可以认为动作已经完成

    return activeIndex >= len(actions)

function execute():

    # 执行当前动作

    actions[activeIndex].execute()

    # 如果当前动作已经完成，则跳转到下一个动作

    if actions[activeIndex].isComplete():

        activeIndex += 1
