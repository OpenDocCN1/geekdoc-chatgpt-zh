//pages/book/book.js

import db from '../../tools/database.js'

Page({

  data: {

    book:null,

    isbn:null

  },

  onLoad: function (options) {

    let isbn = options.isbn;

    db.getBook(isbn).then((res)=>{

      this.setData({

        book: res[0],

        isbn:isbn

      });

      wx.setNavigationBarTitle({

        title: res[0].title,

      })

    }).catch((error)=>{

      wx.showToast({

        title: '获取不到图书信息',

      })

    });

  },

  onReachBottom: function () {

collect:function() {

    db.collectBook(this.data.book.isbn).then(()=>{

        wx.showToast({

            title: '添加收藏成功',

        })

    }).catch(()=>{

    });

}
