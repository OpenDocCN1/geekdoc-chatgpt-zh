def call_deepseek_chat_api(

    messages: List[Dict[str, str]],

    functions: Optional[List[Dict[str, Any]]] = None,

    api_key: str = "",

    api_base: str = "https://api.deepseek.com",

    model: str = "deepseek-chat"

) -> Dict[str, Any]:

    """调用DeepSeek聊天API"""

    payload = {

        "model": model,

        "messages": messages,

        "temperature": 0.7

    }

    if functions:

        payload["functions"] = functions

        payload["function_call"] = "auto"

    try:

        response = requests.post(

            f"{self.api_base}/chat/completions",

            headers=self.headers,

            data=json.dumps(payload)

        )

        response.raise_for_status()

        return response.json()

    except requests.exceptions.RequestException as e:

        print(f"API请求错误: {e}")

        return {"error": str(e)}

def run_tool_augmented_conversation(query: str, api_key: str) -> None:

    """运行工具增强的对话流程"""

    # 定义可用工具函数的描述

    functions = [

        {

            "name": "get_weather",

            "description": "获取指定地点的天气信息",

            "parameters": {

                "type": "object",

                "properties": {

                    "location": {

                        "type": "string",

                        "description": "城市名称，例如：北京"

                    },

                    "date": {

                        "type": "string",

                        "description": "日期，格式：YYYY-MM-DD，默认为今天"

                    }

                },

                "required": ["location"]

            }

        },

        {

            "name": "calculate_sum",

            "description": "计算一组数字的总和",

            "parameters": {

                "type": "object",

                "properties": {

                    "numbers": {

                        "type": "array",

                        "items": {"type": "number"},

                        "description": "需要求和的数字列表"

                    }

                },

                "required": ["numbers"]

            }

        },

        {

            "name": "get_stock_price",

            "description": "获取指定股票的价格信息",

            "parameters": {

                "type": "object",

                "properties": {

                    "symbol": {

                        "type": "string",

                        "description": "股票代码，例如：AAPL"

                    },

                    "date": {

                        "type": "string",

                        "description": "日期，格式：YYYY-MM-DD，默认为今天"

                    }

                },

                "required": ["symbol"]

            }

        }

    ]

    client = DeepSeekClient(api_key)

    messages = [{"role": "user", "content": query}]

    # 主对话循环

    while True:

        # 调用DeepSeek API

        response = client.chat_completion(messages, functions)

        if "error" in response:

            print(f"API错误: {response['error']}")

            break

        assistant_message = response["choices"][0]["message"]

        messages.append(assistant_message)

        # 检查是否有函数调用

        if "function_call" in assistant_message:

            function_call = assistant_message["function_call"]

            function_name = function_call["name"]

            parameters = function_call.get("parameters", {})

            # 执行函数调用

            try:

                function_response = PythonFunctionCaller.execute_function_call(

                    function_name, parameters

                )

            except Exception as e:

                function_response = {"error": f"函数执行错误: {str(e)}"}

            # 将函数调用结果添加到对话中

            messages.append({

                "role": "function",

                "name": function_name,

                "content": json.dumps(function_response)

            })

            print(f"执行函数: {function_name}")

            print(f"参数: {parameters}")

            print(f"结果: {function_response}")

        else:

            # 没有函数调用，直接显示AI回复

            print(f"AI回复: {assistant_message['content']}")

            break

这段实现演示了如何为 DeepSeek 大模型接入工具增强能力，整体由三块组成。

1. PythonFunctionCaller 类：把可供模型调用的外部能力封装起来，示例里包含了查询天气、求数字总和、查询股价三类工具。
2. DeepSeekClient 类：负责与 DeepSeek 服务端交互，支持带函数定义的请求参数传递。
3. run_tool_augmented_conversation 函数：完成了从输入到输出的完整工具对话闭环，具体包括：
    - 预先声明工具的字段与参数描述；
    - 携带函数定义发起对话请求；
    - 解析模型返回的函数调用指令；
    - 在本地执行对应的 Python 工具；
    - 把执行结果回填到对话历史，供模型继续生成最终答案。

想跑通这段示例，需要先准备 DeepSeek 的 API Key，可以写入环境变量 DEEPSEEK_API_KEY，也可直接改代码里的 API_KEY 变量。示例查询同时涉及天气和算术，体现了多工具联动的处理思路。

## 4.5 评估与优化

智能体的评估与优化是决定交付质量和用户体验的核心环节。评估用来量化表现，为改进提供依据；优化则基于评估结果对提示、流程或工具做针对性调整。

### 4.5.1 提示效果评估指标

在 LLM 应用里，提示的质量直接决定输出效果。下面是常用的评估维度。

#### 1. 基础评估指标

1) 响应相关性 Relevance
    - 判断模型回答是否贴合用户真实意图。
    - 采用 1 到 5 分打分，分值越高越相关。

2) 信息完整性 Completeness
    - 判断回答是否覆盖了提示中提出的所有要点。
    - 采用 1 到 5 分打分，分值越高越完整。

3) 流畅度 Fluency
    - 判断文本的语言通顺程度与自然度。
    - 采用 1 到 5 分打分，分值越高越流畅。

#### 2. 高级评估指标

1) 事实准确性 Factual Accuracy
    - 判断陈述事实是否正确，需结合外部知识库校验。
2) 一致性 Consistency
    - 判断回答内部逻辑是否自洽，以及与历史对话的一致程度。
3) 多样性 Diversity
    - 针对创意类任务，衡量生成结果的多样程度。
4) 毒性/偏见评分 Toxicity/Bias
    - 检测输出是否包含有害内容或偏见倾向。

#### 3. 任务特定指标

1) 代码正确性 Code Correctness
    - 对于代码生成任务，检验可执行性与逻辑正确性。
2) 推理质量 Reasoning Quality
    - 对于需要逻辑推导的任务，评估推理链条的合理性。
