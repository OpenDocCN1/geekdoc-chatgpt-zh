def call_deepseek_chat_api(

    messages: List[Dict[str, str]],

    functions: Optional[List[Dict[str, Any]]] = None,

    api_key: str = "",

    api_base: str = "https://api.deepseek.com",

    model: str = "deepseek-chat"

) -> Dict[str, Any]:

    """调用DeepSeek聊天API"""

    payload = {

        "model": model,

        "messages": messages,

        "temperature": 0.7

    }

    if functions:

        payload["functions"] = functions

        payload["function_call"] = "auto"

    try:

        response = requests.post(

            f"{self.api_base}/chat/completions",

            headers=self.headers,

            data=json.dumps(payload)

        )

        response.raise_for_status()

        return response.json()

    except requests.exceptions.RequestException as e:

        print(f"API请求错误: {e}")

        return {"error": str(e)}

def run_tool_augmented_conversation(query: str, api_key: str) -> None:

    """运行工具增强的对话流程"""

    # 定义可用工具函数的描述

    functions = [

        {

            "name": "get_weather",

            "description": "获取指定地点的天气信息",

            "parameters": {

                "type": "object",

                "properties": {

                    "location": {

                        "type": "string",

                        "description": "城市名称，例如：北京"

                    },

                    "date": {

                        "type": "string",

                        "description": "日期，格式：YYYY-MM-DD，默认为今天"

                    }

                },

                "required": ["location"]

            }

        },

        {

            "name": "calculate_sum",

            "description": "计算一组数字的总和",

            "parameters": {

                "type": "object",

                "properties": {

                    "numbers": {

                        "type": "array",

                        "items": {"type": "number"},

                        "description": "需要求和的数字列表"

                    }

                },

                "required": ["numbers"]

            }

        },

        {

            "name": "get_stock_price",

            "description": "获取指定股票的价格信息",

            "parameters": {

                "type": "object",

                "properties": {

                    "symbol": {

                        "type": "string",

                        "description": "股票代码，例如：AAPL"

                    },

                    "date": {

                        "type": "string",

                        "description": "日期，格式：YYYY-MM-DD，默认为今天"

                    }

                },

                "required": ["symbol"]

            }

        }

    ]

    client = DeepSeekClient(api_key)

    messages = [{"role": "user", "content": query}]

    # 主对话循环

    while True:

        # 调用DeepSeek API

        response = client.chat_completion(messages, functions)

        if "error" in response:

            print(f"API错误: {response['error']}")

            break

        assistant_message = response["choices"][0]["message"]

        messages.append(assistant_message)

        # 检查是否有函数调用

        if "function_call" in assistant_message:

            function_call = assistant_message["function_call"]

            function_name = function_call["name"]

            parameters = function_call.get("parameters", {})

            # 执行函数调用

            try:

                function_response = PythonFunctionCaller.execute_function_call(

                    function_name, parameters

                )

            except Exception as e:

                function_response = {"error": f"函数执行错误: {str(e)}"}

            # 将函数调用结果添加到对话中

            messages.append({

                "role": "function",

                "name": function_name,

                "content": json.dumps(function_response)

            })

            print(f"执行函数: {function_name}")

            print(f"参数: {parameters}")

            print(f"结果: {function_response}")

        else:

            # 没有函数调用，直接显示AI回复

            print(f"AI回复: {assistant_message['content']}")

            break

该实现演示了如何让 DeepSeek 模型借助外部能力，整体分为三层。

1. PythonFunctionCaller：把可被模型使用的本地能力打包起来，示例里准备了查天气、求和、查股价三项。

3. run_tool_augmented_conversation：从用户输入到最终答案的完整闭环，包括声明可用能力、带定义发起请求、读取模型提出的调用计划、在本地执行、把返回结果写回会话供模型继续生成。

要跑通，需要提前准备好 DeepSeek 的访问凭证，推荐写入系统环境变量 DEEPSEEK_API_KEY，也可直接在代码中替换 API_KEY。示例查询同时问天气和算术，体现了多步调用的串联思路。

## 4.5 评测与调优
打造可用智能体离不开持续的评测和改进。评测负责量化当前表现，形成改进依据；改进则在评测结果指引下对提示词、流程或接入的能力进行调整。

### 4.5.1 提示质量评估维度
在 LLM 场景里，提示质量直接影响输出。常用的评估维度如下。

#### 1. 基础维度
1) 回复贴合度 Relevance
   - 看输出是否命中用户的真实意图。
   - 采用 1 到 5 分制，分数越高越贴合。
2) 覆盖完整度 Completeness
   - 看一次回答是否把提问涵盖了。
   - 采用 1 到 5 分制，分数越高越全面。
3) 表达顺畅度 Fluency
   - 看文本是否通顺自然。
   - 采用 1 到 5 分制，分数越高越流畅。

#### 2. 进阶维度
1) 事实正确度 Factual Accuracy
   - 判断陈述是否属实，需借助外部知识核对。
2) 自洽性 Consistency
   - 判断回答内部逻辑是否统一，以及与历史对话的连贯程度。
3) 多样性 Diversity
   - 针对创意类任务，衡量输出的多样程度。
4) 风险倾向 Toxicity/Bias
   - 检测是否出现有害内容或偏见表达。

#### 3. 任务专属维度
1) 代码可执行度 Code Correctness
   - 对代码生成任务，检验能否运行且逻辑无误。
2) 推理合理度 Reasoning Quality
   - 对需要逻辑推导的任务，评估推导链条是否合理。
