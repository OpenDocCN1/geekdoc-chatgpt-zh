def findWithBinary(array,data):

    if array == None or len(array) < 1:

        print("参数不合理")

        return False

    # 从矩阵右上角开始查找

    i = 0

    rows = len(array)

    columns = len(array[0])

    j = columns - 1

    while i < rows and j >= 0:

        # 若当前位置命中则返回

        if array[i][j] == data:

            return True

        # 当前值过大则左移一列

        elif array[i][j] > data:

            j -= 1

        # 当前值过小则下移一行

        else:

            i += 1

    return False
