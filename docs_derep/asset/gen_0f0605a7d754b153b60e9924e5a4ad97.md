def Permutation(str,start):

    if str == None or start < 0:

        return

    # 到达末尾时输出当前的一种排列
    if start == len(str)-1:

        print("".join(str),end=' ')

    else:

        i = start

        while i < len(str):

            # 将起始下标与当前下标的字符进行对调
            swap(str,start,i)

            # 固定前缀后，递归处理剩余部分的全排列
            Permutation(str,start+1)

            # 恢复交换前的原始顺序
            swap(str,start,i)

            i += 1

def removeNestedPare(str):

    if str == None:

        return

    Parentheses_num = 0      # 统计未闭合的左括号个数

    if list(str)[0] != '(' or list(str)[-1] != ')':

        return None

    sb = '('

    # 首尾括号可先行剥离
    i = 1

    while i < len(str)-1:

        ch = list(str)[i]

        if ch == '(':

            Parentheses_num += 1

        elif ch == ')':

            Parentheses_num -= 1

        else:

            sb = sb + (list(str)[i])

        i += 1

    # 校验左右括号是否配对
    if Parentheses_num != 0:

        print("由于括号不匹配，因此不做任何操作")

        return None

    # 为结果补上尾部右括号
    sb = sb + ')'

    return sb
