"""基于 tweepy.StreamListener 实现的实时推文处理器。"""

import tweepy

from textblob import TextBlob

class TweetListener(tweepy.StreamListener):

    """负责处理源源不断的推文流。"""

def __init__(self, api, limit=10):

    """初始化计数器与阈值，继承父类构造函数。"""

    self.tweet_count = 0

    self.TWEET_LIMIT = limit

    super().__init__(api)

[PRE7]

def on_status(self, status):

    """每当有新推文推送过来时自动调用。"""

    # 兼容性处理：优先获取完整长文本，失败则退回普通字段
    try:

        tweet_text = status.extended_tweet.full_text

    except:

        tweet_text = status.text

    print(f'Screen name: {status.user.screen_name}')

    print(f'Language: {status.lang}')

    print(f'Status: {tweet_text}')

    # 非英语内容尝试自动翻译
    if status.lang != 'en':

        print(f'Translated: {TextBlob(tweet_text).translate()}')

    print()

    self.tweet_count += 1  # 累计处理条数

    # 达到上限返回 False 终止流，否则继续
    return self.tweet_count != self.TWEET_LIMIT
