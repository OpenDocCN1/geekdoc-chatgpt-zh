def mock_text_to_vector(text: str, dim: int = 64) -> np.ndarray:

    """

    模拟文本到向量的转换，仅用随机数表示，

    真实场景可调用 Sentence-BERT 或 OpenAI 等 Embedding 模型

    """

    rng = np.random.RandomState(abs(hash(text)) % (10**6))

    return rng.rand(dim).astype('float32')
