分布式训练瓶颈实录

最近一次分布式跑训的监控流把瓶颈暴露得非常直观。观察窗口从第5个数据批开始，接下来到第10批的节奏几乎一模一样：搬运模块先把当前批送达，这个过程每次都落在2.00秒左右；随后计算模块接手，真正的前后向更新只用了0.5秒左右就结束；在计算进行的同时，读取模块已经把下一批提前备好。

具体来看，5号批搬运2.00秒后计算0.5秒，读取器已把6号批准备就绪；6号批同样搬运2.00秒、计算0.5秒，7号批已在内存中；7号批搬运2.00秒、计算0.5秒，8号批被提前加载；8号批搬运2.00秒、计算0.5秒，9号批已就位；9号批搬运2.00秒、计算0.5秒，10号批已就位；10号批搬运2.00秒、计算0.5秒后，全部批次结束。整轮运行累计花费25.50秒。

从这段轨迹能看出，算力端几乎一直在空转等待，真正的制约点在数据搬运环节，传输时间是计算时间的四倍左右。

import torch

import torch.nn as nn

import torch.optim as optim

import torchvision

import torchvision.transforms as transforms

import torch.distributed as dist

import torch.multiprocessing as mp

import os

import time

# 定义一个简单的卷积神经网络模型

class SimpleCNN(nn.Module):

    def __init__(self):

        super(SimpleCNN, self).__init__()

        # 卷积层 1：输入通道为 3，输出通道为 32，卷积核为 3*3，填充为 1

        self.conv1 = nn.Conv2d(3, 32, kernel_size=3, padding=1)

        # 卷积层 2：输入通道为 32，输出通道为 64，卷积核为 3*3，填充为 1

        self.conv2 = nn.Conv2d(32, 64, kernel_size=3, padding=1)

        # 全连接层 1：输入特征为 64*8*8，输出特征为 128

        self.fc1 = nn.Linear(64*8*8, 128)

        # 全连接层 2：输入特征为 128，输出特征为 10（类别数）

        self.fc2 = nn.Linear(128, 10)

        # 激活函数(ReLU)

        self.relu = nn.ReLU()

        # 2*2 最大池化

        self.pool = nn.MaxPool2d(2, 2)

    def forward(self, x):

        x = self.relu(self.conv1(x))    # 卷积 1+ReLU

        x = self.pool(x)                # 池化 1

        x = self.relu(self.conv2(x))    # 卷积 2+ReLU

        x = self.pool(x)                # 池化 2

        x = x.view(-1, 64*8*8)          # 展平

        x = self.relu(self.fc1(x))      # 全连接 1+ReLU

        x = self.fc2(x)                 # 全连接 2

        return x

# 初始化分布式环境

def setup(rank, world_size):

    os.environ['MASTER_ADDR'] = 'localhost'    # 主节点地址

    os.environ['MASTER_PORT'] = '12355'        # 主节点端口

    # 初始化进程组，使用 NCCL 作为通信后端

    dist.init_process_group("nccl", rank=rank, world_size=world_size)

# 清理分布式环境

def cleanup():

    dist.destroy_process_group()
