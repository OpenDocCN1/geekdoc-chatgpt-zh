def call_deepseek_chat_api(

    messages: List[Dict[str, str]],

    functions: Optional[List[Dict[str, Any]]] = None,

    api_key: str = "",

    api_base: str = "https://api.deepseek.com",

    model: str = "deepseek-chat"

) -> Dict[str, Any]:

    """调用DeepSeek聊天API"""

    payload = {

        "model": model,

        "messages": messages,

        "temperature": 0.7

    }

    if functions:

        payload["functions"] = functions

        payload["function_call"] = "auto"

    try:

        response = requests.post(

            f"{self.api_base}/chat/completions",

            headers=self.headers,

            data=json.dumps(payload)

        )

        response.raise_for_status()

        return response.json()

    except requests.exceptions.RequestException as e:

        print(f"API请求错误: {e}")

        return {"error": str(e)}

def run_tool_augmented_conversation(query: str, api_key: str) -> None:

    """运行工具增强的对话流程"""

    # 定义可用工具函数的描述

    functions = [

        {

            "name": "get_weather",

            "description": "获取指定地点的天气信息",

            "parameters": {

                "type": "object",

                "properties": {

                    "location": {

                        "type": "string",

                        "description": "城市名称，例如：北京"

                    },

                    "date": {

                        "type": "string",

                        "description": "日期，格式：YYYY-MM-DD，默认为今天"

                    }

                },

                "required": ["location"]

            }

        },

        {

            "name": "calculate_sum",

            "description": "计算一组数字的总和",

            "parameters": {

                "type": "object",

                "properties": {

                    "numbers": {

                        "type": "array",

                        "items": {"type": "number"},

                        "description": "需要求和的数字列表"

                    }

                },

                "required": ["numbers"]

            }

        },

        {

            "name": "get_stock_price",

            "description": "获取指定股票的价格信息",

            "parameters": {

                "type": "object",

                "properties": {

                    "symbol": {

                        "type": "string",

                        "description": "股票代码，例如：AAPL"

                    },

                    "date": {

                        "type": "string",

                        "description": "日期，格式：YYYY-MM-DD，默认为今天"

                    }

                },

                "required": ["symbol"]

            }

        }

    ]

    client = DeepSeekClient(api_key)

    messages = [{"role": "user", "content": query}]

    # 主对话循环

    while True:

        # 调用DeepSeek API

        response = client.chat_completion(messages, functions)

        if "error" in response:

            print(f"API错误: {response['error']}")

            break

        assistant_message = response["choices"][0]["message"]

        messages.append(assistant_message)

        # 检查是否有函数调用

        if "function_call" in assistant_message:

            function_call = assistant_message["function_call"]

            function_name = function_call["name"]

            parameters = function_call.get("parameters", {})

            # 执行函数调用

            try:

                function_response = PythonFunctionCaller.execute_function_call(

                    function_name, parameters

                )

            except Exception as e:

                function_response = {"error": f"函数执行错误: {str(e)}"}

            # 将函数调用结果添加到对话中

            messages.append({

                "role": "function",

                "name": function_name,

                "content": json.dumps(function_response)

            })

            print(f"执行函数: {function_name}")

            print(f"参数: {parameters}")

            print(f"结果: {function_response}")

        else:

            # 没有函数调用，直接显示AI回复

            print(f"AI回复: {assistant_message['content']}")

            break

## 4.5 评估与优化

在智能体研发中，度量与改进是提升效果和体验的核心。度量用来量化当前表现，为后续调整提供依据；改进则依据度量结果对提示、流程或接入能力进行优化。

### 4.5.1 提示效果评估指标

在 LLM 应用里，提示质量直接决定输出水平。常用的评估维度如下。

#### 1. 基础评估指标
1) 响应相关性 Relevance
   - 判断输出是否契合用户的真实意图。
   - 采用 1 到 5 分，分数越高越贴切。
2) 信息完整性 Completeness
   - 看一次回答是否覆盖提问的所有要点。
   - 采用 1 到 5 分，分数越高越全面。
3) 流畅度 Fluency
   - 评估文本是否通顺自然。
   - 采用 1 到 5 分，分数越高越流畅。

#### 2. 高级评估指标
1) 事实准确性 Factual Accuracy
   - 检验陈述是否属实，需要借助外部知识核对。
2) 一致性 Consistency
   - 评估回答内部逻辑是否统一，以及与历史对话的连贯程度。
3) 多样性 Diversity
   - 针对创意类任务，衡量输出的多样程度。
4) 毒性/偏见评分 Toxicity/Bias
   - 检测是否出现有害内容或偏见表述。

#### 3. 任务特定指标
1) 代码正确性 Code Correctness
   - 对代码生成任务，检验能否运行且逻辑正确。
2) 推理质量 Reasoning Quality
   - 对需要逻辑推导的任务，评估推导链条是否合理。
