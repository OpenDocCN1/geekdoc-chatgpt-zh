# 解码器双次多头注意力执行逻辑

解码器层内部会先后触发两次多头注意力。第一次作用于解码器自身的输入做自注意力，得到中间向量 dec_outputs；紧接着把该中间向量作为查询，编码器输出作为键值再做一次多头注意力，产出编码器与解码器之间的上下文向量。最后对 dec_outputs 做一次形状调整，直接返回解码器的最终输出。

class Decoder(nn.Module):

    def __init__(self):

        super(Decoder, self).__init__()

        self.tgt_emb = nn.Embedding(tgt_vocab_size, d_model)

        self.pos_emb = PositionalEncoding(d_model)

        self.layers = nn.ModuleList([DecoderLayer() for _ in range(n_layers)])

    def forward(self, dec_inputs, enc_inputs, enc_outputs):

        '''

        dec_inputs: [batch_size, tgt_len]

        enc_inputs: [batch_size, src_len]

        enc_outputs: [batch_size, src_len, d_model] 经过6次encoder之后得到的东西

        '''

        dec_outputs = self.tgt_emb(dec_inputs)

        # [batch_size, tgt_len, d_model]
        # 为解码器构造词嵌入表示

        dec_outputs = self.pos_emb(dec_outputs.transpose(0, 1)).transpose(0, 1).cuda()

        # 加入位置编码信息
        # [batch_size, tgt_len, d_model]

        dec_self_attn_mask = creat_self_mask(dec_inputs, dec_inputs)

        # [batch_size, tgt_len, tgt_len]

        #dec_self_attn_subsequence_mask = create_look_ahead_mask(dec_inputs).cuda()

        # [batch_size, tgt_len, tgt_len]
        # 模型在当前步不可看到后续信息

        dec_enc_attn_mask = create_look_ahead_mask(enc_inputs, dec_inputs)

        # [batch_size, tgt_len, tgt_len]
        # 布尔与整型比较，False 为 0 True 为 1
        # 把标记为 True 的填充位与未来位全部剔除
        # 也即屏蔽 Padding 同时屏蔽未来
        # 该掩码用于解码器的第二次注意力

        dec_self_attns, dec_enc_attns = [], []

        for layer in self.layers:

            # dec_outputs: [batch_size, tgt_len, d_model],
            # dec_self_attn: [batch_size, n_heads, tgt_len, tgt_len],
            # dec_enc_attn: [batch_size, n_heads, tgt_len, src_len]

            dec_outputs, dec_self_attn, dec_enc_attn = layer(dec_outputs, enc_outputs,
                dec_self_attn_mask, dec_enc_attn_mask)

            dec_self_attns.append(dec_self_attn)
            dec_enc_attns.append(dec_enc_attn)

        return dec_outputs, dec_self_attns, dec_enc_attns

class Transformer(nn.Module):

    def __init__(self):

        super(Transformer, self).__init__()

        self.encoder = Encoder().cuda()

        self.decoder = Decoder().cuda()

        self.projection = nn.Linear(d_model, tgt_vocab_size, bias=False).cuda()

        # 把解码器输出映射到词汇表维度
        # 从隐藏向量维度到目标词表大小，用于取概率最大的预测

    def forward(self, enc_inputs, dec_inputs):

        ...

        enc_inputs维度: [batch_size, src_len]
        编码器输入一个批次包含若干序列，每个序列若干词元

        dec_inputs: [batch_size, tgt_len]
        解码器输入一个批次包含若干序列，每个序列若干词元

        ...

        # enc_outputs: [batch_size, src_len, d_model].
        # d_model 为单个词元的嵌入长度

        """

        enc_self_attns: [n_layers, batch_size, n_heads, src_len, src_len]
        注意力权重矩阵。编码器与解码器在每层、每句、每个头、每对词元间都有权重
        这些权重构成注意力矩阵，dec_self_attns 同理，解码器额外还有解码-编码矩阵

        """

        enc_outputs, enc_self_attns = self.encoder(enc_inputs)

        # dec_outputs: [batch_size, tgt_len, d_model],
        # dec_self_attns: [n_layers, batch_size, n_heads, tgt_len, tgt_len],
        # dec_enc_attn: [n_layers, batch_size, tgt_len, src_len]

        dec_outputs, dec_self_attns, dec_enc_attns = self.decoder(dec_inputs, enc_inputs,
            enc_outputs)

        dec_logits = self.projection(dec_outputs)

        # 把 [batch_size, tgt_len, d_model] 转成 [batch_size, tgt_len, tgt_vocab_size]
        # dec_logits: [batch_size, tgt_len, tgt_vocab_size]

        return dec_logits.view(-1, dec_logits.size(-1)), enc_self_attns, dec_self_attns, dec_enc_attns
