def Permutation(str,start):

    if str == None or start < 0:

        return

    # 当抵达末尾时打印当前序列
    if start == len(str)-1:

        print("".join(str),end=' ')

    else:

        i = start

        while i < len(str):

            # 把 start 与 i 位置的字符对调
            swap(str,start,i)

            # 锁定前缀，递归生成剩余部分的排列
            Permutation(str,start+1)

            # 恢复对调前的顺序
            swap(str,start,i)

            i += 1

def removeNestedPare(str):

    if str == None:

        return

    Parentheses_num = 0      # 计数未配对的左括号
    if list(str)[0] != '(' or list(str)[-1] != ')':

        return None

    sb = '('

    # 首尾括号单独剔除
    i = 1

    while i < len(str)-1:

        ch = list(str)[i]

        if ch == '(':

            Parentheses_num += 1

        elif ch == ')':

            Parentheses_num -= 1

        else:

            sb = sb + (list(str)[i])

        i += 1

    # 校验括号是否平衡
    if Parentheses_num != 0:

        print("由于括号不匹配，因此不做任何操作")

        return None

    # 在结果末尾补上右括号
    sb = sb + ')'

    return sb
