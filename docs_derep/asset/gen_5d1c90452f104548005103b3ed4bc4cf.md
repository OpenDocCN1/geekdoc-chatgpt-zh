import requests

def generate_response(prompt):

    response = requests.post(

        'http://localhost:11434/api/generate',

        json={

            'model': 'deepseek-r1:1.5b',  # 替换为模型名称

            'prompt': prompt,

            'stream': False

        }

    )

    return response.json()['response']

##### 快速试用
