# 企业智能问答服务搭建与验证实战

基于 Flask 搭建的问答服务对外暴露 /query 接口，接受 POST 请求。服务启动时先实例化向量库，请求进来后先校验 context 与 question 是否齐全，缺失则返回 400 错误。校验通过后对上下文做预处理得到向量表达，调用向量库检索最相似的 3 条记录并打印距离与索引，随后把上下文和问题送入生成模型得到答案，最后以 JSON 形式返回。

服务默认监听 0.0.0.0:5000，便于容器化部署。整体依赖集中在 PyTorch 生态与向量检索、Web 框架等。

from flask import Flask, request, jsonify
import torch
from model import generate_answer
from database import VectorDatabase
from utils import preprocess_data

server = Flask(__name__)

vector_store = VectorDatabase()

@server.route('/query', methods=['POST'])
def handle_query():
    """接收查询并返回答案的接口"""
    payload = request.json
    ctx = payload.get('context')
    q = payload.get('question')
    if not ctx or not q:
        return jsonify({'error': 'context and question are required'}), 400

    vec = preprocess_data([ctx])
    dists, idxs = vector_store.search(vec, k=3)
    print('检索结果:', dists, idxs)

    resp = generate_answer(ctx, q)
    return jsonify({'answer': resp})

if __name__ == '__main__':
    server.run(host='0.0.0.0', port=5000)

    for _ in range(num_requests):

        # 模拟生成回答
