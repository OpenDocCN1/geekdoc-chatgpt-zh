## 上电连通性自检

第一次部署建议先跑一次连通校验。自检函数会用 GPIO Zero 新建一个低电平触发的 LED 继电器和数字输入湿度传感器，让继电器吸合两秒再断开，之后连续读取三次传感器电平并换算成干燥或湿润，最后主动释放资源并返回校验结果。

def test_gpiozero():

    print("正在测试 GPIO Zero 访问...")

    try:

        # 初始化设备

        relay = LED(18, active_high=False)  # 大多数继电器模块是低电平触发

        moisture_sensor = DigitalInputDevice(17)

        # 测试继电器

        print("测试继电器控制 - 打开")

        relay.on()  # 激活继电器

        time.sleep(2)

        print("测试继电器控制 - 关闭")

        relay.off()  # 关闭继电器

        # 测试湿度传感器

        print("测试湿度传感器读取:")

        for i in range(3):

            reading = moisture_sensor.value

            print(f" 读数 {i+1}: {reading} ({'干燥' if reading else '湿润'})")

            time.sleep(1)

        # 清理资源

        relay.close()

        moisture_sensor.close()

        print("GPIO Zero 测试成功!")

        return True

    except Exception as e:

        print(f"GPIO Zero 测试失败: {str(e)}")

        return False

## 启动入口与双模式运行

主入口提供两种运行方式。加上 --test 参数只执行连通校验，成功提示系统可用，失败提示检查硬件。默认模式会先输出传感器引脚、继电器引脚、日志路径、检测间隔和浇水时长等运行信息，随后实例化控制器进入循环，可用 Ctrl+C 中断。

def run(self):

    try:

        while True:

            if self.is_soil_dry():

                self.start_irrigation()

                time.sleep(self.irrigation_duration)

                self.stop_irrigation()

            else:

                logging.info("土壤湿润，无须灌溉")

            for _ in range(self.check_interval):

                time.sleep(1)

    except KeyboardInterrupt:

        self.stop_irrigation()
