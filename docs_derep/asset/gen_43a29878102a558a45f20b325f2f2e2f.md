class Blackboard:

    # 供回溯查找的父级黑板

    parent: Blackboard

    # 以关联数组形式存储的数据载荷

    data: HashTable[string -> any]

    function get(name: string) -> any:

        if name in data: return data[name]

        else if parent: return parent.get(name)

        else: return null
