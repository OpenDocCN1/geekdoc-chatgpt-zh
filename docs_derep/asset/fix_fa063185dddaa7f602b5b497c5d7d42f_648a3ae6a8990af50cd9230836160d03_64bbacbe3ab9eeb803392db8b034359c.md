# 冷启动推荐：向量匹配与双协议部署

当新用户还没有行为数据时，可先用内容语义做启发式筛选。把用户兴趣向量与冷启动素材向量送入余弦相似度计算，得到一维得分序列，将该分数写入文章表作为匹配字段，再按从大到小排序即可得到初步的个性化顺序。随后对用户画像表逐行遍历，调用同样的匹配流程并输出文章编号、文本描述与分数，便于人工校验。

示例池里只有三条素材，分别覆盖科技、健康、时尚。运算后科技偏好的用户在人工智能趋势文上得分最高，健康偏好的用户对运动健身技巧文响应最强，时尚偏好的用户则更青睐搭配秘诀。分数分布直接反映了兴趣与内容的贴合，也说明向量粗筛在冷启动阶段具备可行性。

为了让同一套推荐能力同时服务内部调用与外部前端，采用一核双协议的部署方式。核心生成函数目前返回模拟的五条商品，预留后续接入模型推理的位置。远程过程调用侧遵循 Protobuf 契约实现服务接口，监听 50051；Web 侧基于 Flask 提供 /recommend 的 POST 接口，监听 5000。主程序通过线程并发启动两者，实现单进程双通道。

接口契约使用 Protocol Buffer 描述，服务名 Recommendation，包含 GetRecommendations 方法，请求携带 user_id 字符串，响应返回 recommendations 字符串。代码生成通过 grpc_tools.protoc 指定源码目录与输出目录完成。验证环节准备了统一客户端，分别以 insecure_channel 连接本地 50051 调用 Stub，及以 requests POST 到 http://localhost:5000/recommend。同一测试用户在两条通道上均返回相同的五条模拟商品，证明行为一致。

后续模型落地准备采用 ONNX 导出。示例定义了一个 SimpleRecommendationModel，包含用户与物品的 Embedding 以及全连接层，前向过程先查表得到嵌入向量再做逐元素交互，为推理加速奠基。

import grpc

from concurrent import futures

import time

import json

from flask import Flask, request, jsonify

import recommendation_pb2

import recommendation_pb2_grpc

###### 推荐逻辑的核心功能

def generate_recommendations(user_id):

    # 模拟推荐逻辑，实际逻辑可以使用模型推理

    return [f"推荐商品_{i}" for i in range(1, 6)]

###### gRPC 服务实现

class RecommendationService(recommendation_pb2_grpc.RecommendationServicer):

    def GetRecommendations(self, request, context):

        user_id = request.user_id

        recommendations = generate_recommendations(user_id)

        return recommendation_pb2.RecommendationResponse(

            recommendations=json.dumps(recommendations)

        )

###### gRPC 服务启动

def start_grpc_server():

    server = grpc.server(futures.ThreadPoolExecutor(max_workers=10))

    recommendation_pb2_grpc.add_RecommendationServicer_to_server(

        RecommendationService(), server

    )

    server.add_insecure_port('[::]:50051')

    print("gRPC 服务已启动，端口: 50051")

    server.start()

    server.wait_for_termination()

###### Flask 服务实现

app = Flask(__name__)

@app.route('/recommend', methods=['POST'])

def recommend():

    data = request.get_json()

    user_id = data.get('user_id')

    recommendations = generate_recommendations(user_id)

    return jsonify({"recommendations": recommendations})

###### Flask 服务启动

def start_flask_server():

    app.run(port=5000)

if __name__ == '__main__':

    import threading

    threading.Thread(target=start_grpc_server).start()

    start_flask_server()

def test_grpc_call(user_id):

    channel=grpc.insecure_channel("localhost:50051")

    stub=recommendation_pb2_grpc.RecommendationStub(channel)

    response=stub.GetRecommendations(recommendation_pb2.UserRequest(

        user_id=user_id))

    print("gRPC 推荐结果:", response.recommendations)

        "http://localhost:5000/recommend",
