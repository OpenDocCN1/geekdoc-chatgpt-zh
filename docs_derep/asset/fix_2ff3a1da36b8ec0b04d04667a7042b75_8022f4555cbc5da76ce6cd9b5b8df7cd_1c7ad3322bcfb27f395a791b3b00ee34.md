# 4. 引入 Docker 官方仓库
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# 5. 安装 Docker 核心组件
sudo apt install -y docker-ce docker-ce-cli containerd.io

# 6. 验证 Docker 是否安装成功并正常运行
sudo systemctl status docker

# 7. 配置普通用户使用 Docker（无须每次输入 sudo）
sudo usermod -aG docker $USER

# 8. 设置 Docker 开机自启（服务器必做）
sudo systemctl enable docker

**注意：**执行完上述命令后，必须退出当前终端并重新登录，用户权限配置才能生效。

2．Docker Compose 安装

Ubuntu 自带软件源中的 Compose 版本普遍落后，兼容性差，建议直接拉取官方最新发行版，流程如下：

# 1. 下载最新版 Docker Compose
sudo curl -L "https://github.com/docker/compose/releases/latest/download/docker-compose-$(uname-s)-$(uname-m)" -o /usr/local/bin/docker-compose

# 2. 给 Docker Compose 添加执行权限
sudo chmod +x /usr/local/bin/docker-compose

# 3. 验证 Docker Compose 是否安装成功
docker-compose --version

在部署 OpenClaw 之前，先把官方镜像拉到本地。镜像完整地址为 ghcr.io/openclaw/openclaw:latest，latest 代表当前稳定版。执行 docker pull ghcr.io/openclaw/openclaw:latest 后，Docker 引擎会从 GitHub 拉取镜像到本机。

运行容器前要理解数据卷与目录映射的原理。Docker 容器本身是临时性环境，停止或删除后内部产生的一切文件都会消失，配置文件、日志同样丢失。为保证 OpenClaw 的核心资产如 openclaw.json、会话记录、已装技能插件能够长期保留，必须把宿主机目录挂载到容器内部。按官方说明，容器内数据集中在 /root/.openclaw。

### 8.4.3 撰写 Docker Compose 配置
用一条冗长的 docker run 命令也能起容器，但参数繁多且不易维护。工程实践里更推荐使用 Docker Compose，通过一份结构化的 YAML 文件统一描述服务。

在服务器上新建一个专门存放 OpenClaw 数据的目录，并在该目录下创建 docker-compose.yml。文件里定义一个名为 openclaw-gateway 的服务，指定官方镜像，并加上 restart: unless-stopped 策略，保证异常或重启后能自动恢复。

端口上把容器内部 3000 端口映射到宿主机 3000 端口，以便外部访问。数据卷是关键点，将宿主机当前目录下的 ./data 文件夹挂载到容器内部的 /root/.openclaw，实现持久化。

还可通过 environment 注入环境变量，例如把时区设为 Asia/Shenzhen，或把大模型 API Key 以变量形式传入，避免写进明文配置。最后用 command: ["gateway"] 指定启动网关。

OpenClaw 官网提供的标准 Docker Compose 配置如下：

services:
  openclaw-gateway:
    image: ${OPENCLAW_IMAGE:-openclaw:local}
    environment:
      HOME: /home/node
      TERM: xterm-256color
      OPENCLAW_GATEWAY_TOKEN: ${OPENCLAW_GATEWAY_TOKEN:-}
      OPENCLAW_ALLOW_INSECURE_PRIVATE_WS: ${OPENCLAW_ALLOW_INSECURE_PRIVATE_WS:-}
      CLAUDE_AI_SESSION_KEY: ${CLAUDE_AI_SESSION_KEY:-}
      CLAUDE_WEB_SESSION_KEY: ${CLAUDE_WEB_SESSION_KEY:-}
      CLAUDE_WEB_COOKIE: ${CLAUDE_WEB_COOKIE:-}
      TZ: ${OPENCLAW_TZ:-UTC}
    volumes:
      - ${OPENCLAW_CONFIG_DIR}:/home/node/.openclaw
      - ${OPENCLAW_WORKSPACE_DIR}:/home/node/.openclaw/workspace
    ports:
      - "${OPENCLAW_GATEWAY_PORT:-18789}:18789"
      - "${OPENCLAW_BRIDGE_PORT:-18790}:18790"
    init: true
    restart: unless-stopped
    command:
      [
        "node",
        "dist/index.js",
        "gateway",
        "--bind",
        "${OPENCLAW_GATEWAY_BIND:-lan}",
        "--port",
        "18789",
      ]
    healthcheck:
      test:
        [
          "CMD",
          "node",
          "-e",
          "fetch('http://127.0.0.1:18789/healthz').then((r)=>process.exit (r.ok?0:1)).catch(()=>process.exit(1))",
        ]
      interval: 30s
      timeout: 5s
      retries: 5
      start_period: 20s
  openclaw-cli:
    image: ${OPENCLAW_IMAGE:-openclaw:local}
    network_mode: "service:openclaw-gateway"
    cap_drop:
      - NET_RAW
      - NET_ADMIN
    security_opt:
      - no-new-privileges:true
    environment:
      HOME: /home/node
      TERM: xterm-256color
      OPENCLAW_GATEWAY_TOKEN: ${OPENCLAW_GATEWAY_TOKEN:-}
      OPENCLAW_ALLOW_INSECURE_PRIVATE_WS: ${OPENCLAW_ALLOW_INSECURE_PRIVATE_WS:-}
      BROWSER: echo
      CLAUDE_AI_SESSION_KEY: ${CLAUDE_AI_SESSION_KEY:-}
      CLAUDE_WEB_SESSION_KEY: ${CLAUDE_WEB_SESSION_KEY:-}
      CLAUDE_WEB_COOKIE: ${CLAUDE_WEB_COOKIE:-}
      TZ: ${OPENCLAW_TZ:-UTC}
    volumes:
      - ${OPENCLAW_CONFIG_DIR}:/home/node/.openclaw
      - ${OPENCLAW_WORKSPACE_DIR}:/home/node/.openclaw/workspace
    stdin_open: true
    tty: true
    init: true
    entrypoint: ["node", "dist/index.js"]
    depends_on:
      - openclaw-gateway

### 8.4.4 启动与管理服务
配置文件写好后，启动 OpenClaw 变得非常简洁。在 docker-compose.yml 所在目录执行 docker-compose up -d，Compose 会以守护模式在后台创建并运行容器。

运行期间排查问题可执行 docker-compose logs -f 实时查看日志。若要做初始化配置或安装插件，不能直接在宿主机执行，需先用 docker exec -it openclaw-gateway /bin/bash 进入运行中的容器，再在容器内执行 openclaw configure 等命令。

版本更新很流畅：先在同目录执行 docker-compose pull 拉取新镜像，再执行 docker-compose up -d 用新镜像重建容器。借助数据卷挂载，./data 中的配置与历史数据会完整保留并自动接入新版本。

# 第九章 企业协同工具接入
第八章已把 OpenClaw 部署到云端，具备持续待命能力。但一个优秀的 AI Agent 不应孤立运行，而应融入日常沟通场景。OpenClaw 的 Channel 架构支持主流聊天平台，本章重点讲解如何接入国内三大协同工具：飞书、企业微信、钉钉。

## 9.1 飞书接入实战
飞书由字节跳动打造，以流畅体验和开放平台能力著称。OpenClaw 对飞书提供原生支持，核心在于采用飞书开放平台的 WebSocket 长连接事件订阅，而非传统 Webhook。这样无需在服务器暴露公网 URL，也省去反向代理与 SSL 配置，安全性更高。

### 9.1.1 创建企业自建应用与机器人
首先在飞书开放平台创建自建应用作为身份凭证。登录后进入开发者后台，点击创建企业自建应用，填写名称如“OpenClaw 智能助手”、描述并上传图标。

创建成功后进入详情页，在凭证与基础信息中获取两个关键参数：
• App ID，通常以 cli_ 开头；
• App Secret，应用密钥。
这两个值相当于机器人的账号密码，后续校验身份时会用到，需妥善保存。

随后在添加应用功能中打开机器人开关，机器人即成为 OpenClaw 在飞书里的交互入口。

### 9.1.2 权限申请与事件订阅配置
创建应用后需申请 API 权限。在开发配置 → 权限管理 → 应用身份权限中，可批量导入所需权限。以下 JSON 涵盖机器人运行的核心范围：

{
  "scopes": {
    "tenant": [
      "im:chat.members:bot_access",
      "im:message",
      "im:message.group_at_msg:readonly",
      "im:message.p2p_msg:readonly",
      "im:message:readonly",
      "im:message:send_as_bot",
      "im:resource"
    ],
    "user": [
      "contact:contact.base:readonly"
    ]
  }
}

其中 user 下的 contact:contact.base:readonly 用于只读访问企业通讯录基础信息，是机器人识别用户身份、完成上下文匹配的前提。若未开通，用户发起指令时机器人将因缺少数据而失效。
