    "title": "职场女性就该当「乖学生」？",

        "都不是！"

    ],

    "tags": [

        "自我觉醒",

        "价值提升",

        "职场真相"

    ]

}

async def main(args: Args) -> Output:
    params = args.params
    input = params.get("input")
    length = len(input)
    # 构建输出对象
    ret: Output = {
        "length" : length
    }
    return ret
