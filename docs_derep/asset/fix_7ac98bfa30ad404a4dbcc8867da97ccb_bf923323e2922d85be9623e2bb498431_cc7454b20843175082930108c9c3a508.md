GPU 数据搬运提速：三组实测拆解

训练与推理里，算力被数据搬运拖住是常见现象。CPU 侧到显存、显存到显存的走法不一样，实测帧率就会差很多。下面用三段可复现的实验，把常见的加速手段效果摊开来看。

CPU 到 GPU：页锁定带来的实感加速
先造一个 1000 万个 float32 的随机张量，分别用普通可分页内存和先 pin 好的内存各搬一次到同一张卡。普通路径前后做一次 cuda 同步，锁定路径打开 non_blocking 再同步，搬完后用范数比对数值。
在千万元素量级，普通路径明显更慢，页锁定版本更快，数值差异是零，说明提速并不丢精度。页锁定让 DMA 直接读写主机页，省掉了系统里的一次中间拷贝，这正是主机到设备提速的关键。

多卡协同：CPU 回环与 GPU 直连的差距
多卡配合时走法不同结果差很多。一种是把 GPU0 的张量先写回 CPU 再送到 GPU1，走 PCIe 回环；另一种是 GPU0 直接推给 GPU1，借助 NVLink 直连并配合非阻塞拷贝。
在 10MB 到 1000MB 五个档位计时，直连路径的耗时基本是回环路径的一半左右，提速倍数稳定在 2.4 到 2.5 倍。随着规模增大，耗时线性增长，倍数却基本保持。

import torch

import time

def transfer_between_gpus(size_in_mb):

    # 确保至少有两个 GPU 可用

    if torch.cuda.device_count() < 2:

        print("需要至少两个 GPU 来运行本示例。")

        return

    # 定义设备

    device0 = torch.device('cuda:0')

    device1 = torch.device('cuda:1')

    # 创建一个大的张量并将其移动到 GPU 0 上

    tensor_size = size_in_mb * 1024 * 1024 // 4  # float32 占 4 字节

    tensor = torch.randn(tensor_size, dtype=torch.float32, device=device0)

    # 同步设备，确保所有操作完成

    torch.cuda.synchronize()

    # 使用 PCIe 进行数据传输

    start_time_pcie = time.time()

    tensor_cpu = tensor.to('cpu')

    tensor_gpu1_pcie = tensor_cpu.to(device1)

    torch.cuda.synchronize()

    end_time_pcie = time.time()

    time_pcie = end_time_pcie - start_time_pcie

    # 使用 NVLink 进行数据传输（假设 GPU 0 和 GPU 1 通过 NVLink 互联）

    # 直接在 GPU 间传输

    start_time_nvlink = time.time()

    tensor_gpu1_nvlink = tensor.to(device1, non_blocking=True)

    torch.cuda.synchronize()

    end_time_nvlink = time.time()

    time_nvlink = end_time_nvlink - start_time_nvlink

    # 输出结果

    print(f"数据传输大小：{size_in_mb} MB")

    print(f"通过 PCIe 传输的时间：{time_pcie:.6f} 秒")

    print(f"通过 NVLink 传输的时间：{time_nvlink:.6f} 秒")

    print(f"速度提升：{time_pcie / time_nvlink:.2f} 倍")

def main():

    # 定义传输数据大小（单位为 MB）

    sizes = [10, 50, 100, 500, 1000]

    for size in sizes:

        print("\n----------------------------------------")

        transfer_between_gpus(size)

if __name__ == "__main__":

    main()

TensorRT 推理里缓冲区的准备思路
TensorRT 推理流程中，主机侧与设备侧的内存要提前准备好。加载序列化引擎后，按绑定遍历，依据形状和批量大小算出所需元素数，用对应 numpy 类型申请页锁定主机内存，再用 cuda.mem_alloc 申请等量的设备内存。输入输出分开归类，bindings 统一成设备指针的整数句柄，并建一个 CUDA 流供后续异步拷贝。
这种先用页锁定主机缓冲区的做法，正是前面提到的提速原理在推理框架里的落地，能让 H2D 拷贝摆脱额外复制，成为高吞吐推理的基础。

import tensorrt as trt

import pycuda.driver as cuda

import pycuda.autoinit

import numpy as np

# 定义 TensorRT 日志记录器

TRT_LOGGER = trt.Logger(trt.Logger.WARNING)

def load_engine(trt_runtime, engine_path):

    """加载序列化的 TensorRT 引擎"""

    with open(engine_path, 'rb') as f:

        engine_data = f.read()

    engine = trt_runtime.deserialize_cuda_engine(engine_data)

    return engine

def allocate_buffers(engine):

    """为输入和输出分配 CUDA 缓冲区"""

    inputs = []

    outputs = []

    bindings = []

    stream = cuda.Stream()

    for binding in engine:

        size = trt.volume(engine.get_binding_shape(binding)) * engine.max_batch_size

        dtype = trt.nptype(engine.get_binding_dtype(binding))

        # 分配主机内存

        host_mem = cuda.pagelocked_empty(size, dtype)

        # 分配设备内存

        device_mem = cuda.mem_alloc(host_mem.nbytes)

        # 添加到列表中

        bindings.append(int(device_mem))

        if engine.binding_is_input(binding):

            inputs.append({'host': host_mem, 'device': device_mem})

        else:

            outputs.append({'host': host_mem, 'device': device_mem})

    return inputs, outputs, bindings, stream
