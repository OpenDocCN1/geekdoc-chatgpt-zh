内部工具编排：从登记到调度

在内部系统里，新能力往往希望能零改动接入，调用时又要可控可审计。下面通过两个可运行的样例，把登记、检查、分发、缓存、记录这一整条链路串起来。

先看函数层面的登记与运行时调用。整体思路是先在注册表中确认名字是否存在，不存在就给出未登记提示；再拿 schema 对传入参数做合规检查，不通过就返回校验提示；找到映射的实现后执行，运行期间的异常统一封装成调用错误返回。

演示脚本先把两份示例 schema 写进注册表，随后打印当前可用的名称列表。用注入器进行四次试跑：第一次查询员工档案，带上编号 EMP12345，返回姓名 Alice、部门 HR、状态 Active 的完整数据；第二次处理发票，带上发票号 INV98765 和金额 1050.75，正常返回；第三次只给发票号不给金额，参数检查被拦截；第四次尝试调用一个从未登记的函数，直接得到未注册反馈。入口函数最后会统计总耗时，方便评估效率。

再看对外工具的调度层，这里补充了缓存复用和链路留痕两个工程手段。

外部调用被模拟成一个带随机等待的函数，用来体现网络抖动。结果缓存采用加锁的字典实现，键通过把工具名与请求内容拼接后求 SHA256 得到，相同请求只算一次。链路记录器写入时加锁，保存工具名、请求内容、返回结果、耗时和时间戳，便于事后排查。

调度器把缓存与记录融合：先算出键查缓存，命中就直接返回并输出命中提示；未命中则记下开始时间、执行真实调用、算出耗时、写入缓存并把本次调用写入链路，最后输出实际耗时。这种组合在重复请求频繁的场景下能压低尾延迟，同时保留完整的执行轨迹。

import time
import random
import threading
from queue import Queue
import hashlib
# 模拟外部工具调用函数
def external_tool_call(tool_name, request_data):
    """
    模拟调用外部工具的函数，返回处理结果。
    参数:
        tool_name: 工具名称 (str)
        request_data: 请求数据 (str)
    返回:
        result: 处理结果 (str)
    """
    # 模拟不同工具调用的延时
    delay = random.uniform(0.1, 0.3)
    time.sleep(delay)
    result = f"{tool_name} processed: {request_data}"
    return result
# 定义缓存类，用于存储函数调用结果
class ResultCache:
    def __init__(self):
        self.cache = {}
        self.lock = threading.Lock()
    def generate_key(self, tool_name, request_data):
        """
        根据工具名称和请求数据生成唯一缓存键。
        参数:
            tool_name: 工具名称 (str)
            request_data: 请求数据 (str)
        返回:
            key: 唯一字符串键
        """
        hash_input = f"{tool_name}:{request_data}"
        key = hashlib.sha256(hash_input.encode('utf-8')).hexdigest()
        return key
    def get(self, key):
        """
        根据缓存键获取缓存结果。
        参数:
            key: 缓存键 (str)
        返回:
            result: 缓存中存储的结果，如果不存在，则返回 None
        """
        with self.lock:
            return self.cache.get(key, None)
    def set(self, key, result):
        """
        将结果存入缓存。
        参数:
            key: 缓存键 (str)
            result: 要缓存的结果
        """
        with self.lock:
            self.cache[key] = result
# 定义调用链追踪器，记录工具调用路径和时间信息
class CallChainTracker:
    def __init__(self):
        self.chain = []
        self.lock = threading.Lock()
    def record_call(self, tool_name, request_data, result, exec_time):
        """
        记录一次工具调用的详细信息。
        参数:
            tool_name: 工具名称 (str)
            request_data: 请求数据 (str)
            result: 工具调用结果 (str)
            exec_time: 执行时间 (float)
        """
        record = {
            "tool": tool_name,
            "request": request_data,
            "result": result,
            "time": exec_time,
            "timestamp": time.strftime("%Y-%m-%d %H:%M:%S", time.localtime())
        }
        with self.lock:
            self.chain.append(record)
    def get_chain(self):
        """
        返回完整的调用链记录。
        """
        with self.lock:
            return self.chain.copy()
# 定义工具调用调度器，整合工具调用链追踪与结果缓存优化机制
class ToolCallDispatcher:
    def __init__(self):
        self.cache = ResultCache()
        self.tracker = CallChainTracker()
    def call_tool(self, tool_name, request_data):
        """
        调度工具调用。首先检查缓存，如果存在，则直接返回缓存结果，
        否则调用外部工具并记录调用链信息。
        参数：
            tool_name: 工具名称 (str)
            request_data: 请求数据 (str)
        返回：
            result: 工具调用结果 (str)
        """
        key = self.cache.generate_key(tool_name, request_data)
        cached_result = self.cache.get(key)
        if cached_result is not None:
            print(f"[Cache Hit] {tool_name} with request: {request_data}")
            return cached_result
        # 未命中缓存，执行调用
        start = time.time()
        result = external_tool_call(tool_name, request_data)
        exec_time = time.time() - start
        # 存入缓存
        self.cache.set(key, result)
        # 记录调用链
        self.tracker.record_call(tool_name, request_data, result, exec_time)
        print(f"[Call] {tool_name} processed request: {request_data} in {exec_time:.4f} seconds")
