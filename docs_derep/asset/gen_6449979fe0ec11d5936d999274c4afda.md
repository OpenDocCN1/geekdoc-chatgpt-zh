from gpiozero import LED, DigitalInputDevice
import time
import logging
from datetime import datetime

# 初始化日志文件与格式
log_file = 'irrigation_log3.txt'
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    filename=log_file
)

# 同时在终端打印日志
console = logging.StreamHandler()
console.setLevel(logging.INFO)
formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
console.setFormatter(formatter)
logging.getLogger('').addHandler(console)

class IrrigationController:
    def __init__(self):
        # 硬件引脚映射
        self.MOISTURE_SENSOR_PIN = 17
        self.RELAY_PIN = 18

        # 通过库创建设备对象
        try:
            self.relay = LED(self.RELAY_PIN, active_high=True, initial_value=False)
            # self.relay = LED(self.RELAY_PIN, active_high=False, initial_value=False)
            self.moisture_sensor = DigitalInputDevice(self.MOISTURE_SENSOR_PIN)

            logging.info("GPIO端口配置完毕")
            print("GPIO端口配置完毕")
        except Exception as e:
            logging.error(f"GPIO初始化异常: {str(e)}")
            print(f"GPIO初始化异常: {str(e)}")
            self.relay = None
            self.moisture_sensor = None

        # 运行参数
        self.check_interval = 2
        self.irrigation_duration = 15
        self.dry_threshold = 1

        logging.info("灌溉控制模块初始化完成")

    def read_moisture(self):
        """获取土壤湿度传感器的当前电平"""
        try:
            if self.moisture_sensor:
                return 1 if self.moisture_sensor.value else 0
            else:
                raise RuntimeError("湿度传感器未准备就绪")
        except Exception as e:
            logging.error(f"传感器读取出错: {str(e)}")
            return 0

    def is_soil_dry(self):
        """评估土壤干湿程度"""
        try:
            readings = []
            for _ in range(3):
                readings.append(self.read_moisture())
                time.sleep(1)

            avg_reading = sum(readings) / len(readings)
            is_dry = avg_reading >= 0.5
            status = "干燥" if is_dry else "湿润"
            logging.info(f"土壤判定为{status}，三次读数{readings}")
            return is_dry
        except Exception as e:
            logging.error(f"土壤状态判断失败: {str(e)}")
            return False

    def start_irrigation(self):
        """启动水泵供电"""
        try:
            if self.relay:
                logging.info("打开灌溉回路")
                self.relay.on()
                print("灌溉已启动")
            else:
                raise RuntimeError("继电器未就绪")
        except Exception as e:
            logging.error(f"灌溉启动失败: {str(e)}")
            print(f"灌溉启动失败: {str(e)}")

    def stop_irrigation(self):
        """关闭水泵供电"""
        try:
            if self.relay:
                logging.info("关闭灌溉回路")
                self.relay.off()
                print("灌溉已停止")
            else:
                raise RuntimeError("继电器未就绪")
        except Exception as e:
            logging.error(f"灌溉停止失败: {str(e)}")
            print(f"灌溉停止失败: {str(e)}")

    def run(self):
        """主循环控制"""
        logging.info("灌溉系统开始运行")
        print("灌溉系统开始运行")
        try:
            while True:
                if self.is_soil_dry():
                    self.start_irrigation()
                    time.sleep(self.irrigation_duration)
                    self.stop_irrigation()
                else:
                    logging.info("土壤湿度充足，跳过本次灌溉")
                    print("土壤湿度充足，跳过本次灌溉")

                logging.info(f"下次检测前休眠 {self.check_interval} 秒")
                print(f"下次检测前休眠 {self.check_interval} 秒")
                for _ in range(self.check_interval):
                    time.sleep(1)
        except KeyboardInterrupt:
            logging.info("用户手动终止程序")
            print("用户手动终止程序")
            self.stop_irrigation()
