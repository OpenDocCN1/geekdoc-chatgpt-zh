def run_conv(messages,
             api_key,
             tools=None,
             functions_list=None,
             model="deepseek-chat"):
    """
    支持自动调用外部函数的对话模型
    :param messages: 必需，传入给大模型的 messages
    :param api_key: 必需，用于调用大模型的 API Key
    :param tools: 可选，默认 None，可填入外部函数列表
    :param model: 可选，大模型名称，默认 deepseek-chat
    :return: 大模型最终生成的文本
    """
    user_messages = messages
    client = OpenAI(api_key=api_key,
                    base_url="https://api.deepseek.com")
    # 若未配置工具，则执行标准对话
    if tools == None:
        response = client.chat.completions.create(
            model=model,
            messages=user_messages
        )
        final_response = response.choices[0].message.content
    # 若已配置工具，则按需调用外部函数作答
    else:
        # 构建外部函数映射
        available_functions = {func.__name__: func for func in functions_list}
        # 准备本次对话的消息列表
        messages = user_messages
        # 首轮请求
        response = client.chat.completions.create(
            model=model,
            messages=user_messages,
            tools=tools,
        )
        response_message = response.choices[0].message
        # 读取模型选择的函数名称
        function_name = response_message.tool_calls[0].function.name
        # 定位对应的函数实现
        fuction_to_call = available_functions[function_name]
        # 解析模型返回的参数
        function_args = json.loads(response_message.tool_calls[0].function.arguments)
        # 执行函数并获取返回结果
        function_response = fuction_to_call(**function_args)
        # 将首轮模型回复追加到消息历史
        user_messages.append(response_message.model_dump())
        # 将工具执行结果以 tool 角色追加
        user_messages.append(
            {
                "role": "tool",
                "content": function_response,
                "tool_call_id": response_message.tool_calls[0].id
            }
        )
        # 再次调用大模型
        second_response = client.chat.completions.create(
            model=model,
            messages=user_messages)
        # 取出最终回复
        final_response = second_response.choices[0].message.content
    return final_response
