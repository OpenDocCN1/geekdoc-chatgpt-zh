###### 初始化OpenAI客户端

from openai import OpenAI

import os

client = OpenAI(

    base_url="https://api.deepseek.com",

    api_key=os.getenv("DEEPSEEK_API_KEY")

)

###### 设置查询

user_query = "查询太原市的AAAAA级景区及其当月游客量"

###### 准备生成SQL的提示词

prompt = f"""

以下是数据库的结构描述：

{schema_description}

用户的自然语言问题如下：

"{user_query}"

请注意：

1. scenic_spots表中的city字段存储的是城市名称，对应city_info表中的city_name

2. 两张表之间的关联应该使用city_name和city进行匹配

3. 请只返回SQL查询语句，不要包含任何解释、注释或格式标记（如sql）

"""

###### 调用大模型生成SQL语句

response = client.chat.completions.create(

    model="deepseek-chat",

    messages=[

        {"role": "system", "content": "请只返回SQL语句，不要包含任何Markdown格式或其他说明。"},

        {"role": "user", "content": prompt}

    ],

    temperature=0

)

###### 清理SQL语句，移除可能的Markdown标记

sql = response.choices[0].message.content.strip()

sql = sql.replace('sql', '').replace('```', '')

##### 4. 结合Agent、SQL-Data-Function和Tool Calls的方案

小雪：咖哥，Text2SQL在简单问题上挺好用，可是一碰到需要多步推理、业务约束比较多的情况，生成的SQL经常不对，稳定性也很难保证。实际落地有什么更稳的思路吗？

咖哥：我跟业内聊过不少案例，确实这是Text2SQL的痛点。经过反复迭代，大家慢慢摸出了一条兼顾效率与可靠的中间路线。

> 这个思路来自咖哥的朋友卢向东在2024年6月掘金大会上的分享。他的公众号“土猛的员外”持续输出RAG落地的实战经验。

核心是让Agent的Tool Calls参见人民邮电出版社出版的《大模型应用开发 动手做AI Agent》。接管调度，配合事先封装好的数据函数Data-Function模板，里面预置了可复用的SQL，再加上经过人工补充的元数据，形成从自然语言到结构化操作的稳定通道。

小雪：元数据提取具体指什么？

咖哥：就是把数据库的表结构、字段含义、视图描述等信息抽出来，再人工补上业务上的关联说明和使用场景。这样加工后的元数据才适合做向量化检索。

小雪：数据函数又怎么理解？

咖哥：可以把它看成一套函数模板库，覆盖常见查询模式。给特定业务写数据函数，和写MyBatis、Hibernate这类ORM模板一个道理，把复杂的SQL逻辑封装起来，提升复用性和可控性。

整体流程是智能化调度：函数调用负责分发和执行，大模型负责理解意图，检索匹配的元数据和数据函数，完成单步或多步的SQL组装与查询。

下面的示例展示了如何用模块化工具定义来注册工具描述并实现具体逻辑。系统提示中会注入表字段的业务含义和关联关系，帮助模型更安全地完成查询。
