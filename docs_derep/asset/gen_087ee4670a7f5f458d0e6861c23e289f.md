class TextImageDataModule(LightningDataModule):

    def __init__(self,

                 folder: str,

                 batch_size: int,

                 num_workers=0,

                 image_size=224,

                 resize_ratio=0.75,

                 shuffle=False,

                 custom_tokenizer=None

    ):

        """ 从包含文本和图像文件的文件夹中创建一个文本图像数据模块

        参数：

            folder (str): 包含文本和图像文件的文件夹，它们通过各自路径的 stem 进行匹配

            batch_size (int): 每个数据加载器的批处理大小

            num_workers (int, optional): DataLoader 中的工作线程数，默认为 0

            image_size (int, optional): 输出图像的大小，默认为 224

            resize_ratio (float, optional): 裁剪时包含的最小图像比例，默认为 0.75

            shuffle (bool, optional): 是否在采样过程中进行打乱，默认为 False

            custom_tokenizer (transformers.AutoTokenizer, optional): 用于文本的分词器，默认为 None

        """

        super().__init__()

        self.folder = folder

        self.batch_size = batch_size

        self.num_workers = num_workers

        self.image_size = image_size

        self.resize_ratio = resize_ratio

        self.shuffle = shuffle

        self.custom_tokenizer = custom_tokenizer

@staticmethod

def add_argparse_args(parent_parser):

    parser = argparse.ArgumentParser(parents=[parent_parser], add_help=False)

    parser.add_argument(

        '--folder', type=str, required=True, help='你的训练文件夹的目录')

    parser.add_argument('--batch_size', type=int, help='批处理大小')

    parser.add_argument(

        '--num_workers', type=int, default=0, help='数据加载器的工作线程数')

    parser.add_argument(

        '--image_size', type=int, default=224, help='图像的大小')

    parser.add_argument(

        '--resize_ratio', type=float, default=0.75,

        help='随机裁剪时图像的最小尺寸')

    parser.add_argument(

        '--shuffle', type=bool, default=False, help='采样时是否打乱顺序')

    return parser

def setup(self, stage=None):

    self.dataset = TextImageDataset(

        self.folder, image_size=self.image_size,

        resize_ratio=self.resize_ratio, shuffle=self.shuffle,

        custom_tokenizer=not self.custom_tokenizer is None)

def train_dataloader(self):

    return DataLoader(

        self.dataset, batch_size=self.batch_size,

        shuffle=self.shuffle, num_workers=self.num_workers,

        drop_last=True, collate_fn=self.dl_collate_fn)

def dl_collate_fn(self, batch):

    if self.custom_tokenizer is None:

        return torch.stack([row[0] for row in batch]),

               torch.stack([row[1] for row in batch])

    else:

        return torch.stack(

            [row[0] for row in batch]),

            self.custom_tokenizer([row[1] for row in batch],

            padding=True, truncation=True, return_tensors="pt")

下面对上述实现做一个通俗拆解：

1. TextImageDataset 的核心任务是把指定目录下的图像和文本成对读入，按文件名 stem 匹配，再做图像变换和文本分词，为下游模型准备干净的输入。
2. __init__ 负责实例化数据集，扫描文件夹并记录文件列表，同时把图像尺寸、裁剪比例、是否打乱、是否使用自定义分词器等超参固化下来。
3. __len__ 仅返回当前配对样本的总数，方便 DataLoader 估算迭代轮次。
4. fix_img 用来统一图像通道，确保所有读取到的图都被强制转为 RGB，避免不同模式导致的预处理不一致。
5. random_sample 在启用打乱时会被调用，从样本池里随机抽取一对图文。
6. sequential_sample 按索引顺序取下一个样本，走到末尾时自动回绕到开头，实现循环读取。
7. skip_sample 根据 shuffle 标志决定出问题的样本是随机跳过还是顺序跳过，保证数据流不中断。
8. __getitem__ 依据索引定位具体文件，先做图像解码与转换，再做文本分词，最后以张量形式返回图像和文本。
9. TextImageDataModule 是一个轻量的数据管理外壳，内部封装 TextImageDataset，并把 DataLoader 的各项配置集中管理。
10. 其 __init__ 保存数据根目录、批次大小、工作进程数、图像尺寸、resize_ratio、shuffle 开关以及自定义分词器引用。
11. add_argparse_args 是静态方法，用来把文件夹路径、批大小、工作线程、图像尺寸、裁剪比例、是否打乱等参数注册到命令行解析器。
12. setup 方法在训练阶段创建并初始化 TextImageDataset 实例，完成数据集就绪。
13. train_dataloader 依据已设定的批大小、打乱策略、工作线程数并开启 drop_last，返回可直接用于训练的数据加载器。
14. dl_collate_fn 是批拼接逻辑，默认情况下直接堆叠图像和文本张量；若提供了自定义分词器，则对文本列表统一做 padding、truncation 并返回张量。

### 8.2.3 创建模型
本示例的 models 目录里放着 CLIP 模型本体以及 Lightning 包装器的完整代码。CLIPWrapper 充当 CLIP 的 Lightning 训练验证外壳；CustomCLIPWrapper 在其基础上叠加了自蒸馏等定制优化手段，用来提升效果。配套的 RN.yaml 与 ViT.yaml 提供了不同视觉骨干的配置参数，整体构成可训练可评估的完整环境。

#### 1. 实现CLIP模型
文件 model.py 实现了 CLIP 架构。视觉编码器可选改进版 ResNet 或视觉 Transformer，文本编码器采用自研 Transformer 结构。代码中还定义了 Bottleneck、AttentionPool2d、ModifiedResNet、LayerNorm、QuickGELU、ResidualAttentionBlock、Transformer、VisualTransformer 等基础模块，分别负责特征提取、池化、归一化、非线性激活与注意力计算。顶层 CLIP 类把视觉和文本两路特征映射到同一语义空间，并通过余弦相似度衡量图文匹配程度，支撑多模态检索与对齐任务。
