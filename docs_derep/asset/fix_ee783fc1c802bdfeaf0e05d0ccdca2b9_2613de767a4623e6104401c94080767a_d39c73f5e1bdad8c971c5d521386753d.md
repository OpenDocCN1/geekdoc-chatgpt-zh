void checkWi-FiConnection( ) {

  if (Wi-Fi.status( ) != WL_CONNECTED) {

    Wi-Fi.reconnect( );               // 使用快速重连方法

    int attempts = 0;

    while (Wi-Fi.status( ) != WL_CONNECTED && attempts < 10) {

      feedWatchdog( );                 // 重连过程中持续喂狗

      delay(500);

      attempts++;

    }

  }

}
