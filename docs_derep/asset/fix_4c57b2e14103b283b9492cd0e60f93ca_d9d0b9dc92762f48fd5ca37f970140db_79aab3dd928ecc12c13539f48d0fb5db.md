def mins(a,b,c):
    mins = a if a < b else b
    mins = mins if mins < c else c
    return mins

def MinDistance2(a,b,c):
    curDist = 0
    minsd = 0
    minDist = 2**32
    i = 0                # i 指向序列 a 的当前位置
    j = 0                # j 指向序列 b 的当前位置
    k = 0                # k 指向序列 c 的当前位置
    while True:
        curDist = max(abs(a[i] - b[j]),abs(a[i] - c[k]),abs(b[j] - c[k]))
        if curDist < minDist:
            minDist = curDist
        # 定位三指针当前所指元素中的最优者
        minsd = mins(a[i],b[j],c[k])
        if minsd == a[i]:
            i += 1
            if i >= len(a):
                break
        elif minsd == b[j]:
            j += 1
            if j >= len(b):
                break
        else:
            k += 1
            if k >= len(c):
                break
    return minDist

def MinDistance3(a,b,c):
    MinSum = 0                # 记录目前得到的最优距离和
    Sum = 0                   # 当前三两距离之和的临时值
    MinOfabc = 0              # 三指针当前值的最优者
    cnt = 0                   # 迭代计数，合计不会超过三序列长度相加
    i = j = k = 0
    MinSum = (abs(a[i]-b[j])+abs(a[i]-c[k])+abs(b[j]-c[k])) // 2
    while cnt <= len(a)+len(b)+len(c):
        Sum = (abs(a[i]-b[j])+abs(a[i]-c[k])+abs(b[j]-c[k])) // 2
        MinSum = MinSum if MinSum < Sum else Sum
        MinOfabc = mins(a[i],b[j],c[k])      # 提取三指针当前值的最优者
        # 依据最优者来源决定对应指针递增
        if MinOfabc == a[i]:
            i += 1
            if i >= len(a):
                break
        if MinOfabc == b[j]:
            j += 1
            if j >= len(b):
                break
        if MinOfabc == c[k]:
            k += 1
            if k >= len(c):
                break
        cnt += 1
    return MinSum

def MatrixChainOrder1(p,i,j):
    if i == j:
        return 0
    mins = 2**32
    #""
    # 在首个可分点尝试所有括号划分方式
    # 子区间同样采用递归方式求解
    #""
    k = i
    while k < j:
        count = MatrixChainOrder1(p,i,k) + MatrixChainOrder1(p,k+1,j) + p[i-1]*p[k]*p[j]
        if count < mins:
            mins = count
        k += 1
    return mins

def MatrixChainOrder2(p,n):
    cost = [[None]*n for i in range(n)]
    i=1
    while i < n:
        cost[i][i]=0
        i+=1
    cLen =2
    while cLen < n:
        i=1
        while i < n-cLen+1:
            j = i+cLen-1
            cost[i][j]=2**31
            k = i
            while k <=j-1:
                q = cost[i][k]+cost[k+1][j]+p[i-1]*p[k]*p[j]
                if q < cost[i][j]:
                    cost[i][j]=q
                k +=1
            i+=1
        cLen +=1
    return cost[1][n-1]
