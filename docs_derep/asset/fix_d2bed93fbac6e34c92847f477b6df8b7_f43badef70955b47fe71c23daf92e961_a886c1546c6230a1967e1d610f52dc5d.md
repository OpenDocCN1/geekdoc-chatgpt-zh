#include <iostream>
using namespace std;
const int N = 1000;
int c[N];

常量 N=1000 用来限制数组最大长度；全局数组 c 用以保存中间计算结果。

int logic(int x, int y){
    return (x & y) ^ ((x ^ y) | (~x & y));
}

logic 通过按位组合实现特定逻辑。
（1）x & y 提取两数同为 1 的位。
（2）x ^ y 提取两数不同的位。
（3）~x & y 提取 y 为 1 且 x 为 0 的位。
（4）上述结果组合后返回。

void generate(int a, int b, int *c){
    for(int i=0;i<b;i++){
        c[i]=logic(a,i)%(b+1);
    }
}

generate 用来填充数组 c。遍历 0 到 b-1，对每个 i 先调用 logic(a,i) 再取模 b+1，保证值落在 [0,b]。

void recursion(int depth, int *arr, int size){
    if(depth<=0 || size<=1) return;
    int pivot=arr[0];
    int i=0,j=size-1;
    while(i<=j){
        while(arr[i]<pivot) i++;
        while(arr[j]>pivot) j--;
        if(i<=j){
            int temp=arr[i]; arr[i]=arr[j]; arr[j]=temp;
            i++; j--;
        }
    }
    recursion(depth-1, arr, j+1);
    recursion(depth-1, arr+i, size-i);
}

recursion 模拟分区排序。
终止条件为深度耗尽或子数组长度为 1。取首元素为基准，用双指针把小于基准的移到左边、大于的移到右边，随后对左右段递归，深度每次减一。

int main(){
    int a,b,d;
    cin>>a>>b>>d;
    generate(a,b,c);
    recursion(d,c,b);
    for(int i=0;i<b;++i) cout<<c[i]<<" ";
    cout<<endl;
}

主程序读入 a、b、d，先生成数组，再按指定深度进行分区，最后输出结果。

**【程序总结】**

输入 a、b、d，按位逻辑与取模生成数组 c；以受深度 d 约束的递归分区方式处理数组；输出处理后的序列。

**一、位运算**

按位与 &：对应位同为 1 才为 1。

**二、递归思想**

递归包含终止条件、自身调用、回传结果。
如求和 sum(n)=1+...+n，终止于 n==1，递归调用 sum(n-1)。

**三、快速排序分区思想**

选基准 pivot，双指针把小于基准的放左、大于的放右，递归处理左右子数组。
