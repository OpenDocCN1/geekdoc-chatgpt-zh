async def run_client():

    print("=== 客户端等待 5 秒后开始连接 ... ===")

    await asyncio.sleep(5)

    # MCP stdio transport

    from mcp.client.stdio import StdioServerParameters, stdio_client

    server_params = StdioServerParameters(

        command="python",

        args=[os.path.abspath(__file__)]

    )

    async with stdio_client(server_params) as (read, write):

        async with ClientSession(read, write) as session:

            print("[Client] Finance QA session init...")

            await session.initialize()

            # Step1: 查询金融术语

            glossary_res = await session.call_tool("tool_finance_glossary", {"term": "Yield Curve"})

            print("[Client] 金融术语查询:", glossary_res)

            # Step2: 模拟某产品收益

            product_sim = await session.call_tool("tool_product_simulation", {

                "product_id": "FundA",

                "principal": 10000,

                "years": 2

            })

            print("[Client] 产品收益模拟:", product_sim)

            # Step3: 查看用户账户

            account_view = await session.call_tool("tool_check_account", {"user_id": "U3002"})

            print("[Client] 用户账户信息:", account_view)

            # Step4: 风险评估问卷

            answers = ["中", "longterm"]

            risk_eval = await session.call_tool("tool_risk_survey", {

                "answers": answers

            })

            print("[Client] 风险偏好:", risk_eval)

            # Step5: 列出可投产品

            prod_list = await session.call_tool("tool_list_products", {})

            print("[Client] 可投产品列表:", prod_list)

async def main():

    import asyncio

    server_task = asyncio.create_task(run_server())

    client_task = asyncio.create_task(run_client())

    await asyncio.gather(server_task, client_task)

if __name__ == "__main__":

    asyncio.run(main())

这个 demo 把 MCP 在金融问答里的多工具协同跑了一遍。

1) tool_finance_glossary 用来检索专业名词的基础释义，这里演示了 Yield Curve 的查询。

整条链路里，MCP 把原本分散的能力封装成各自独立的 Tool，客户端只按名称传参即可依次调用，完成从术语解释、收益测算、账户查询、风险画像到产品盘点的多阶段金融服务。正是这种封装，让上下文能被结构化沉淀，业务被解耦，也让调用过程保持安全可控。

#### 10.1.3 智能体工作流平台的MCP落地方案

随着大模型与智能体一起演进，编排复杂任务、让多个智能体协作的工作流平台愈发关键。MCP 以标准化上下文协议的身份出现，兼具上下文建模、工具调用、资源调度和语义封装的能力，为搭建稳定、可扩展、可控的智能体工作流提供统一的交互与编排底座。基于 MCP 的平台目标是打通智能体间的语义鸿沟，让任务驱动下上下文能持续传递、状态可被追踪、操作可原子化执行，特别适合企业里高并发、多轮对话、异步任务链的管控场景。

在架构层面，智能体被视作 MCP 下的能力单元 Capability Entity。它的生命周期由上下文 Slot 承载，调度则依赖 MCP 工具系统 Tool API 触发。工作流引擎在此之上做任务分派、状态维护和路径规划，实际执行时智能体接收 ToolCall 指令，完成能力落地并回写反馈。为支持多阶段链路，MCP 支持状态注入，比如用 Slot 记录当前阶段、任务状态以及中间产物，从而维持语义一致性和上下文同步的可靠性。

任务一般由业务侧发起，平台通过 MCP 客户端发起构建请求，服务端依据已注册智能体的能力声明 Capabilities 匹配合适的执行体。这不是传统 RPC 的指令式调用，而是一种语义化的消息触发，天然带有上下文传递、资源注入和状态持久化的特性。一个任务可拆成多个有序或并行的步骤，每个步骤由一个或多个智能体承担，而 MCP 的工具调用机制让这些步骤以 Tool 的形式声明，并能热插拔地接入工作流。

平台还内置 Slot 生命周期管理，用来动态维护执行过程中产生的上下文，例如初始请求、智能体中间产出、最终结构化结果，都通过 Slot 的写入与更新实现跨智能体共享与边界隔离。再加上 MCP 标准里的错误治理能力，例如错误分级、异常恢复、上下文重试与回滚，让生产环境下智能体的协同变得可控、可观测。
