## 注意力模块实现

class Attention(torch.nn.Module):

    def __init__(self, fill: bool = True):

        super(Attention, self).__init__()

        self.scaling_coeff = 1.0 / torch.sqrt(torch.tensor(d_main, device=device))  # 点积缩放系数，稳定梯度
        self.lower_tri = torch.tril(torch.ones(1, 1, device=device)) if fill else None  # 因果掩码用的下三角矩阵
        self.linear_k = torch.nn.Linear(in_features=d_main, out_features=d_main, bias=False)  # K 投影层
        self.linear_q = torch.nn.Linear(in_features=d_main, out_features=d_main, bias=False)  # Q 投影层
        self.linear_v = torch.nn.Linear(in_features=d_main, out_features=d_main, bias=False)  # V 投影层
        self.linear_attention = torch.nn.Linear(in_features=d_main, out_features=d_main, bias=False)  # 输出线性映射层

    def forward(self, x, x_lengths):

        examples, length, dim = x.shape  # 拆解输入：样本数、序列长、特征维
        k = self.linear_k(x)  # K 投影，结果形状 [样本数, 序列长, 特征维]
        q = self.linear_q(x)  # Q 投影，结果形状 [样本数, 序列长, 特征维]
        v = self.linear_v(x)  # V 投影，结果形状 [样本数, 序列长, 特征维]

        scaled_dotproduct = (q @ k.transpose(-1, -2)) * self.scaling_coeff  # 缩放点积分数，形状 [样本数, 序列长, 序列长]

        if x_lengths is not None:  # 批次内序列长度不一致时
            numbers = torch.arange(length, device=device).unsqueeze(dim=0)  # 位置索引，形状 [1, 序列长]
            seq_mask_vector = x_lengths.unsqueeze(dim=-1) > numbers  # 长度掩码向量，形状 [样本数, 序列长]
            seq_mask_matrix = seq_mask_vector.unsqueeze(dim=-1) * seq_mask_vector.unsqueeze(dim=1)  # 二维掩码，形状 [样本数, 序列长, 序列长]
            seq_mask_matrix = seq_mask_matrix * self.lower_tri[:length, :length] if self.lower_tri is not None else seq_mask_matrix  # 叠加因果下三角
            masked_scaled_dotproduct = scaled_dotproduct.masked_fill(seq_mask_matrix == 0, float('-inf'))  # 掩码位置置负无穷
            weight = torch.nn.functional.softmax(masked_scaled_dotproduct, dim=-1)  # 行 softmax 得到注意力权重，形状 [样本数, 序列长, 序列长]
            weight = weight.masked_fill(seq_mask_vector.unsqueeze(dim=-1) == 0, 0)  # 填充位的权重置零
        else:  # 批次内所有序列等长时
            masked_scaled_dotproduct = scaled_dotproduct.masked_fill(self.lower_tri[:length, :length] == 0, float('-inf')) if self.lower_tri is not None else scaled_dotproduct  # 因果掩码，非法位置置负无穷
            weight = torch.nn.functional.softmax(masked_scaled_dotproduct, dim=-1)  # 行 softmax 得到注意力权重，形状 [样本数, 序列长, 序列长]

        agg = weight @ v  # 加权聚合上下文，形状 [样本数, 序列长, 特征维]
