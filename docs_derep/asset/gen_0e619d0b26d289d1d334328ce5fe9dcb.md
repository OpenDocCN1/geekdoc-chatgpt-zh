```python
# 随机抽取 24 张图片可视化检查
import numpy as np

idx = np.random.choice(np.arange(len(x_train)), 24, replace=False)
fig, axes = plt.subplots(nrows=4, ncols=6, figsize=(16, 9))

for ax, img, label in zip(axes.ravel(), x_train[idx], y_train[idx]):
    ax.imshow(img, cmap=plt.cm.gray_r)
    ax.set_xticks([])
    ax.set_yticks([])
    ax.set_title(label)

plt.tight_layout()
```
