# 时间换空间：Checkpoint 到底能省多少显存

训大模型时，限制往往不是算力，而是 GPU 内存。PyTorch 的梯度检查点就是一种显式的交换：前向时不完整保留中间激活，反向需要时再把对应片段重跑一遍。内存高峰被压下来，付出的代价是额外的计算。下面用两组可复现的实验看清取舍。

推理侧先看卷积网络。把主干拆成前半段和后半段，原始版走完整图，启用重算的版本把两段都包进 checkpoint。构造 16 张 224×224 的随机输入，连续跑 10 轮，每轮前后都清空缓存并同步，重置峰值统计，保证两次环境一致。整体来看，重算版的内存峰值明显回落，累计时长只小幅增加，很适合显存受限的推理任务。

class CheckpointedCNN(nn.Module):

    def __init__(self):

        super(CheckpointedCNN,self).__init__()

        self.layer1=nn.Sequential(

            nn.Conv2d(3,64,kernel_size=3,padding=1),

            nn.ReLU(),

            nn.Conv2d(64,64,kernel_size=3,padding=1),

            nn.ReLU(),

            nn.MaxPool2d(2)

        )

        self.layer2=nn.Sequential(

            nn.Conv2d(64,128,kernel_size=3,padding=1),

            nn.ReLU(),

            nn.Conv2d(128,128,kernel_size=3,padding=1),

            nn.ReLU(),

            nn.MaxPool2d(2)

        )

        self.fc=nn.Linear(128*56*56,1000)

    def forward(self,x):

        # 使用检查点函数来节省显存

import torch

import torch.nn as nn

import torch.utils.checkpoint as checkpoint

import time

# 定义一个深层神经网络模型

class DeepNet(nn.Module):

    def __init__(self):

        super(DeepNet,self).__init__()

        self.layers=nn.Sequential(

            nn.Linear(1024,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU(),

            nn.Linear(2048,1024),

            nn.ReLU(),

            nn.Linear(1024,10)

        )

    def forward(self,x):

        return self.layers(x)

# 定义使用梯度检查点的深层神经网络模型

class CheckpointedDeepNet(nn.Module):

    def __init__(self):

        super(CheckpointedDeepNet,self).__init__()

        self.layer1=nn.Sequential(

            nn.Linear(1024,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU()

        )

        self.layer2=nn.Sequential(

            nn.Linear(2048,2048),

            nn.ReLU(),

            nn.Linear(2048,2048),

            nn.ReLU()

        )

        self.layer3=nn.Sequential(

            nn.Linear(2048,1024),

            nn.ReLU(),

            nn.Linear(1024,10)

        )

    def forward(self,x):

        x=checkpoint.checkpoint(self.layer1,x)

        x=checkpoint.checkpoint(self.layer2,x)

        x=self.layer3(x)

        return x

def train_model(model,optimizer,input,target,num_epochs=5):

    model.train()

    for epoch in range(num_epochs):

        optimizer.zero_grad()

        output=model(input)

        loss=criterion(output,target)

        loss.backward()

        optimizer.step()
