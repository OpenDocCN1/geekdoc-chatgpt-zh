接下来进入状态定义和图初始化。State 用 TypedDict 声明，messages 字段借助 add_messages 做累加，这样每轮交互都能在图里顺畅传递。StateGraph 在这里被实例化，后面就可以往里挂节点了。

#Chapter9/Agent.ipynb

from typing import Annotated

from typing_extensions import TypedDict

from langgraph.graph import StateGraph, START, END

from langgraph.graph.message import add_messages

class State(TypedDict):

    #messages 的类型为 list

    #messages 通过 add_messages 方法来进行更新

    messages: Annotated[list, add_messages]

graph_builder = StateGraph(State)
模型接入口选用 ChatOpenAI，API Key 通过 getpass 交互式输入避免写入明文。chatbot 函数把当前 State 喂给模型并把结果写回 messages。add_node 第一个参数是节点唯一标识，第二个参数是可调用的处理函数。
[PRE2]
import getpass

import os

from langchain_openai import ChatOpenAI

os.environ["OPENAI_API_KEY"] = getpass.getpass()

llm = ChatOpenAI(temperature=0)

def chatbot(state: State):

    return {"messages": [llm.invoke(state["messages"])]}

#第 1 个参数必须是全局唯一的节点名

#第 2 个参数是这个节点可调用的一种方法或者对象

graph_builder.add_node("chatbot", chatbot)
把起点和终点连起来，最简路径就是 START 指向 chatbot，再从 chatbot 到 END，形成一条直线流程。

如果希望模型能做算术，先统一输入结构，再实现四则运算函数并包装成 StructuredTool，最后把工具集绑定到 LLM。
[PRE8]
from langchain_core.pydantic_v1 import BaseModel, Field

class Calculation_input(BaseModel):

    op1: int = Field()

    op2: int = Field()

    pass

def calculate_sum(op1: int, op2: int) -> str:

    print("调用加法",op1,op2)

    return str(op1 + op2)

def calculate_mul(op1: int, op2: int) -> str:

    print("调用乘法", op1, op2)

    return str(op1 * op2)

def calculate_div(op1: int, op2: int) -> str:

    print("调用除法", op1, op2)

    return str(op1 / op2)

def calculate_sub(op1: int, op2: int) -> str:

    print("调用减法", op1, op2)

    return str(op1 - op2)

sum_tool = StructuredTool(

    name='calculate_sum',

    func=calculate_sum,

    description="计算两个数的和",

    args_schema=Calculation_input,

)

mul_tool = StructuredTool(

    name='calculate_mul',

    func=calculate_mul,

    description="计算两个数的积",

    args_schema=Calculation_input,

)

sub_tool = StructuredTool(

    name='calculate_sub',

    func=calculate_sub,

    description="计算两个数的差",

    args_schema=Calculation_input,

)

div_tool = StructuredTool(

    name='calculate_div',

    func=calculate_div,

    description="计算两个数的商",

    args_schema=Calculation_input,

)

tools = [sum_tool,mul_tool,div_tool,sub_tool]

llm_with_tools = llm.bind_tools(tools)
工具绑定完成后，重建图并把 chatbot 换成调用带工具的模型。
[PRE9]
graph_builder = StateGraph(State)

def chatbot(state: State):

    return {"messages": [llm_with_tools.invoke(state["messages"])]}

#添加 chatbot 节点

graph_builder.add_node("chatbot", chatbot)
