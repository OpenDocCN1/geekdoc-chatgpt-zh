app = FastMCP("customer-service-demo")

@app.tool()

def tool_faq_query(keyword: str) -> Dict[str, Any]:

    """

    模拟 FAQ 库关键字查询：

    - 在 FAQ_DB 中查找与 keyword 最相关的问题

    - 若找不到，返回提示

    """

    result = {}

    for k, v in FAQ_DB.items():

        if keyword.lower() in k.lower():

            result[k] = v

    if not result:

        return {"message": f"No FAQ found for '{keyword}'"}

    return {"faq_found": result}

@app.tool()

def tool_submit_ticket(user_id: str, subject: str, content: str) -> Dict[str, Any]:

    """

    提交工单，返回 ticket_id

    """

    t_id = create_ticket(user_id, subject, content)

    return {"ticket_id": t_id, "status": "open"}

@app.tool()

def tool_get_ticket(ticket_id: str) -> Dict[str, Any]:

    """

    查看工单详情

    """

    return get_ticket(ticket_id)

#### 10.1.2 金融问答系统实战解析

金融类问答既要消化海量专业术语，又要满足严格合规，模型本身的领域准确度只是起点，上下文把控和权限的可控性同样关键。MCP 以结构化资源描述和统一的 Tool 机制为基础，把知识库、用户提问、模型推理拆分开，再通过标准接口协同作业。

下面这个示例展示了一套基于 MCP 的金融问答搭建：通过注册多组 Tool 分别完成术语释义、收益试算、账单查询、风险问卷评估；服务端借助 Context Slot 保存对话历史并动态注入资源。这样即便面对多轮追问或强合规流程，系统也能保持灵活扩展且操作可审计。案例紧扣金融场景的专属需求，是一套清晰可跑通的完整工程范例。
