# 用 Python 拼一个跨部门任务调度流水线

最近在做流程自动化时，经常会遇到一个场景：不同事业部的处理节点不一样，彼此之间又有先后依赖，既要分发对路，又要等前置完成才敢往下推。下面这段示例把这个思路拆成了三层实现，逻辑很清晰，也便于直接改造到真实业务里。

整体分工是这样的：最底层是代表各部门能力的工具对象，中间是负责路由的调度中心，最上层是带着队列和依赖关系的任务流引擎，最后用一个企业模拟脚本演示依赖链如何串起来。引擎内部起四个工作线程并发消费队列，遇到依赖未满足的任务会暂时后退，等条件达成再继续。

工具类只做一件事，模拟一次耗时的业务处理，通过 `name` 和 `processing_time` 两个属性区分不同部门的处理特征，`process` 方法内部用 `time.sleep` 制造延迟，最后返回带部门前缀的结果字符串。

调度中心在初始化时会把 HR、Finance、Legal、Sales 四个部门的工具实例预先注册到字典里。调度方法接受部门名和任务数据，如果部门存在就取出对应工具执行，否则返回错误提示。

任务流引擎负责把任务入队、记录结果、检查依赖。它用 `Queue` 保存待办，用字典缓存已完成产出，用锁保证结果写入线程安全。`add_task` 把任务 ID、部门、数据以及依赖列表打包入队。`process_tasks` 轮询队列，依赖全部在结果字典里的任务才会真正派发到调度中心执行，完成后再打印；依赖不足的任务会被重新放到队尾等待。`run` 方法启动四个线程同时跑 `process_tasks`，全部结束后再输出收尾信息。

模拟部分先初始化调度中心和引擎，然后按业务顺序添加四个任务。T1 由 HR 负责更新 Q1 员工档案，没有前置；T2 Finance 调取季度营收、T3 Legal 审查合同合规，都依赖 T1；T4 Sales 生成销售绩效报告，则要等 T2 和 T3 都完成。

这套结构把“工具 - 调度 - 引擎 - 模拟”解耦清楚，依赖检查和线程池并行也保留了扩展空间。

import time

import random

import threading

from queue import Queue

# 定义一个工具类，模拟不同部门或服务的工具调用

class Tool:

    def __init__(self, name, processing_time):

        """

        初始化工具示例

        :param name: 工具名称 (str)

        :param processing_time: 模拟处理所需时间 (float)

        """

        self.name = name

        self.processing_time = processing_time

    def process(self, task_data):

        """

        模拟工具处理任务

        :param task_data: 输入任务数据 (str)

        :return: 处理结果 (str)

        """

        # 模拟处理延时

        time.sleep(self.processing_time)

        return f"{self.name} processed: {task_data}"

# 定义调度中心类，管理多个工具并调度任务

class ToolScheduler:

    def __init__(self):

        """

        初始化调度中心，注册多个工具示例

        """

        self.tools = {

            "HR": Tool("HR_Tool", 0.2),

            "Finance": Tool("Finance_Tool", 0.3),

            "Legal": Tool("Legal_Tool", 0.25),

            "Sales": Tool("Sales_Tool", 0.15)

        }

    def schedule_task(self, department, task_data):

        """

        根据部门名称调度任务到对应工具

        :param department: 部门名称 (str)

        :param task_data: 输入任务数据 (str)

        :return: 工具处理结果 (str)

        """

        if department in self.tools:

            tool = self.tools[department]

            result = tool.process(task_data)

            return result

        else:

            return f"Error: Department '{department}' not found."

# 定义任务流引擎类，并管理任务队列和任务依赖关系

class TaskFlowEngine:

    def __init__(self, scheduler):

        """

        初始化任务流引擎，绑定调度中心

        :param scheduler: ToolScheduler 示例

        """

        self.scheduler = scheduler

        self.task_queue = Queue()

        self.results = {}

        self.lock = threading.Lock()

    def add_task(self, task_id, department, task_data, dependencies=None):

        """

        将任务添加到任务队列，包含任务依赖关系

        :param task_id: 任务标识 (str)

        :param department: 所属部门 (str)

        :param task_data: 任务数据 (str)

        :param dependencies: 依赖的任务 ID 列表 (list)

        """

        task = {

            "task_id": task_id,

            "department": department,

            "task_data": task_data,

            "dependencies": dependencies or []

        }

        self.task_queue.put(task)

    def process_tasks(self):

        """

        处理任务队列中的任务，考虑依赖关系

        """

        while not self.task_queue.empty():

            task = self.task_queue.get()

            # 检查依赖任务是否已完成

            if all(dep in self.results for dep in task["dependencies"]):

                result = self.scheduler.schedule_task(task["department"], task["task_data"])

                with self.lock:

                    self.results[task["task_id"]] = result

                print(f"Task {task['task_id']} completed: {result}")

            else:

                # 如果依赖未完成，将任务重新放入队列末尾

                self.task_queue.put(task)

                print(f"Task {task['task_id']} deferred due to unmet dependencies.")

            self.task_queue.task_done()

    def run(self):

        """

        启动任务处理，开启多个线程以并行处理任务

        """

        threads = []

        for _ in range(4):  # 使用4个工作线程

            t = threading.Thread(target=self.process_tasks)

            t.start()

            threads.append(t)

        for t in threads:

            t.join()

        print("All tasks processed.")

# 模拟下游应用：执行企业多部门任务流

def simulate_enterprise_task_flow():

    scheduler = ToolScheduler()

    engine = TaskFlowEngine(scheduler)

    # 添加任务，任务之间具有依赖关系

    engine.add_task("T1", "HR", "Update employee records for Q1")

    engine.add_task("T2", "Finance", "Retrieve quarterly revenue figures", dependencies=["T1"])

    engine.add_task("T3", "Legal", "Review contract compliance", dependencies=["T1"])

    engine.add_task("T4", "Sales", "Generate sales performance report", dependencies=["T2", "T3"])
