void quickSort(int arr[], int l, int r) {

    if (l >= r) return;

    int pivot = arr[l], i = l, j = r;

    while (i <= j) {

         while (arr[i] < pivot) i++;

         while (arr[j] > pivot) j--;

         if (i <= j) swap(arr[i++], arr[j--]);

    }

    quickSort(arr, l, j);

    quickSort(arr, i, r);

}

3. 耗时与空间评估
在常见输入下耗时约为 O(n log n)。当序列几乎有序时最不利情形会退化到 O(n²)。递归调用所占的栈空间约为 O(log n)。

【题目总结】
本例集合了按位操作的高效运用、递归分治的思维以及快排的实现。吃透这三块，无论解题还是调优都会更顺手。把握好递归的终止条件，灵活使用按位技巧，挑选合适的基准值，练稳这三招就能向编程高手靠拢。

2. 代码分段说明
1）全局声明
const int P = 998244353, N = 1e4 + 10, M = 20;
int n, m;
string s;
int dp[1 << M];
P 为防止溢出的模数。dp 用来存放压缩后的状态结果，规模为 2^M。

2）solve——状态压缩动态规划
int solve() {
    dp[0] = 1;
    for (int i = 0; i < n; ++i) {
         for (int j = (1 << (m - 1)) - 1; j >= 0; --j) {
              int k = (j << 1) | (s[i] - '0');
              if (j != 0 || s[i] == '1')
                   dp[k] = (dp[k] + dp[j]) % P;
         }
    }
    int ans = 0;
    for (int i = 0; i < (1 << m); ++i) {
         ans = (ans + 111 * i * dp[i]) % P;
    }
    return ans;
}
外层按字符推进，内层逆序遍历状态，左移一位再拼入当前字符实现转移，最后按权重 111*i*dp[i] 求和取模。

3）solve2——暴力枚举
int solve2() {
    int ans = 0;
    for (int i = 0; i < (1 << n); ++i) {
         int cnt = 0;
         int num = 0;
         for (int j = 0; j < n; ++j) {
              if (i & (1 << j)) {
                   num = num * 2 + (s[j] - '0');
                   cnt++;
              }
         }
         if (cnt <= m) (ans += num) %= P;
    }
    return ans;
}
用 i 的二进制位表示是否选取字符，拼接成数值并在选中个数不超过 m 时累加。
