## 前馈网络实现
class FeedForward(nn.Module):

    def __init__(self, d_model, dim_feedforward, dropout=0.1):

        super(FeedForward, self).__init__()

        self.linear1 = nn.Linear(d_model, dim_feedforward)

        self.dropout = nn.Dropout(dropout)

        self.linear2 = nn.Linear(dim_feedforward, d_model)

    def forward(self, x):

        return self.linear2(self.dropout(

            torch.nn.functional.relu(self.linear1(x))))

## 编码单层实现
class TransformerEncoderLayer(nn.Module):

    def __init__(self, d_model, num_heads, dim_feedforward, dropout=0.1):

        super(TransformerEncoderLayer, self).__init__()

        self.self_attn = MultiHeadSelfAttention(d_model, num_heads)

        self.feed_forward = FeedForward(d_model, dim_feedforward, dropout)

        self.norm1 = nn.LayerNorm(d_model)

        self.norm2 = nn.LayerNorm(d_model)

        self.dropout1 = nn.Dropout(dropout)

        self.dropout2 = nn.Dropout(dropout)

    def forward(self, src):

        src2 = self.self_attn(src)

        src = src + self.dropout1(src2)

        src = self.norm1(src)

        src2 = self.feed_forward(src)

        src = src + self.dropout2(src2)

        src = self.norm2(src)

        return src

## 解码单层实现
class TransformerDecoderLayer(nn.Module):

    def __init__(self, d_model, num_heads, dim_feedforward, dropout=0.1):

        super(TransformerDecoderLayer, self).__init__()

        self.self_attn = MultiHeadSelfAttention(d_model, num_heads)

        self.encoder_attn = MultiHeadSelfAttention(d_model, num_heads)

        self.feed_forward = FeedForward(d_model, dim_feedforward, dropout)

        self.norm1 = nn.LayerNorm(d_model)

        self.norm2 = nn.LayerNorm(d_model)

        self.norm3 = nn.LayerNorm(d_model)

        self.dropout1 = nn.Dropout(dropout)

        self.dropout2 = nn.Dropout(dropout)

        self.dropout3 = nn.Dropout(dropout)

    def forward(self, tgt, memory):

        tgt2 = self.self_attn(tgt)

        tgt = tgt + self.dropout1(tgt2)

        tgt = self.norm1(tgt)

        tgt2 = self.encoder_attn(tgt, memory)

        tgt = tgt + self.dropout2(tgt2)

        tgt = self.norm2(tgt)

        tgt2 = self.feed_forward(tgt)

        tgt = tgt + self.dropout3(tgt2)

        tgt = self.norm3(tgt)

        return tgt

## 整体模型实现
class Transformer(nn.Module):

    def __init__(self, d_model, num_heads, num_layers, dim_feedforward,

                 dropout=0.1):

        super(Transformer, self).__init__()

        self.encoder = nn.ModuleList([TransformerEncoderLayer(d_model,

            num_heads, dim_feedforward, dropout) for _ in range(num_layers)])

        self.decoder = nn.ModuleList([TransformerDecoderLayer(d_model,

            num_heads, dim_feedforward, dropout) for _ in range(num_layers)])

        self.pos_encoder = PositionalEncoding(d_model)

        self.pos_decoder = PositionalEncoding(d_model)

    def forward(self, src, tgt):

        src = self.pos_encoder(src)

        tgt = self.pos_decoder(tgt)

        memory = src

        for layer in self.encoder:

            memory = layer(memory)

        output = tgt

        for layer in self.decoder:

            output = layer(output, memory)

        return output

要点说明：

1) PositionalEncoding 位置嵌入：借助正弦余弦曲线合成位置向量并叠加到输入，使模型具备顺序感知。
2) MultiHeadSelfAttention 多头注意力：并行多路计算注意力权重，用以刻画序列内部不同位置的相互作用。
3) FeedForward 双层前馈：先线性扩张经 ReLU 非线性，再经丢弃正则后线性回缩，提升表征能力。
4) TransformerEncoderLayer 编码单层：自注意力后接前馈，两段均采用残差加法与层标准化。
5) TransformerDecoderLayer 解码单层：依次进行自注意力、编码-解码交叉注意与前馈，同样配合残差与标准化稳定训练。
6) Transformer 整体装配：多层编码与解码堆叠，并前后接入位置嵌入，完成源到目标的贯通式映射。

#### 2.1.2 注意力与前馈的层级分工

在 Transformer 内部，编码与解码均由注意力模块与前馈模块交替堆叠，形成逐层递进的深度结构。注意力负责从多视角挖掘词间关联，前馈负责对提取表征进行非线性重塑，二者协同提升建模效果，典型示意图见图2-2。

多头注意力本质上是同一层内的多路并行分析。输入被拆成若干头，各自独立计算权重后再融合，使模型能同时关注句子的不同侧面。

图2-2 经典的多头注意力机制

单头注意力好比一位观察者，只能盯住一对关系，比如“足球”与“踢”。多头注意力则像一组观察者，各自承担不同视角：

- 第一位聚焦“足球”与“踢”的动作-宾语关系。
- 第二位聚焦“小明”与“踢”的主语-动作关系。
- 第三位聚焦“今天”与“公园”的时间-地点关系。
