**Brown 语料切块与词块共现表**
先把 Brown 语料库前 5400 个词拼接成一段长文本，按 800 词为单位进行切分。切分结果整理成序号与文本并存的字典列表。随后用词袋模型设定词在各块中出现的最小次数与最大文档占比阈值，得到块与词的计数矩阵。词表经 numpy 转换后输出，块统一命名为 Chunk-1 递增，最后以等宽对齐的方式打印每个词在各块中的频次表格。

import numpy as np

from sklearn.feature_extraction.text import CountVectorizer

from nltk.corpus import brown

from text_chunker import chunker 
[PRE1]python

# Read the data from the Brown corpus

input_data = ' '.join(brown.words()[:5400]) 
[PRE2]python

# Number of words in each chunk 

chunk_size = 800 
[PRE3]python

text_chunks = chunker(input_data, chunk_size) 
[PRE4]python

# Convert to dict items

chunks = []

for count, chunk in enumerate(text_chunks):

    d = {'index': count, 'text': chunk}

    chunks.append(d) 
[PRE5]python

# Extract the document term matrix

count_vectorizer = CountVectorizer(min_df=7, max_df=20)

document_term_matrix = count_vectorizer.fit_transform([chunk['text'] for chunk in chunks]) 
[PRE6]python

# Extract the vocabulary and display it

vocabulary = np.array(count_vectorizer.get_feature_names())

print("\nVocabulary:\n", vocabulary) 
[PRE7]python

# Generate names for chunks

chunk_names = []

for i in range(len(text_chunks)):

    chunk_names.append('Chunk-' + str(i+1)) 
[PRE8]python

# Print the document term matrix

print("\nDocument term matrix:")

formatted_text = '{:>12}' * (len(chunk_names) + 1)

print('\n', formatted_text.format('Word', *chunk_names), '\n')

for word, item in zip(vocabulary, document_term_matrix.T):

    # 'item' is a 'csr_matrix' data structure

    output = [word] + [str(freq) for freq in item.data]

    print(formatted_text.format(*output))
