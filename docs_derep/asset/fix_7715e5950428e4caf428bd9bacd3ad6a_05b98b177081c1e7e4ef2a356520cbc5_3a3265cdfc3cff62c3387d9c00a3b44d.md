## 自注意力模块

class Attention(torch.nn.Module):

    def __init__(self, fill: bool = True):

        super(Attention, self).__init__()

        self.scaling_coeff = 1.0 / torch.sqrt(torch.tensor(d_main, device=device))  # 点积缩放系数
        self.lower_tri = torch.tril(torch.ones(1, 1, device=device)) if fill else None  # 下三角掩码模板
        self.linear_k = torch.nn.Linear(in_features=d_main, out_features=d_main, bias=False)  # K投影
        self.linear_q = torch.nn.Linear(in_features=d_main, out_features=d_main, bias=False)  # Q投影
        self.linear_v = torch.nn.Linear(in_features=d_main, out_features=d_main, bias=False)  # V投影
        self.linear_attention = torch.nn.Linear(in_features=d_main, out_features=d_main, bias=False)  # 输出投影

    def forward(self, x, x_lengths):

        examples, length, dim = x.shape  # 解包输入三维
        k = self.linear_k(x)  # K特征
        q = self.linear_q(x)  # Q特征
        v = self.linear_v(x)  # V特征

        scaled_dotproduct = (q @ k.transpose(-1, -2)) * self.scaling_coeff  # 缩放点积

        if x_lengths is not None:  # 序列长度不一致时
            numbers = torch.arange(length, device=device).unsqueeze(dim=0)  # 位置索引
            seq_mask_vector = x_lengths.unsqueeze(dim=-1) > numbers  # 长度掩码向量
            seq_mask_matrix = seq_mask_vector.unsqueeze(dim=-1) * seq_mask_vector.unsqueeze(dim=1)  # 长度掩码矩阵
            seq_mask_matrix = seq_mask_matrix * self.lower_tri[:length, :length] if self.lower_tri is not None else seq_mask_matrix  # 融合因果掩码
            masked_scaled_dotproduct = scaled_dotproduct.masked_fill(seq_mask_matrix == 0, float('-inf'))  # 失效位置置 -inf
            weight = torch.nn.functional.softmax(masked_scaled_dotproduct, dim=-1)  # 注意力权重
            weight = weight.masked_fill(seq_mask_vector.unsqueeze(dim=-1) == 0, 0)  # 填充位置权重清零
        else:  # 所有序列等长时
            masked_scaled_dotproduct = scaled_dotproduct.masked_fill(self.lower_tri[:length, :length] == 0, float('-inf')) if self.lower_tri is not None else scaled_dotproduct  # 应用因果掩码
            weight = torch.nn.functional.softmax(masked_scaled_dotproduct, dim=-1)  # 得到权重

        agg = weight @ v  # 加权聚合输出
