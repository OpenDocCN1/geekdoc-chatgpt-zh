把 {{input}} 里的原文抽取出来，统一打包成 JSON。输出结构固定为三项：title 标题、content 内容列表、tags 标签列表。

######## 示例

{
    "title": "职场女性就该当「乖学生」？",

    "content": [

        "是迎合规则？",

        "保持谦逊？",

        "隐藏锋芒？",

        "都不是！"

    ],

    "tags": [

        "自我觉醒",

        "价值提升",

        "职场真相"

    ]

}

async def main(args: Args) -> Output:
    params = args.params
    input = params.get("input")
    length = len(input)
    # 构建输出对象
    ret: Output = {
        "length" : length
    }
    return ret
