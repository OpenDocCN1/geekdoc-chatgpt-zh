将输入的文本映射为标准 JSON 结构，输出需要包含 title、content 列表、tags 列表这三个字段。

待处理的原文为 {{input}}。

######## 示例

{
    "title": "职场女性就该当「乖学生」？",
    "content": [
        "是迎合规则？",
        "保持谦逊？",
        "隐藏锋芒？",
        "都不是！"
    ],
    "tags": [
        "自我觉醒",
        "价值提升",
        "职场真相"
    ]
}
[PRE1]python
async def main(args: Args) -> Output:
    params = args.params
    input = params.get("input")
    length = len(input)
    # 构建输出对象
    ret: Output = {
        "length" : length
    }
    return ret
