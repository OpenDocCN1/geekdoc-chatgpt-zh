# 解码 logits 维度拆解与训练实战

Transformer 在解码阶段返回的 logits 张量形状为 `[batch_size * tgt_len, tgt_vocab_size]`。换句话说，批次中的所有目标位置被拉平成若干行，每行对应一个词位，该词位在目标词表上的概率分布共有 `tgt_vocab_size` 个维度，推理时一般取概率最大的那一维作为预测。整体来看，模型的工作方式就是先经 Encoder 编码，再经 Decoder 解码，最后输出结果。

下面是模型初始化与训练循环的配置片段。训练共进行 1024 个 epoch，使用 tqdm 展示进度，数据在进入计算前都搬到指定设备。前向得到输出以及编码自关注、解码自关注、解码对编码的关注三组注意力，损失采用忽略下标 0 的交叉熵，目标先展平成一维。每次迭代执行梯度清零、反向传播、参数更新，并把当前轮次和损失写到进度条，轮次结束保存模型权重。

for epoch in range(1024):

    pbar = tqdm((loader), total=len(loader))  # 显示进度条

    for enc_inputs, dec_inputs, dec_outputs in pbar:

        enc_inputs, dec_inputs, dec_outputs = enc_inputs.to(device), dec_inputs.to(device), dec_outputs.to(device)

        # outputs: [batch_size * tgt_len, tgt_vocab_size]

        outputs, enc_self_attns, dec_self_attns, dec_enc_attns = model(enc_inputs, dec_inputs)

        loss = criterion(outputs, dec_outputs.view(-1))

        optimizer.zero_grad()

        loss.backward()

        optimizer.step()

        pbar.set_description(f"epoch {epoch + 1} : train loss {loss.item():.6f} ")  # : learn_rate {lr_scheduler.get_last_lr()[0]:.6f}

    torch.save(model.state_dict(), save_path)

class Memory:

    def __init__(self):

        """初始化"""

        self.actions = []  # 行动(共4种)

        self.states = []   # 状态, 由8个数字组成

        self.logprobs = [] # 概率

        self.rewards = []  # 奖励

        self.is_dones = [] # 游戏是否结束: is_terminals?

    def clear_memory(self):

        del self.actions[:]

        del self.states[:]

        del self.logprobs[:]

        del self.rewards[:]

        del self.is_dones[:]

class Action(torch.nn.Module):

    def __init__(self, state_dim=8, action_dim=4):

        super().__init__()

        # actor

            torch.nn.Linear(state_dim, 128),

            torch.nn.ReLU(),

            torch.nn.Linear(128, 64),

            torch.nn.ReLU(),

            torch.nn.Linear(64, action_dim),
