### 6.3 分词器构造思路

大模型里常用的子词切分方案主要是 BPE 与 BBPE，二者实现路径差别明显。下面围绕 BPE 做展开说明。

**6.3.1 BPE**

**1.核心思路**

BPE 起源于数据压缩，后被引入 NLP 用于子词分割。整体做法是通过不断合并高频字节对来扩充词表。具体包含：

●起步阶段：把文本里每个字符当作独立符号，比如把 30000 个基础汉字都作为初始词元，并统计出现次数。

●循环合并：每轮挑选出现次数最多的相邻符号对，合并成新符号，并同步刷新词表与频次。

●结束条件：达到预设词表规模或合并轮数后停止。

这样既能缓解未登录词问题，也能在词表大小和模型效果之间取得折中。

**2.伪代码示例**

Python

def BPE（text, vocab_size）:

  #初始化词汇表

  vocab = {char: freq for char, freq in count_characters（text）}

  while len（vocab） < vocab_size:

    #统计所有相邻符号对的出现频率

    pairs = get_most_frequent_pair（vocab）

    #如果没有更多的符号对可以合并，退出循环，停止增加词表个数

    if not pairs:

      break

    #合并出现最频繁的符号对

    new_vocab = merge_pair（vocab, pairs[0]）

    #更新词汇表

    vocab = new_vocab

  return vocab

def count_characters（text）:

  #统计每个字符出现的频率

  return Counter（text）

def get_most_frequent_pair（vocab）:

  #统计所有相邻符号对出现的频率

  pairs = {}

  for word, freq in vocab.items（）:

    symbols = word.split（）

    for i in range（len（symbols） - 1）:

      pair = （symbols[i], symbols[i+1]）

      pairs[pair] = pairs.get（pair, 0） + freq

  #返回出现频率最高的符号对

  return sorted（pairs, key=pairs.get, reverse=True）

def merge_pair（vocab, pair）:

  #合并符号对

  new_vocab = {}

  for word, freq in vocab.items（）:

    new_word = word.replace（' '.join（pair）, ''.join（pair））

    new_vocab[new_word] = freq

  return new_vocab

from collections import defaultdict, Counter

class BPETokenizer:

  def __init__（self, vocab_size=1000）:

    self.vocab_size = vocab_size            #目标词汇表大小

    self.vocab = defaultdict（int）           #初始化词汇表

    self.merges = []                  #存储合并的符号对

  def train（self, text）:

    #初始化词汇表，统计每个字符的出现频率

    self.vocab = Counter（text.split（））

    #迭代合并符号对，直到词汇表达到目标大小

    while len（self.vocab） < self.vocab_size:

      #统计所有相邻符号对的出现频率

      pairs = self.get_pairs（）

      if not pairs:

        break                    #如果没有可合并的符号对，停止送代

      #找到出现最频繁的符号对

      most_frequent_pair = max（pairs, key=pairs.get）

      #合并符号对，并更新词汇表

      self.merge_pair（most_frequent_pair）

      #记录合并的符号对

      self.merges.append（most_frequent_pair）

 def get_pairs（self）:

    #统计所有相邻符号对的出现频率

    pairs = defaultdict（int）

    for word, freq in self.vocab.items（）:

      symbols = word.split（）            #将单词拆分为符号

      for i in range（len（symbols） - 1）:

        pair = （symbols[i], symbols[i+1]）     #获取相邻符号对

        pairs[pair] += freq             #更新符号对的出现频率

    return pairs

  def merge_pair（self, pair）:

    #合并符号对，更新词汇表

    new_vocab = defaultdict（int）

    for word, freq in self.vocab.items（）:

      #将符号对合并为一个新 Token

      new_word = word.replace（' '.join（pair）, ''.join（pair））

      new_vocab[new_word] += freq           #更新新词汇表

    self.vocab = new_vocab

  def Tokenize（self, text）:

    #将输入文本分词为子词符号

    words = text.split（）

    Tokens = []

    for word in words:

      symbols = list（word）             #将单词拆分为字符

      #应用所有合并规则

      for pair in self.merges:

        symbols = self.apply_merge（symbols, pair）

      Tokens.extend（symbols）            #将符号加入结果列表

    return Tokens

  def apply_merge（self, symbols, pair）:

    #应用合并规则到符号列表

    new_symbols = []

    i = 0

    while i < len（symbols）:

      #如果当前符号和下一个符号匹配合并规则

      if i < len（symbols） - 1 and （symbols[i], symbols[i+1]） == pair:

        new_symbols.append（''.join（pair））    #合并符号对

        i += 2                   #跳过下一个符号

      else:

        new_symbols.append（symbols[i]）      #保留当前符号

        i += 1

    return new_symbols
