## 悟空图库多模态向量化实战

想给黑神话悟空的截图做语义检索，先要把图片和文字描述投影到同一个向量空间。这份实现用 Visualized_BGE 作为编码核心，先封装一个统一的编码器，再用 dataclass 描述单张图的信息，最后批量跑一遍数据集生成向量，为后续 Milvus 入库做准备。

class WukongEncoder:
    """统一封装视觉与文本到向量空间的编码器"""
    def __init__(self, model_name: str, model_path: str):
        self.model = Visualized_BGE(model_name_bge=model_name, model_weight=model_path)
        self.model.eval()

    def encode_query(self, image_path: str, text: str) -> list[float]:
        """将图片与文字联合编码为查询向量"""
        with torch.no_grad():
            query_emb = self.model.encode(image=image_path, text=text)
        return query_emb.tolist()[0]

    def encode_image(self, image_path: str) -> list[float]:
        """仅对单张图片做向量化"""
        with torch.no_grad():
            query_emb = self.model.encode(image=image_path)
        return query_emb.tolist()[0]

class WukongDataset:
    """负责读取 JSON 元数据并维护图像清单"""
    def __init__(self, data_dir: str, metadata_path: str):
        self.data_dir = data_dir
        self.metadata_path = metadata_path
        self.images: List[WukongImage] = []
        self._load_metadata()

    def _load_metadata(self):
        """解析元文件构建内存图像列表"""
        with open(self.metadata_path, 'r', encoding='utf-8') as f:
            data = json.load(f)
        for img_data in data['images']:
            self.images.append(WukongImage(**img_data))
