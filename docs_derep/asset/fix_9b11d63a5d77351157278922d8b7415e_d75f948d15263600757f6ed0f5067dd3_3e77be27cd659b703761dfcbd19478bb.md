def run_pytorch_inference(model,input_data):

    """

    使用PyTorch模型进行推理

    """

    model.eval()

    with torch.no_grad():

        input_tensor=torch.from_numpy(input_data).float()

        pytorch_output=model(input_tensor)

    return pytorch_output.numpy()
