async def main():

    # 初始化 MCP 客户端实例

    client = MCPClient("https://mcp.pay***t.example.com")

    # 列出服务端资源，获取汇率和手续费数据

    # 读取汇率数据

    exchange_data = await client.read_resource("resource://exchange_rate")

    # 读取手续费数据

    fee_data = await client.read_resource("resource://payment_fee")

    # 更新本地上下文，存储资源数据

    await client.update_context("exchange_rate", exchange_data["USD_CNY"])

    await client.update_context("fee_rate", fee_data["credit_card_fee"])

    # 调用支付计算工具：计算支付金额

    tool_params = {

        "amount": 100,  # 假设支付原始金额为 100 美元

        "exchange_rate": client.context.get("exchange_rate", 1),

        "fee_rate": client.context.get("fee_rate", 0)

    }

    tool_result = await client.call_tool("calculate_payment", tool_params)

    await client.update_context("payment_result", tool_result)

    # 发起支付处理请求，整合上下文数据

    invoke_params = {

        "session": {

            "client_id": "client_001",

            "context": client.context

        "order": {

            "order_id": "order_123456",

            "amount": 100

    }

#### 5.2.3 开发包增强与个性化实践

官方开发包已能覆盖多数常规需求，但在引入提供方钩子、扩充呼叫方请求处理、搭建跨服务的智能体协作等情形下，需要动手做个性化增强。这种需求往往源于业务形态多样且持续演进，对记忆注入方式、能力函数执行以及数据节点管控提出了更高的灵活要求。

MCP 的设计理念是提供一套可扩展的基础骨架，开发者在不改动核心协议的前提下，可通过插件、中间件或语义化注入把业务逻辑精准嵌入模型交互过程。

##### 1. 主要增强方向
- 能力插件扩展：可自行编写并注册专属的计算或检索函数，例如数据挖掘、NLP 辅助等，让提供方拥有更丰富的处理手段，呼叫方仍保持统一的参数与状态调用习惯。
- 数据节点访问策略：在读写数据节点前加入鉴权、预处理或缓存逻辑，通过重载开发包中的数据管理接口，实现对访问全链路的自定义控制。
- 状态注入规则：支持在请求发出前自动拼装或改写状态槽位，也支持在响应返回后做二次加工。借助中间件或钩子函数，能够动态管理状态片段，适配复杂调度与多智能体协同。
- 异常处理与重试机制：针对网络波动、模型超时、能力函数执行异常等情况，结合场景规则实现更细化的异常捕获、错误码归类和自动重试或降级。

##### 2. 常见业务落地
在跨境电商里，可以为 MCP 提供方挂载多语言翻译、海关费用测算、订单追踪查询等额外能力；在知识管理平台里，则需要扩展数据处理钩子以实现机密文档的访问控制与版本管理。不同场景都能通过扩展开发包接口或挂载自定义方法实现，并在呼叫方保持一致的调用形态。
