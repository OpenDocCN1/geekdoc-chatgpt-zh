async def main():

    # 初始化 MCP 客户端实例

    client = MCPClient("https://mcp.pay***t.example.com")

    await client.initialize()

    # 列出服务端资源，获取汇率和手续费数据

    resources = await client.list_resources()

    # 读取汇率数据

    exchange_data = await client.read_resource("resource://exchange_rate")

    # 读取手续费数据

    fee_data = await client.read_resource("resource://payment_fee")

    # 更新本地上下文，存储资源数据

    await client.update_context("exchange_rate", exchange_data["USD_CNY"])

    await client.update_context("fee_rate", fee_data["credit_card_fee"])

    # 调用支付计算工具：计算支付金额

    tool_params = {

        "amount": 100,  # 假设支付原始金额为 100 美元

        "exchange_rate": client.context.get("exchange_rate", 1),

        "fee_rate": client.context.get("fee_rate", 0)

    }

    tool_result = await client.call_tool("calculate_payment", tool_params)

    await client.update_context("payment_result", tool_result)

    # 发起支付处理请求，整合上下文数据

    invoke_params = {

        "session": {

            "client_id": "client_001",

            "context": client.context

        },

        "order": {

            "order_id": "order_123456",

            "amount": 100

        }

    }

    response = await client.invoke("process_payment", invoke_params)

    print("最终支付处理响应:")

    print(json.dumps(response, indent=2, ensure_ascii=False))

if __name__ == "__main__":

#### 5.2.3 SDK 定制与二次开发

官方 SDK 已经能满足大部分常规用法，但当业务要做更深度的改造时，比如新增服务端 Hook、扩展客户端请求处理逻辑、搭建跨服务的智能体协作，就得动手定制。需求的本质来自业务的多样化与演进，对上下文的注入方式、工具函数的执行和资源的管控提出了更高的灵活性要求。

MCP 的设计思路是提供一套可扩展的基础框架，开发者可以在不破坏核心协议的前提下，通过插件、中间件或语义化注入，把业务逻辑精准地嵌入到模型交互流程中。

##### 1. 定制的主要方向

- 工具层扩展：可以自行编写并注册专用的工具函数，例如数据挖掘、NLP 辅助等，让服务端具备更丰富的计算与检索能力，客户端保持统一的参数与上下文调用方式即可。
- 资源访问策略：在读或写资源前加入权限校验、数据预处理或缓存机制，通过重载 SDK 中的资源管理接口，对访问全过程做自定义控制。
- 上下文注入规则：支持在请求发出前自动拼装或改写上下文 Slot，也支持在响应返回后做二次加工。借助中间件或钩子函数，能动态管理上下文片段，适配复杂调度与多智能体协同。
- 错误处理与重试机制：针对网络抖动、模型超时、工具执行异常等情况，结合业务规则实现更细化的异常捕获、错误码分类和自动重试或降级。

##### 2. 典型应用场景

跨境电商里，可以为 MCP 服务端挂载多语言翻译、海关费用计算、订单追踪查询等额外工具；知识管理平台里，则需要扩展资源处理钩子来做机密文档的访问控制与版本管理。不同业务都能通过扩展 SDK 接口或挂载自定义方法实现，并在客户端保持一致的调用形态。
