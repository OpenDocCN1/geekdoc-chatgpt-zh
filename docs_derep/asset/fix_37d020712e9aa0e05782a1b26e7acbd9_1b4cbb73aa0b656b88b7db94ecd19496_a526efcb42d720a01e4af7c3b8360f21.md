# 企业知识问答快速验证方案

本方案展示一套可直接跑起来的问答原型，以及围绕它做的依赖、样例、单元、集成、压力和资源监控检查。用 Flask 暴露一个 POST 接口，启动时就创建向量库实例。请求进来后先检查请求体是否同时携带上下文和问题，缺失则直接返回 400。校验通过后先对上下文做预处理得到向量表达，再用向量库取回最接近的前 3 条记录并打印距离与索引，随后把上下文和问题交给生成模块产出答案，最后以 JSON 形式回传。服务绑定 0.0.0.0 的 5000 端口，便于容器化部署。

技术栈集中在 PyTorch 生态的模型推理、Transformer、faiss-cpu 向量检索、Flask Web 框架以及 NumPy。

样例数据准备了四条企业内部政策问答，每条都包含上下文、提问和期望出现的关键词，用于人工核对。单元测试覆盖生成模块是否能在答案中包含带薪年假与 10 天等关键信息，以及向量检索是否能稳定返回 3 条候选。集成测试模拟一次年终奖金查询，演示检索与生成两步协同的工作顺序。压力测试循环 20 次调用生成函数并计算平均耗时。资源监控脚本借助 psutil 定期打印 CPU 占用、内存使用率和剩余可用内存，连续采集 5 次。

from flask import Flask, request, jsonify

import torch

from model import generate_answer

from database import VectorDatabase

from utils import preprocess_data

app = Flask(__name__)

# 初始化向量数据库

vector_db = VectorDatabase()

@app.route('/query', methods=['POST'])

def query():

    """

    API端点，接收查询请求并返回回答

    """

    data = request.json

    context = data.get("context")

    question = data.get("question")

    if not context or not question:

        return jsonify({"error": "context and question are required"}), 400

    # 检索最相似的向量

    embeddings = preprocess_data([context])

    distances, indices = vector_db.search(embeddings, k=3)

    print("检索结果:", distances, indices)

    # 调用生成模型

    answer = generate_answer(context, question)

    return jsonify({"answer": answer})

if __name__ == "__main__":

    app.run(host="0.0.0.0", port=5000)

import time

def load_test(num_requests):

    """

    负载测试：模拟多次请求并记录响应时间

    :param num_requests: 请求次数

    """

    query="公司关于远程工作的政策是什么？"

    context="公司规定员工可以选择每周至少1天的远程工作，但需提前申请。"

    start_time=time.time()

    for _ in range(num_requests):

        # 模拟生成回答

        answer=generate_optimized_answer(context, query)

        print(answer)  # 可选择性打印

    end_time=time.time()

    avg_time=(end_time-start_time) / num_requests

    print(f"平均响应时间: {avg_time:.2f} 秒")

import psutil

def monitor_performance():

    """

    性能监控：打印CPU和内存的当前使用情况

    """

    # 获取CPU和内存的使用情况

    cpu_usage=psutil.cpu_percent(interval=1)

    memory_info=psutil.virtual_memory()

    memory_usage=memory_info.percent

    print(f"当前CPU使用率: {cpu_usage}%")

    print(f"当前内存使用率: {memory_usage}%")

    print(f"当前可用内存: {memory_info.available / 1024**2:.2f} MB")

# 定期调用性能监控函数

for _ in range(5):  # 模拟监控多次

    monitor_performance()

    time.sleep(2)
