T5-small 快速搭建可部署的文本分析服务

这个案例以 t5-small 为基座，给出一个可以直接落地的小型 NLP 服务骨架。初始化阶段会一次性加载分词器与序列到序列模型，随后对外暴露四项能力：对输入做去首尾空白并转小写的归一化，对提示做基于预训练模型的续写，通过分词统计单词出现频次作为基础分析，以及在推理时关闭梯度来降低开销。所有功能收拢在同一个类里，便于统一调用与验证。

验证层使用 unittest 完成。类级别的 setUpClass 只创建一次平台实例，之后覆盖五种场景。端到端用例先拿一段前后带空格的英文句子做清洗，期望得到全小写的规范文本；随后调用续写接口确认返回非空字符串，再用统计接口确认返回非空字典，最后用无梯度推理确认同样能产出结果。单点用例里，清洗以 “   Hello World!   ” 验证空白剔除与大小写统一；续写以 “The future of AI is” 验证输出类型；频次统计以 “hello world hello” 验证 hello 出现 2 次、world 出现 1 次；推理以 “The impact of NLP on businesses is” 验证输出非空。五项均通过，整体耗时约 3.567 秒。

镜像化选用 python:3.9-slim 为基础，工作目录固定为 /app。先复制源码，随后复制 requirements.txt 并离线安装依赖，开放 5000 端口，启动命令指向 app.py。构建镜像用 docker build -t my-text-analysis-platform .，运行用 docker run -p 5000:5000 my-text-analysis-platform。日常运维可通过 docker images、docker ps、docker logs 查看镜像、容器状态与日志。

生产侧采用 Kubernetes Deployment 上线。部署名为 text-analysis-deployment，标签 app: text-analysis，副本数 3 以提升可用性。Pod 模板复用相同标签，容器名为 text-analysis-container，使用镜像 my-text-analysis-platform:latest，监听 5000 端口。资源设置上请求 256Mi 内存与 250m CPU，上限 512Mi 内存与 500m CPU，避免争抢。

import unittest

from transformers import AutoTokenizer, AutoModelForSeq2SeqLM

import torch

# 模拟的主模块，用于集成各个子模块

class NLPTextAnalysisPlatform:

    def __init__(self):

        # 加载预训练模型和分词器

        self.tokenizer=AutoTokenizer.from_pretrained("t5-small")

        self.model=AutoModelForSeq2SeqLM.from_pretrained("t5-small")

    def preprocess_text(self, text):

        # 数据预处理示例：去除多余空格并处理大小写

        return text.strip().lower()

    def generate_text(self, prompt):

        # 基于预训练模型的文本生成

        inputs=self.tokenizer(prompt, return_tensors="pt")

        outputs=self.model.generate(**inputs, max_length=50)

        generated_text=self.tokenizer.decode(outputs[0],

                                             skip_special_tokens=True)

        return generated_text

    def analyze_text(self, text):

        # 高级文本分析：统计单词频率作为示例

        words=text.split()

        word_freq={word: words.count(word) for word in set(words)}

        return word_freq

    def optimize_and_infer(self, prompt):

        # 模型推理性能测试

        with torch.no_grad():

            inputs=self.tokenizer(prompt, return_tensors="pt")

            outputs=self.model.generate(**inputs, max_length=50)

            return self.tokenizer.decode(outputs[0],

                                       skip_special_tokens=True)

# 集成测试类

class TestNLPTextAnalysisPlatformIntegration(unittest.TestCase):

    @classmethod

    def setUpClass(cls):

        cls.platform=NLPTextAnalysisPlatform()

    def test_end_to_end_pipeline(self):

        # 测试整个NLP分析流程的集成：数据预处理、文本生成、内容分析

        raw_text=" Natural Language Processing is an exciting field of AI. "

        preprocessed_text=self.platform.preprocess_text(raw_text)

        self.assertEqual(preprocessed_text,

                         "natural language processing is an exciting field of ai.",

                         "预处理结果与预期不符")

        # 检查文本生成模块

        generated_text=self.platform.generate_text(preprocessed_text)

        self.assertTrue(isinstance(generated_text, str) and len(generated_text) > 0,

                        "文本生成模块输出不符合预期")

        # 检查文本分析模块

        analysis_result=self.platform.analyze_text(generated_text)

        self.assertTrue(isinstance(analysis_result, dict) and len(analysis_result) > 0,

                        "文本分析结果不符合预期")

        # 检查优化与推理性能模块

        optimized_output=self.platform.optimize_and_infer(preprocessed_text)

        self.assertTrue(isinstance(optimized_output, str) and len(optimized_output) > 0,

                        "推理性能优化模块输出不符合预期")

    def test_preprocess_text(self):

        # 测试预处理功能

        text="   Hello World!   "

        expected="hello world!"

        result=self.platform.preprocess_text(text)

        self.assertEqual(result, expected, "预处理失败：未正确去除空白和调整大小写")

    def test_generate_text(self):

        # 测试文本生成模块

        prompt="The future of AI is"

        generated_text=self.platform.generate_text(prompt)

        self.assertTrue(isinstance(generated_text, str) and len(generated_text) > 0, "生成的文本不符合预期格式")

    def test_analyze_text(self):

        # 测试内容分析模块

    def test_optimize_and_infer(self):

        # 测试推理优化模块

        prompt="The impact of NLP on businesses is"

        self.assertTrue(isinstance(optimized_output, str) and len(optimized_output) > 0, 

                        "优化后的推理输出不符合预期")
