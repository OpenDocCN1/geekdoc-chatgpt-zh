import requests

# DeepSeek API配置

API_URL="https://api.deepseek.com/v1/chat/completions"  # 对话生成接口

API_KEY="your_api_key_here" # 替换为实际的API密钥

# 定义请求头

HEADERS={

    "Authorization": f"Bearer {API_KEY}",

    "Content-Type": "application/json"

}

# 数学问题求解函数

def solve_math_problem(prompt):

    """

    调用DeepSeek API解决数学问题

    :param prompt: 数学问题描述

    :return: 模型的解答

    """

    data={

        "model": "deepseek-v3",

        "messages": [{"role":"user","content":prompt}],

        "max_tokens": 150,

        "temperature": 0.0

    }

    response=requests.post(API_URL,headers=HEADERS,json=data)

    if response.status_code == 200:

        return response.json().get("choices")[0].get("message",{}).get("content","").strip()

    else:

        return f"请求失败，状态码:{response.status_code},错误信息: {response.text}"

# 示例题目

math_problems=[

    "计算定积分f(x^2+3x+2)dx在区间[0,1]上的值。",

    "求解线性代数方程组:2x+3y=5和x-y=2。",

    "求函数f(x)=e^(-x^2)在x=0处的导数值。",

    "计算数列的前10项和，已知通项公式为a_n=n^2+1。",

    "求矩阵[[1,2],[3,4]]的行列式值。"

]

# 执行评估

if __name__ == "__main__":

    print("2025年考研数学一常规题目评估开始。\n")

    for i, problem in enumerate(math_problems, 1):

        print(f"题目 {i}: {problem}")

        solution=solve_math_problem(problem)

        print(f"DeepSeek-V3解答: {solution}\n")

这里选用 deepseek-v3，输出上限 150 token，温度调到 0，追求可重复的确定答案。题单覆盖定积分、线性方程组、一点求导、有限项求和、二阶行列式。运行时会逐条打印序号、原题及模型返回，便于对照 2025 考研数学一的基础项。

import requests

API_URL="https://api.deepseek.com/v1/chat/completions" # 对话生成接口

API_KEY="your_api_key_here" # 替换为实际的 API 密钥

# 定义请求头

HEADERS={

    "Authorization": f"Bearer {API_KEY}",

    "Content-Type":"application/json"

}

### 请求 DeepSeek 生成算法

def generate_algorithm(prompt):

    """

    调用 DeepSeek 生成算法代码

    :param prompt: 提示描述

    :return: 模型生成的代码

    """

    data={

        "model": "deepseek-v3",

        "messages": [{"role": "user", "content": prompt}],

        "max_tokens": 300,

        "temperature": 0.7

    }

    response=requests.post(API_URL, headers=HEADERS, json=data)

    if response.status_code == 200:

        return response.json().get("choices", [{}])[0].get("message", {}).get("content", "").strip()

    else:

        return f"请求失败，状态码：{response.status_code}，错误信息：\n{response.text}"

# 示例：生成快速排序算法

if __name__ == "__main__": 

    prompt="生成一个 Python 实现的快速排序算法，并添加注释。"

    print("调用 DeepSeek 生成算法中，请稍等...\n")

    algorithm_code=generate_algorithm(prompt)

    print("生成的快速排序算法代码：\n")

    print(algorithm_code)

# 测试生成的代码

print("\n测试生成的快速排序算法:")

这次把 max_tokens 放到 300，temperature 用 0.7，让表达更灵活。提示只有一句话，要求输出带注释的 Python 快速排序。日志先提示正在生成，随后打印代码并直接执行。对 [3, 6, 8, 10, 1, 2, 1] 调用 quicksort，得到 [1, 1, 2, 3, 6, 8, 10]，排序正确。

要点提炼

1. 任务描述：仅用“生成一个 Python 实现的快速排序算法，并添加注释”这一自然指令，模型就能给出符合预期的实现。
2. 代码水准：产出结构清晰、注释到位、逻辑自洽，可直接用于教学演示或工程复用。
3. 可迁移性：同类调用可延伸到动态规划、图搜索等更复杂算法的快速落地。

实验观察

1. 响应效率：在极短时间内产出完整排序实现，体现出模型在算法辅助开发上的速度优势。
2. 正确性：生成片段无语法错误，运行逻辑通顺，且具备一定扩展空间。
3. 使用场景：适合课堂算法展示、科研快速验证，以及项目里基础算法的原型实现。

总体看，DeepSeek-V3 在辅助算法开发上兼顾效率与实用，能迅速产出高质量代码，加快编码节奏，为教学和工程提供明显支撑。
