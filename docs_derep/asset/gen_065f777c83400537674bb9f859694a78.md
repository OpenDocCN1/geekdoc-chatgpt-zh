# OpenCV 实时框选跟踪实战

class ObjectTracker(object):

    def __init__(self, scaling_factor=0.5):

        # 初始化摄像头采集器
        self.cap = cv2.VideoCapture(0)

[PRE0]
 # 读取摄像头首帧
        _, self.frame = self.cap.read()

[PRE1]
 # 记录画面缩放系数
        self.scaling_factor = scaling_factor

[PRE2]
 # 将画面按比例缩放
        self.frame = cv2.resize(self.frame, None,
                fx=self.scaling_factor, fy=self.scaling_factor,
                interpolation=cv2.INTER_AREA)

[PRE3]
 # 创建用于展示的窗口
        cv2.namedWindow('Object Tracker')

[PRE4]
 # 注册鼠标事件处理函数
        cv2.setMouseCallback('Object Tracker', self.mouse_event)

[PRE5]
 # 框选区域初始化为空
        self.selection = None
        # 拖拽起点初始化为空
        self.drag_start = None
        # 跟踪状态初始为未开始
        self.tracking_state = 0

[PRE6]
 # 定义鼠标交互处理方法
    def mouse_event(self, event, x, y, flags, param):

        # 将坐标转为 int16 类型
        x, y = np.int16([x, y])

[PRE7]
 # 检测左键按下事件
        if event == cv2.EVENT_LBUTTONDOWN:
            self.drag_start = (x, y)
            self.tracking_state = 0

[PRE8]
 # 判断是否进入框选流程
        if self.drag_start:
            if flags & cv2.EVENT_FLAG_LBUTTON:
                # 获取当前帧高宽
                h, w = self.frame.shape[:2]

[PRE9]
 # 读取拖拽起始坐标
                xi, yi = self.drag_start

[PRE10]
 # 计算左上与右下边界
                x0, y0 = np.maximum(0, np.minimum([xi, yi], [x, y]))
                x1, y1 = np.minimum([w, h], np.maximum([xi, yi], [x, y]))

[PRE11]
 # 先清空旧的选中区域
                self.selection = None

[PRE12]
 # 当宽高有效时确认选区
                if x1-x0 > 0 and y1-y0 > 0:
                    self.selection = (x0, y0, x1, y1)

[PRE13]
 else:
                # 释放鼠标后进入跟踪模式
                self.drag_start = None
                if self.selection is not None:
                    self.tracking_state = 1

[PRE14]
 # 启动持续跟踪的主循环
    def start_tracking(self):
        # 循环运行直到按下 Esc
        while True:
            # 实时抓取下一帧
            _, self.frame = self.cap.read()

[PRE15]
 # 对新帧做相同缩放
            self.frame = cv2.resize(self.frame, None,
                    fx=self.scaling_factor, fy=self.scaling_factor,
                    interpolation=cv2.INTER_AREA)

[PRE16]
 # 复制一帧用于可视化
            vis = self.frame.copy()

[PRE17]
 # 转为 HSV 色彩空间
            hsv = cv2.cvtColor(self.frame, cv2.COLOR_BGR2HSV)

[PRE18]
 # 按阈值生成颜色掩膜
            mask = cv2.inRange(hsv, np.array((0., 60., 32.)),
            np.array((180., 255., 255.)))

[PRE19]
 # 若存在用户选区则处理
            if self.selection:
                # 解析选区四个坐标
                x0, y0, x1, y1 = self.selection
                # 构建跟踪窗口参数
                self.track_window = (x0, y0, x1-x0, y1-y0)

[PRE20]
 # 截取感兴趣区域
                hsv_roi = hsv[y0:y1, x0:x1]
                mask_roi = mask[y0:y1, x0:x1]
                # 计算 ROI 的 H 通道直方图
                # 仅统计掩膜内的像素
                hist = cv2.calcHist( [hsv_roi], [0], mask_roi,
                        [16], [0, 180] )

[PRE21]
 # 归一化并拉平成一维
                cv2.normalize(hist, hist, 0, 255, cv2.NORM_MINMAX)
                self.hist = hist.reshape(-1)

[PRE22]
 # 取出可视化用的 ROI
                vis_roi = vis[y0:y1, x0:x1]

[PRE23]
 # 反色处理仅用于显示效果
                cv2.bitwise_not(vis_roi, vis_roi)
                vis[mask == 0] = 0

[PRE24]
 # 判断当前是否处于跟踪阶段
            if self.tracking_state == 1:
                # 清空选区避免重复建模
                self.selection = None

[PRE25]
 # 执行直方图反向投影
                hsv_backproj = cv2.calcBackProject([hsv], [0],
                        self.hist, [0, 180], 1)

[PRE26]
 # 对反向投影结果进行掩膜过滤
                # 两者做按位与操作
                hsv_backproj &= mask

[PRE27]
 # 设置迭代终止条件
                term_crit = (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 10, 1)

[PRE28]
 # 用 CAMShift 更新目标位置
                track_box, self.track_window = cv2.CamShift(hsv_backproj, self.track_window, term_crit)

[PRE29]
 # 在画面上绘制椭圆框
                cv2.ellipse(vis, track_box, (0, 255, 0), 2)
            # 实时显示处理结果
            cv2.imshow('Object Tracker', vis)

[PRE30]
 # 检测 Esc 键退出循环
            c = cv2.waitKey(5)
            if c == 27:
                break

[PRE31]
 # 释放所有窗口资源
        cv2.destroyAllWindows()

[PRE32]
if __name__ == '__main__':
    # 程序入口启动跟踪器
    ObjectTracker().start_tracking()
