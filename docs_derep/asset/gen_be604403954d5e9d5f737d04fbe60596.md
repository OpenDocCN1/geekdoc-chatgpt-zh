def decrypt_key_with_private_key(private_key, encrypted_key):

        encrypted_key,

        padding.OAEP(

            mgf=padding.MGF1(algorithm=hashes.SHA256()),

            algorithm=hashes.SHA256(),

            label=None

        )

    )

    return key
