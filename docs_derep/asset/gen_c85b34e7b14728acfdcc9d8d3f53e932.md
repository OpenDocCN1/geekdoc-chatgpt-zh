想用人类评分来引导模型输出更符合预期的回答，核心就是把对话拆成输入、目标和分数，然后照常规语言模型的方式做前向、算损失、反向、更新。下面是一个可以直接跑通的简化流程，先看训练循环里发生了什么，再看环境怎么搭、数据怎么准备。

        input_batch, target_batch = input_batch.to(device), target_batch.to(device) # 把输入和目标都搬到当前设备上
        outputs = model(input_batch) # 让模型做一次前向推理
        logits = outputs.logits # 取出前向结果中的 logits
        loss = criterion(logits.view(-1, len(vocab)), target_batch.view(-1)) # 展平后计算交叉熵损失
        loss.backward() # 反向传播得到梯度
        optimizer.step() # 用优化器刷新参数

    if (epoch + 1) % 100 == 0: # 训练 100 个轮次打印一次
        print(f'Epoch: {epoch+1:04d}, cost={loss:.6f}')

class RLHFDataset(Dataset): # 创建一个数据集类，继承自 PyTorch 的 Dataset

    def __init__(self, data, tokenizer, vocab): # 类的初始化函数

        self.tokenizer = tokenizer # 分词器，用于将文本数据转换为模型可以理解的形式

        self.vocab = vocab # 词汇表，存储所有可能的词汇，以便模型能理解

        # 处理输入数据，将其分解为输入数据、目标数据和评分数据

        self.input_data, self.target_data, self.scores = self.process_data(data)
