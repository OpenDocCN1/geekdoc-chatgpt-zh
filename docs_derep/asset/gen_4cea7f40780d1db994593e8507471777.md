带KV缓存的GPT-2训练与推理全流程拆解

这套实现把原生的GPT-2改造为支持增量缓存的版本，并配套了基于SentencePiece的分词器、中文情感语料的采样数据集以及带余弦退火的训练循环。核心思路是在首次完整编码后把每一层的键值对保留下来，后续只对新来的一步做计算，显著降低重复计算开销。

缓存初始化与重置
模型在开始新一轮生成前会把历史键值清空，形状为层数×头数×0×2*head_size的零张量，作为空缓存的占位符。

def reset_kv_cache(self):

首次填充阶段
当缓存为空时，整段输入先被嵌入并加上位置编码，经过dropout后逐层送入GPTBlock。每个Block返回当前步的键值对，收集后沿层维度拼接，成为初始的past_key_values。

import sentencepiece as spm

class Tokenizer:

    def __init__(self, spm_path = './vocab_new/spm_model.model'):

        super().__init__()

        self.sp = spm.SentencePieceProcessor()

        self.sp.Load(spm_path)

        self.end_id = 4

    def encode(self, text):

        token = self.sp.EncodeAsIds(text)

        return token

    def decode(self, token):

        _text = self.sp.DecodeIds(token)

        return (_text)

    def vocab_size(self):

        return len(self.sp)

import config

tokenizer_emo = config.Tokenizer()

token_list = []

with open("../dataset/ChnSentiCorp.txt", mode="r", encoding="UTF-8") as emotion_file:

    for line in tqdm(emotion_file.readlines()):

        line = line.strip().split(",")

        text = "".join(line[1:]) + '<|end_of_sentence|>'

        token = tokenizer_emo.encode(text)

        for id in token:

            token_list.append(id)

token_list = torch.tensor(token_list * 4)

class TextSamplerDataset(torch.utils.data.Dataset):

    def __init__(self, data = token_list, seq_len = 48):

        super().__init__()

        self.data = data

        self.seq_len = seq_len

    def __getitem__(self, index):

        rand_start = torch.randint(0, self.data.size(0) - self.seq_len, (1,))

        full_seq = self.data[rand_start : rand_start + self.seq_len + 1].long()

        return full_seq[:-1],full_seq[1:]

    def __len__(self):

        return self.data.size(0) // self.seq_len
