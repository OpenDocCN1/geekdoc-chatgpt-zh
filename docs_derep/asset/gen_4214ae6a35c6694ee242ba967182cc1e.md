import bm25_RAG_demo

@mcp.tool()

def customer_chat(query: str) -> dict:

    """
    该方法充当智能客服的决策中枢，专门承接普通用户发起的多样化咨询任务。

    其运行方式类似于一位资深客服顾问，会先接收用户的具体提问、相关的业务场景信息以及用户的基础标识，

    再基于内部知识库、业务规则引擎或对接的外部专业接口，对咨询内容进行理解分析并给出精准答复。

    同时会全程保存对话数据，为后续的服务质量评估、问题溯源和知识库迭代提供依据，帮助实现高效、准确且具个性化的咨询体验。

    参数：
        query (str): 用户输入的咨询原文，需明确说明想获取的内容，例如产品功能、办理流程、政策法规等。

    返回：
        dict: 以清晰结构返回的应答结果，包含回复文本及配套信息，便于前端展示和用户阅读。
    """

    # 以下演示了检索与生成的基本流程，生产环境会接入真实的知识库接口和规则引擎

    with open("phone.txt", "r", encoding="utf-8") as f:

        content = f.read()

    informations = bm25_RAG_demo.split_text_by_word_count(content, max_word_count=128)

    bm25_results = bm25_RAG_demo.get_top_n_sim_text(query, informations)

    response = bm25_RAG_demo.customer_chat(query, bm25_results)

    return {

        "answer": response,

        "response_time": datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S"),

        "status": "success"

    }
