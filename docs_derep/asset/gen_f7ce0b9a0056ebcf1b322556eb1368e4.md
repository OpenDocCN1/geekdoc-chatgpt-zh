class TrieNode:

    def __init__(self):

        CHAR_COUNT=11

        self.isLeaf=False

        self.url=None

        self.child=[None]*CHAR_COUNT    # 分支数组预留 11 个位置

        i=0

        while i<CHAR_COUNT:

            self.child[i]=None

            i +=1

class DNSCache:

    def __init__(self):

        self.CHAR_COUNT=11    # 数字 0-9 加点号共 11 种字符
        self.root=TrieNode() # 从根节点开始构建前缀树

    def insert(self,ip,url):

        # 计算待写入 IP 的字符总数
        lens = len(ip)

        pCrawl = self.root

        level=0

        while level<lens:

            # 将当前字符映射为分支下标
            index = getIndexFromChar(ip[level])

            # 缺失分支时即时创建
            if pCrawl.child[index] ==None:

                pCrawl.child[index] = TrieNode()

            # 进入下一层节点
            pCrawl = pCrawl.child[index]

            # 标记为终止节点并写入 URL
            pCrawl.isLeaf = True

            pCrawl.url = url

            level +=1

    # 根据 IP 查表获取 URL
    def searchDNSCache(self,ip):

        pCrawl = self.root

        lens = len(ip)

        # 按字符逐层向下匹配
        level=0

        while level<lens:

            index = getIndexFromChar(ip[level])

            if pCrawl.child[index] ==None:

                return None

            pCrawl = pCrawl.child[index]

            level +=1

        # 到达末尾检查是否为有效记录
        if pCrawl!=None and pCrawl.isLeaf:

            return pCrawl.url

        return  None

"""
**** 功能概述: 在整数数组中寻找第 k 小的元素
**** 输入说明: array 为整数序列; low 为起始下标; high 为结束下标; k 为期望排名
**** 输出说明: 返回数组中排名第 k 的数值
"""

def findSmallK(array,low,high,k):

    i = low

    j = high

    splitElem = array[i]

    # 划分左右两部分，使左侧 <= 枢轴，右侧 > 枢轴
    while i < j:

        while i < j and array[j] >= splitElem:

            j -= 1

        if i < j:

            array[i] = array[j]

            i += 1

        while i < j and array[i] <= splitElem:

            i += 1

        if i < j:

            array[j] = array[i]

    array[i] = splitElem

    # 枢轴在当前区间中的相对位置
    subArrayIndex = i - low

    # 恰好命中目标排名
    if subArrayIndex == k-1:

        return array[i]

    # 目标在左侧子区间
    elif subArrayIndex > k-1:

        return findSmallK(array,low,i-1,k)

    # 目标在右侧子区间，需调整 k
    else:

        return findSmallK(array,i+1,high,k-(i-low)-1)
