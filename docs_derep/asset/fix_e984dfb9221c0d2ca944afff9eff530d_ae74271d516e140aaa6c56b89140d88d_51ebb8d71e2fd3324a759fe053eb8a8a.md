# 训练到推理：边缘轻量与服务器提速实战

模型只在训练脚本里跑通还不够，能在目标硬件上稳定、可预期地给出结果才是闭环。下面用两条常见路径演示完整链路，思路可直接迁移到自己的项目。

先看面向移动端的轻量方案。选用 Keras 建一个极简序列网络，输入是 256 维向量，先过 128 单元的 ReLU，再接 10 类 Softmax。用随机样本做一次模拟训练，优化器选 Adam，损失用稀疏分类交叉熵，批大小 32、迭代一轮即可。训练结束后借助 TFLite 转换器把 Keras 结构导出为 .tflite 并写入磁盘。随后用解释器加载文件、分配运行时内存、查询输入输出索引，准备一个 1×256 的随机浮点样本写入输入区，计时执行一次前向，再读回结果并打印耗时，就能直观感受到转换后的延迟表现。

再看服务端追求吞吐的路线。以 PyTorch 为例，先定义一个双卷积的骨架：首层把 3 通道提升到 32 通道，3×3 卷积保持填充，ReLU 激活后做 2×2 最大池化；第二层把 32 通道提升到 64 通道，处理方式一致再下采样一次；展平后接 128 维线性层，再到 10 维输出，中间激活仍用 ReLU。训练时在指定设备上循环数据，每处理百个小批次就输出一次累计损失，结束提示完成。评测时关闭梯度统计准确率。训练收敛后把权重固化为 ONNX，选用 11 版算子集并开启常量折叠，随后用 TensorRT 读取 ONNX，通过解析器载入网络，工作区设为 1GB 并开启半精度，生成可序列化的推理引擎并落盘。

# 定义一个简单的卷积神经网络模型
class SimpleCNN(nn.Module):
    def __init__(self):
        super(SimpleCNN, self).__init__()
        # 卷积层
        # 输入通道为 3，输出通道为 32，卷积核为 3*3
        self.conv1=nn.Conv2d(3, 32, kernel_size=3, padding=1)
        # 输入通道为 32，输出通道为 64，卷积核为 3*3
        self.conv2=nn.Conv2d(32, 64, kernel_size=3, padding=1)
        # 全连接层
        self.fc1=nn.Linear(64*8*8, 128)  # 输入特征为 64*8*8，输出特征为 128
        self.fc2=nn.Linear(128, 10)  # 输入特征为 128，输出特征为 10（类别数）
        # 激活函数
        self.relu=nn.ReLU()
        # 池化层
        self.pool=nn.MaxPool2d(2, 2)  # 2*2 最大池化
    def forward(self, x):
        x=self.relu(self.conv1(x))  # 卷积 1+ReLU
        x=self.pool(x)  # 池化 1
        x=self.relu(self.conv2(x))  # 卷积 2+ReLU
        x=self.pool(x)  # 池化 2
        x=x.view(-1, 64*8*8)  # 展平
        x=self.relu(self.fc1(x))  # 全连接 1+ReLU
        x=self.fc2(x)  # 全连接 2
        return x
# 训练函数
def train_model(model, trainloader, criterion, optimizer, device, epochs=5):
    model.to(device)
    model.train()
    for epoch in range(epochs):
        running_loss=0.0
        for i, data in enumerate(trainloader, 0):
            inputs, labels=data[0].to(device), data[1].to(device)
            optimizer.zero_grad()
            outputs=model(inputs)
            loss=criterion(outputs, labels)
            loss.backward()
            optimizer.step()
            running_loss += loss.item()
            if i % 100 == 99:  # 每 100 批输出一次
                print(f"[Epoch {epoch+1}, Batch {i+1}] 损失: {running_loss/100:.3f}")
                running_loss=0.0
    print("训练完成")
# 测试函数
def test_model(model, testloader, device):
    model.to(device)
    model.eval()
    correct=0
    total=0
    with torch.no_grad():
        for data in testloader:
            images, labels=data[0].to(device), data[1].to(device)
            outputs=model(images)
            _, predicted=torch.max(outputs.data, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()
    print(f"测试准确率: {100*correct/total:.2f}%")
# 将 PyTorch 模型导出为 ONNX 格式
def export_onnx(model, device, onnx_file_path='model.onnx'):
    model.to(device)
    model.eval()
    dummy_input=torch.randn(1, 3, 32, 32).to(device)
    torch.onnx.export(model, dummy_input, onnx_file_path,
                      export_params=True,
                      opset_version=11,
                      do_constant_folding=True,
                      input_names=['input'],
                      output_names=['output'])
    print(f"模型已被导出为 ONNX 格式, 路径: {onnx_file_path}")
# 使用 TensorRT 构建优化的推理引擎
def build_engine(onnx_file_path, engine_file_path='model.trt'):
    TRT_LOGGER=trt.Logger(trt.Logger.WARNING)
    builder=trt.Builder(TRT_LOGGER)
    network=builder.create_network(1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH))
    parser=trt.OnnxParser(network, TRT_LOGGER)
    with open(onnx_file_path, 'rb') as model:
        if not parser.parse(model.read()):
            print('ERROR: Failed to parse the ONNX file.')
            for error in range(parser.num_errors):
                print(parser.get_error(error))
            return None
    config=builder.create_builder_config()
    config.max_workspace_size=1 << 30  # 1GB
    config.set_flag(trt.BuilderFlag.FP16)  # 启用 FP16 模式
    engine=builder.build_engine(network, config)
    if engine is None:
        print("Failed to create the engine")
        return None
    with open(engine_file_path, 'wb') as f:
        f.write(engine.serialize())
    print(f"TensorRT 引擎已被构建并保存, 路径: {engine_file_path}")
    return engine

推理时从引擎实例化一个执行上下文，并在显存上分配输入输出缓存，用 CUDA 流实现输入的异步搬运，触发异步前向计算，完成后把结果搬回主机供后续使用。这种显存与流并行的写法是高吞吐部署的常用做法。

def infer(engine, inputs):
    TRT_LOGGER = trt.Logger(trt.Logger.WARNING)
    # 创建执行上下文
    context = engine.create_execution_context()
    # 分配 GPU 内存
    d_input = cuda.mem_alloc(1 * inputs.nbytes)
    # 假设输出为 10 个类别，每个类别占用 4 字节的内存空间
    d_output = cuda.mem_alloc(1 * 10 * 4)
    bindings = [int(d_input), int(d_output)]
    stream = cuda.Stream()
    # 将输入数据复制到 GPU 上
    cuda.memcpy_htod_async(d_input, inputs, stream)
    # 执行推理
    context.execute_async_v2(bindings=bindings, stream_handle=stream.handle)
    # 创建输出数组
    output = np.empty(10, dtype=np.float32)
    # 将输出数据复制回 CPU
