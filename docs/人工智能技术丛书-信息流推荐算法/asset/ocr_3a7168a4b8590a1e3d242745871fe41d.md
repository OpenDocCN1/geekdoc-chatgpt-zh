$$\frac{\partial f(\boldsymbol{x}_i)}{\partial \theta_j} = - \frac{1}{(1+e^{-\boldsymbol{\theta}^T \boldsymbol{x}_i})^2} e^{-\boldsymbol{\theta}^T \boldsymbol{x}_i} x_{ij}
= - \frac{1+e^{-\boldsymbol{\theta}^T \boldsymbol{x}_i} - 1}{(1+e^{-\boldsymbol{\theta}^T \boldsymbol{x}_i})^2} x_{ij} \quad (5-5)
= -x_{ij} \frac{1}{1+e^{-\boldsymbol{\theta}^T \boldsymbol{x}_i}} \left( 1 - \frac{1}{1+e^{-\boldsymbol{\theta}^T \boldsymbol{x}_i}} \right)
= -x_{ij} f(\boldsymbol{x}_i) (1 - f(\boldsymbol{x}_i))$$
