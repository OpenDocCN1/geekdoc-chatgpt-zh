```python
Print 'ndarray 数组的转置和轴对换'
k = numpy.arange(9) # [0,1,...,8]
m = k.reshape((3,3)) # 改变数组的 shape，复制成 2 维的数组，每个维度长度为 3
print k # [0 1 2 3 4 5 6 7 8]
print m # [[0 1 2] [3 4 5] [6 7 8]]
