```
import sensor, image

sensor.reset()
sensor.set_pixformat(sensor.RGB565)
sensor.set_framesize(sensor.QVGA)
sensor.skip_frames(30)
sensor.set_auto_gain(False) #必须关闭它以防止图像冲刷...
while(True) :
    img = sensor.snapshot()
    img.lens_corr(1.3) #图像的畸变校正，经测试 1.3 效果可以完成需求
    for code in img.find_qrcodes() :
        print(code)
```
