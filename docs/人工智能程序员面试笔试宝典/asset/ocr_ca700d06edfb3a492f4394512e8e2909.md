def getMaxSubStr(str1,str2):
    len1 = len(str1)
    len2 = len(str2)
    sb = ""
    maxs = 0
    maxI = 0
    # 申请新的空间来记录公共子串长度信息
    M = [[0 for i in range(len1+1)] for j in range(len2+1)]
    # 通过利用通用公式写新建立的二维数组（公共子串的长度信息）
    i = 0
    while i < len1+1:
        M[i][0] = 0
        i += 1
    j = 0
    while j < len2+1:
        M[0][j] = 0
        j += 1
    i = 1
    while i < len1+1:
        j = 1
        while j < len2+1:
            if list(str1)[i-1] == list(str2)[j-1]:
                M[i][j] = M[i-1][j-1] + 1
                if M[i][j] > maxs:
                    maxs = M[i][j]
                    maxI = i
            else:
                M[i][j] = 0
            j += 1
        i += 1
    # 找出公共子串
    i = maxI - maxs
    while i < maxI:
        sb += "".join(list(str1)[i])
        i += 1
    return sb

if __name__ == "__main__":
    str1 = "abcade"
    str2 = "dcadde"
    print(getMaxSubStr(str1,str2))

# 算法性能分析：

由于这种方法使用了二重循环分别遍历两个字符数组，因此时间复杂度为O(m×n)（其中，m和n分别为两个字符串的长度）。此外，由于这种方法申请了一个m×n的二维数组，因此，算法的空间复杂度也为O(m×n)。很显然，这种方法的主要缺点为申请了m×n个额外的存储空间。
