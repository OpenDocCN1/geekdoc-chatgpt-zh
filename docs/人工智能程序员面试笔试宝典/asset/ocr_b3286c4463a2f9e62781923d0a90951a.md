class EditDistance:
    def mins(self,a,b,c):
        tmp = a if a < b else b
        return tmp if tmp < c else c
    # 参数 replaceWight 用来表示替换操作与插入删除操作的倍数
    def edit(self,s1,s2,replaceWight):
        # 两个空串的编辑距离为 0
        if s1 == None and s2 == None:
            return 0
        # 其中一个为空串，那么编辑距离为另一个字符串的长度
        if s1 == None:
            return len(s2)
        if s2 == None:
            return len(s1)
        # 申请二维数组保存中间的计算结果
        D = [[None]*(len(s2)+1) for i in range(len(s1)+1)]
        i = 0
        while i < len(s1)+1:
            D[i][0] = i
            i += 1
        i = 0
        while i < len(s2)+1:
            D[0][i] = i
            i += 1
        i = 1
        while i < len(s1)+1:
            j = 1
            while j < len(s2)+1:
                if list(s1)[i-1] == list(s2)[j-1]:
                    D[i][j] = self.mins(D[i-1][j]+1,D[i][j-1]+1,D[i-1][j-1])
                else:
                    D[i][j] = min(D[i-1][j]+1,D[i][j-1]+1,D[i-1][j-1]+replaceWight)
                j += 1
            i += 1
        print("-----------------------------")
        i = 0
        while i < len(s1)+1:
            j = 0
            while j < len(s2)+1:
                print(D[i][j],end=' ')
                j += 1
            print()
            i += 1
        i += 1
        print("-----------------------------")
        dis = D[len(s1)][len(s2)]
        return dis
