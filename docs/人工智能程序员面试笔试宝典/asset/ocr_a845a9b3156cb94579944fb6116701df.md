def FindlastK(head, k):
    if head is None or head.next is None:
        return
    slow = LNode()
    fast = LNode()
    slow = head.next
    fast = head.next
    i = 0
    while i < k and fast is not None:
        fast = fast.next
        i += 1
    if i < k:
        return None
    while fast is not None:
        slow = slow.next
        fast = fast.next
    return slow
