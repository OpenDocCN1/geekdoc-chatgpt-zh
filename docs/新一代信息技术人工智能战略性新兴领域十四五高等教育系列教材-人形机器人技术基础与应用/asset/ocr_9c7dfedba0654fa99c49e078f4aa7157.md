```
// 发布高层位置速度信息
ChannelPublisher <unitree_go ::msg::dds_:: SportModeState_ >high_state_puber ( TOPIC_HIGHSTATE );
high_state_puber.InitChannel ();
// 接收控制指令
ChannelSubscriber <unitree_go ::msg::dds_::LowCmd_ > low_cmd_suber ( TOPIC_LOWCMD );
// 初始化完成后，如果接收到 LowCmd_ 指令就会进入 HighStateHander 回调函数
low_cmd_suber.InitChannel ( HighStateHander );
while ( 1 ) {
    // 进行一步仿真计算，计算结果会存入 mj_data
    mj_step ( mj_model , mj_data );
    // 获取仿真数据
    GetLowState ();
    GetHighState ();
    // 发布仿真数据
    low_state_puber ->Write ( low_state );
    high_state_puber ->Write ( high_state );
}
// 释放内存
mj_deleteData ( mj_data );
mj_deleteModel ( mj_model );
return 0;
}
```

对于仿真数据的获取，仿真传感器的数据都储存于 mj_data->sensordata 中，可以按照传感器在机器人的 MJCF 描述文件中的顺序，从中依次取出对应的传感器数据，然后将其存入 Unitree SDK2 定义好的数据结构中，再使用 Channel Publisher 来发布数据。实现方法如下：
