```
//批量获取图书信息
getBooks: function (isbns) {
    return new Promise((res, rej) => {
        if (isbns.length == 0) {
            rej("error");
            return;
        }

        let db = wx.cloud.database();
        const _ = db.command
        let books = db.collection("book");
        books.where({
            isbn: _.in(isbns)
        }).get({
            success: function (data) {
                res(data.data);
            }
        });
    });
},
//获取用户房信息
getUserHome: function(openId) {
    return new Promise((res, rej) => {
        if (openId.length == 0) {
            rej("error");
            return;
        }

        let db = wx.cloud.database();
        let homes = db.collection("home");
        homes.where({
            _openid: openId
        }).get({
            success: function (data) {
                res(data.data);
            }
        });
    });
},
```
