getComments:function(isbn){
return new Promise((res,rej)=>{
    let db = wx.cloud.database();
    let comments = db.collection("comment");
    comments.where({
        isbn:isbn
    }).get({
        success:function(comments){
            res(comments);
        },
        fail:function(){
            wx.showToast({
                title: '书评不翼而飞了!! ',
            })
            rej();
        }
    });
});
