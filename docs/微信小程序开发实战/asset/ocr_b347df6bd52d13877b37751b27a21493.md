```
login:function(){
    return new Promise((resolve, reject) => {
        wx.cloud.callFunction({
            name: 'login',
            success: function (res) {
                resolve(res);
            },
            fail: function (error) {
                wx.showToast({
                    title: '登录失败',
                })
                reject(error);
            }
        });
    });
},
```
