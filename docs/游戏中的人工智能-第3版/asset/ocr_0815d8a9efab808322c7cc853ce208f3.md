```
class PlanningDecomposer extends Decomposer:
    graph: Graph
    heuristic: function(GraphNode, GraphNode) -> float

    function decompose(character: Kinematic, goal: Goal) -> Goal:
        # 首先，可以量化当前的位置和目标
        # 并转换为图形中的节点
        start: GraphNode = graph.getNode(kinematic.position)
        end: GraphNode = graph.getNode(goal.position)

        # 如果起始节点和终点是相等的，则无须规划路径
        if startNode == endNode:
            return goal

        # 否则，按以下方式规划路线
        path = pathfindAStar(graph, start, end, heuristic)

        # 获取路径中的第一个节点并对它进行局部化
        firstNode: GraphNode = path[0].asNode
        position: Vector = graph.getPosition(firstNode)

        # 更新该目标并返回
        goal.position = position
        goal.hasPosition = true
        return goal
```
