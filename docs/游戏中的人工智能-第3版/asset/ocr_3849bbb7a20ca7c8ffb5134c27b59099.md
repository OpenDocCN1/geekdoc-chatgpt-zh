```
# 步骤 4, 按每个位置计算
function flow(x: int, y: int, srcMat: float[int][int]):
    dirns = [(0, 1, 0), (1, 0, 1), (2, 0, -1), (3, -1, 0)]
    # 对每个相邻位置应用水流
    for i, dx, dy in dirns:
        waterFlow = waterFlow[x][y][i]
        prop = waterFlow / totalFlow[x][y]
        materialFlow = prop * srcMat[x][y]

        water[x][y] -= waterFlow
        water[x + dx][y + dy] += waterFlow
        material[x][y] -= materialFlow
        material[x + dx][y + dy] += materialFlow
```

```
function evaporate(x: int, y: int):
    water[x][y] *= (1 - EVAPORATION)
```
