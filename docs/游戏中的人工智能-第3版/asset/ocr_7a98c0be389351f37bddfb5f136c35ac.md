endNodeRecord.connection = connection
endNodeRecord.estimatedTotalCost = endNodeCost +
    endNodeHeuristic
# 将它添加到开放列表
if not open.contains(endNode):
    open += endNodeRecord
# 至此已经完成对当前节点的连接检查
# 所以可将它添加到封闭列表
# 将从开放列表中删除它
open -= current
closed += current
# 跳转到这里表示已经找到了目标节点
# 或者没有更多的节点需要搜索
if current.node != goal:
    # 已经遍历了所有的节点但是是未发现目标
    # 因此没有解决方案
    return null
else:
    # 编译路径中连接的列表
    path = []
    # 沿着路径回溯，累积连接
    while current.node != start:
        path += current.connection
        current = current.connection.getFromNode()
    # 反转路径并返回该路径
    return reverse(path)
