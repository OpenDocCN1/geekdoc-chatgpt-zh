# 窗口大小 + 1, 即 N 值
nValue: int

# 使用预测器注册一组动作
# 更新其数据
# 假设在这一组动作中已经包含明确的 nValue 个元素
function registerSequence(actions: any[]):
    # 将序列拆分为 key 键和 value 值
    key = actions[0..nValue]
    value = actions[nValue]

    # 确认已经获得存储
    if not key in data:
        keyData = data[key] = new KeyDataRecord()
    else:
        keyData = data[key]

    # 添加到总计以及该值的次数统计中
    keyData.counts[value] += 1
    keyData.total += 1

# 从给定的一个结果中获取下一次很可能采取的行动
# 假定动作有 nValue - 1 个元素在其中
# 即窗口大小
function getMostLikely(actions: any[]) -> any:
    # 获取 key 数据
    keyData = data[actions]

    # 找到最高的概率
    highestValue = 0
    bestAction = null

    # 获取存储的动作列表
    actions = keyData.counts.getKeys()

    # 遍历动作列表中的每一个动作
    for action in actions:
        # 检查最高概率的值
        if keyData.counts[action] > highestValue:
            # 存储该动作
            highestValue = keyData.counts[action]
