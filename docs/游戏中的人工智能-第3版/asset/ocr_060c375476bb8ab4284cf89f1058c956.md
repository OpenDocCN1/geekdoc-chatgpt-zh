function rain(x: int, y: int):

# 步骤 2, 按每个位置计算
function calculateFlow(x: int, y: int):
    dirns = [(0, 1, 0), (1, 0, 1), (2, 0, -1), (3, -1, 0)]
    # 流到每个相邻位置
    totalFlow[x][y] = 0
    for i, dx, dy in dirns:
        dh = elevation[x][y] - elevation[x + dx][y + dy]
        dw = water[x][y] - water[x + dx][y + dy]
        waterFlow[x][y][i] += FLOW_RATE * (dh + dw)
        if waterFlow[x][y][i] < 0:
            waterFlow[x][y][i] = 0
        totalFlow[x][y] += waterFlow[x][y][i]

    # 没有更多的水了
    if totalFlow[x][y] > water[x][y]:
        prop = water[x][y] / totalFlow[x][y]
        for i in 0..4:
            waterFlow[x][y][i] *= prop

    # 全面流过
    hOut = waterFlow[x][y][0] - waterFlow[x][y][3]
    hIn = waterFlow[x+1][y][3] - waterFlow[x-1][y][0]
    vOut = waterFlow[x][y][1] - waterFlow[x][y][2]
    vIn = waterFlow[x][y+1][2] - waterFlow[x][y-1][1] .
    waterVelocity[x][y] = Vector2(hOut + hIn, vOut + vIn)

# 步骤 3, 按每个位置计算
function erodeDeposit(x: int, y: int):
    waterSpeed = waterVelocity[x][y].magnitude()

    if waterSpeed > PICK_THRESHOLD:
        pick = waterSpeed * SOLUBILITY
        material[x][y] += pick
        elevation[x][y] -= pick
