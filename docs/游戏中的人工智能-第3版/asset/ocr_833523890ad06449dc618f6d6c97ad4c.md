```
```python
# 从存储中获取当前 Q 值
Q = store.getQValue(state, action)

# 从新状态获取最佳动作的 Q 值
maxQ = store.getQValue(
    newState, store.getBestAction(newState))

# 执行 Q-Learning
Q = (1 - alpha) * Q + alpha * (reward + gamma * maxQ)

# 存储新 Q 值
store.storeQValue(state, action, Q)

# 更新该状态
state = newState
