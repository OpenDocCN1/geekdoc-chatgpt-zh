```
# 检查是否有转换需要激发
if triggeredTransition:
    # 重置计时器
    currentTime = resetTime

    # 使用矩阵和状态向量相乘
    matrix = triggeredTransition.getMatrix()
    state = matrix * state

    # 返回已触发转换的动作列表
    return triggeredTransition.getActions()
else:
    # 否则检查计时器
    currentTime -= 1

    if currentTime <= 0:
        # 执行默认转换
        state = defaultTransitionMatrix * state
        currentTime = resetTime

    # 不返回任何动作，因为没有触发任何转换
    return []
```

    function isTriggered()
    function getMatrix()
    function getActions()
