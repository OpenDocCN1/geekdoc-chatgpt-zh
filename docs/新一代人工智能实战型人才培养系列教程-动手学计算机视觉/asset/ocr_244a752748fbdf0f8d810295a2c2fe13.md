            # 更新当前最匹配的特征点
            maxmatch[vec1_index] = distance
            # 记录当前特征点对的距离
            max_index[vec1_index] = vec2_index

        elif distance < secmatch[vec1_index]:
            # 更新第二近特征点对的距离
