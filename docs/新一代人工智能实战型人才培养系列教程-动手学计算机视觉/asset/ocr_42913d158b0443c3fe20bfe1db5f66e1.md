def normalize(pts, T=None):
    """
    对点集进行归一化
    """
    # 如果 T 参数为空
    if T is None:
        # 求点集的平均值
        u = np.mean(pts, 0)
        # 求点集中每个点与原点之间的距离之和
        d = np.sum(np.sqrt(np.sum(np.power(pts, 2), 1)))
        # 计算归一化矩阵
        T = np.array([
            [np.sqrt(2) / d, 0, -(np.sqrt(2) / d * u[0])],
            [0, np.sqrt(2) / d, -(np.sqrt(2) / d * u[1])],
            [0, 0, 1]
        ])
    # 对点集进行归一化
    return homoco_pts_2_euco_pts(np.matmul(T,
        euco_pts_2_homoco_pts(pts).T).T), T
