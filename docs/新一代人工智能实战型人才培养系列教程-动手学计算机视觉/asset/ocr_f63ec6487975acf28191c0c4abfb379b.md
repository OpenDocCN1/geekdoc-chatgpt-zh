```
from utils import *
# 设定高斯核的大小与方差
kernel_size = 5
sigma = 1.5

# 导入一幅图像
img = Image.open('lena.jpeg')

# 将其转为灰度图像
img = img.convert('L')
img = np.array(img)

# 构造高斯卷积核
kernel = gaussian_kernel(kernel_size, sigma)

# 与图像进行卷积
smoothed = cv2.filter2D(img, -1, kernel)

# 展示图像
plot_image(img, 'Original image')
plot_image(smoothed, 'Smoothed image')
```
