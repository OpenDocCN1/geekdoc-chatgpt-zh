def binary_search(sorted_list, val):
    """
    二分查找
    :param sorted_list:
    :param val:
    :return:
    """
    low = 0
    high = len(sorted_list) - 1
    while low <= high:
        mid = (low + high) // 2
        if sorted_list[mid] == val:
            return mid
    elif sorted_list[mid] > val:
        high = mid - 1
    else:
        low = mid + 1
    return -1
