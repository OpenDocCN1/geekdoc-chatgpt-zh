```
head_cla = i
# 添加到距离最小的簇头对应的聚类列表中
classes[head_cla].append(members[n])
# 将簇头作为头节点添加到聚类列表中
iter_classes.append(classes)
# 0 个簇头的情况
if dist_min == 1:
    print("本轮没有簇头!")
    break

return iter_classes
```

def show_plt(classes):

    显示分类图

    for i in range(len(classes)):
        centor = classes[i][0]
        for point in classes[i]:
            ax1.plot([centor[0], point[0]], [centor[1], point[1]], c=color[i % 6], marker=icon[i % 5],
                     alpha=0.4)

    plt.show()
