    """
    根据阈值选取簇头节点
    :param r: 轮数
    :param nodes: 节点列表
    :param flags: 选择标志
    :param P: 比例因子
    :return: 簇头列表 heads, 簇成员列表 members
    """
    # 阈值函数 Tn 使用 LEACH 计算
    P = 0.05 * (100 / len(nodes))
    Tn = P / (1 - P * (r % (1 / P)))
    # print("阈值为:", Tn)
    # 簇头列表

    # 簇成员列表

    # 本轮簇头数
