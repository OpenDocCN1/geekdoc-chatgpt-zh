        result=img_object.inference(img,None)
        frame = img_object.base64_to_image(result["result_image"])
        #图像显示
        cv.imshow('frame',frame)
        key=cv.waitKey(1)
        if key==ord('q'):
            break
    cap.release()
    cv.destroyAllWindows()
else :
