void setup(){
pinMode(13,OUTPUT);// 设置数字端口 13 为输出模式。
}
void loop(){
int val;// 定义整型变量 val。
val=analogRead(0);// 读出模拟端口 A0 的值，赋给变量 val。
if(val>800){// 如果 val>800,
for(int i=0;i<4;i++){// 循环 4 次。
delay(500);// 延时 500ms。
if(val>800){// 如果 val>800,
digitalWrite(13,1);// 数字端口 13 输出高电平。
delay(5000);// 延时 5000ms。
digitalWrite(13,0);// 数字端口 13 输出低电平。
val=0;// 变量 val 清零。
}
}
}
}
}
