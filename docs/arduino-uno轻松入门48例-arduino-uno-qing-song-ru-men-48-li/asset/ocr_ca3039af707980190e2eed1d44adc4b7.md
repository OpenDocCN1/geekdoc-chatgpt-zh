```
void setup(){
for(int i=0;i<14;i++){// 循环执行，从 i=0 开始，到 i=13 结束。
pinMode(i, OUTPUT);// 设置数字端口 0 ~ 13 为输出模式。
}
}
void loop(){
for(int i=0;i<14;i++){// 循环执行，i=0 ~ 13。
digitalWrite(i,1);// 数字端口输出高电平。
delay(100);// 延时 100ms。
digitalWrite(i,0);// 数字端口输出低电平。
}
}
```
