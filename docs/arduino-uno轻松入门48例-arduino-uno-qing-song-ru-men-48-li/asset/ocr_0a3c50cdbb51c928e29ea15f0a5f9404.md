LiquidCrystal lcd(3,4,5,8,9,10,11);
// 创建 LiquidCrystal(rs,rw,enable,d4,d5,d6,d7) 类实例 lcd。
void setup(){
  lcd.clear();// 清除屏幕。
  lcd.begin(16,2);// 设定显示屏尺寸。
}
void loop(){
  S0=(S0+1)%10;
  if(S0==0){
    S1=(S1+1)%6;
    if(S1==0){
      M0=(M0+1)%10;
      if(M0==0){
        M1=(M1+1)%6;
        if(M1==0){
          H0=(H0+1)%10;
          if(H0==0){
            H1=(H1+1)%10;
          }
        }
      }
    }
  }
}
lcd.setCursor(0,0);// 设置光标位置为 (0,0)。
lcd.print("JISHI T=");// 输出字符串。
