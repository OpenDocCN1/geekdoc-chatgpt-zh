def NET_G(ngf=64, block_n=6, downsampling_n=2, upsampling_n=2, image_size = 256):
    """
    生成网络
    采用resnet结构

    [第一层] 大小为7的卷积核 通道数量 3->ngf
    [下采样] 大小为3的卷积核 步长为2 每层通道数量倍增
    [残差网络] 9个block叠加
    [上采样]
    [最后一层] 通道数量变回3
    """
