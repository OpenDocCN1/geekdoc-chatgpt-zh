```
# X: 储存3*3像素格的灰度值
# Y: 储存中心像素格的色彩彩值
# X, Y构成了从3*3像素格的灰度值到中心像素格色彩值的映射
def read_style_image(file_name):
    img = rgb2lab(io.imread(file_name))
    w, h = img.shape[:2]
    X = []
    Y = []
    for x in range(size, w - size):
        for y in range(size, h - size):
            X.append(img[x - size : x + size + 1, y - size : y + size + 1,
0].reshape(-1))
            Y.append(img[x, y, 1:])
    return X, Y
```
