import pandas as pd
from sklearn import tree
from sklearn.preprocessing import LabelEncoder
#1.选取特征，并进行预处理

one_hot_feature = ['Outlook', 'Temperature', 'Humidity', 'Windy', 'Play']

lbc = LabelEncoder()
for feature in one_hot_feature:

from sklearn import tree
from sklearn.externals.six import StringIO
import pydotplus
